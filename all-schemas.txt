create materialized view public.inv_stock_analytics_mv as
with
  lot_ranked as (
    select
      lt_1.item_id,
      upper(COALESCE(lt_1.location, 'MAIN'::text)) as location,
      TRIM(
        both
        from
          COALESCE(lt_1.batch_no, ''::text)
      ) as batch_no,
      lt_1.receipt_date,
      lt_1.created_at,
      COALESCE(lt_1.qty_available, 0::numeric) as qty_available,
      row_number() over (
        partition by
          lt_1.item_id,
          (upper(COALESCE(lt_1.location, 'MAIN'::text)))
        order by
          lt_1.receipt_date,
          lt_1.created_at,
          lt_1.id
      ) as rn
    from
      inv_lots lt_1
    where
      COALESCE(lt_1.qty_available, 0::numeric) > 0::numeric
  ),
  lot_agg as (
    select
      lr.item_id,
      lr.location,
      count(*)::integer as batch_count,
      min(lr.receipt_date) as oldest_receipt_at,
      max(
        case
          when lr.rn = 1 then lr.batch_no
          else null::text
        end
      ) as next_batch_no
    from
      lot_ranked lr
    group by
      lr.item_id,
      lr.location
  ),
  issue_30 as (
    select
      l.item_id,
      upper(COALESCE(l.location, 'MAIN'::text)) as location,
      sum(COALESCE(l.qty_out, 0::numeric)) as issue_qty_30
    from
      inv_ledger l
    where
      l.ref_type = 'ISSUE'::text
      and l.created_at >= (now() - '30 days'::interval)
    group by
      l.item_id,
      (upper(COALESCE(l.location, 'MAIN'::text)))
  ),
  pr_first_receipt as (
    select
      l.item_id,
      upper(COALESCE(l.location, 'MAIN'::text)) as location,
      l.ref_no as pr_no,
      min(l.created_at) as first_receipt_at
    from
      inv_ledger l
    where
      l.ref_type = 'PR-RECEIPT'::text
      and COALESCE(l.qty_in, 0::numeric) > 0::numeric
      and l.ref_no is not null
    group by
      l.item_id,
      (upper(COALESCE(l.location, 'MAIN'::text))),
      l.ref_no
  ),
  lead_time as (
    select
      pr.item_id,
      r.location,
      avg(
        GREATEST(
          EXTRACT(
            epoch
            from
              r.first_receipt_at - pr.created_at
          ) / 86400.0,
          0::numeric
        )
      ) as avg_lead_time_days
    from
      inv_purchase_requests pr
      join pr_first_receipt r on r.pr_no = pr.pr_no
      and r.item_id = pr.item_id
    group by
      pr.item_id,
      r.location
  )
select
  s.item_id,
  s.location,
  s.last_movement_at,
  la.oldest_receipt_at,
  case
    when COALESCE(la.oldest_receipt_at, s.last_movement_at) is null then null::integer
    else ceil(
      EXTRACT(
        epoch
        from
          now() - COALESCE(la.oldest_receipt_at, s.last_movement_at)
      ) / 86400.0
    )::integer
  end as ageing_days,
  case
    when COALESCE(la.oldest_receipt_at, s.last_movement_at) is null then 'UNKNOWN'::text
    when ceil(
      EXTRACT(
        epoch
        from
          now() - COALESCE(la.oldest_receipt_at, s.last_movement_at)
      ) / 86400.0
    ) > 60::numeric then 'NON_MOVING'::text
    when ceil(
      EXTRACT(
        epoch
        from
          now() - COALESCE(la.oldest_receipt_at, s.last_movement_at)
      ) / 86400.0
    ) > 30::numeric then 'SLOW_MOVING'::text
    else 'FAST_MOVING'::text
  end as movement_class,
  COALESCE(la.next_batch_no, ''::text) as next_batch_no,
  COALESCE(la.batch_count, 0) as batch_count,
  round(COALESCE(i30.issue_qty_30, 0::numeric) / 30.0, 4) as avg_daily_consumption,
  round(COALESCE(lt.avg_lead_time_days, 7::numeric), 2) as avg_lead_time_days,
  1.2 as safety_factor,
  round(
    COALESCE(lt.avg_lead_time_days, 7::numeric) * (COALESCE(i30.issue_qty_30, 0::numeric) / 30.0) * 1.2,
    2
  ) as minimum_stock_level
from
  inv_stock_mv s
  left join lot_agg la on la.item_id = s.item_id
  and la.location = s.location
  left join issue_30 i30 on i30.item_id = s.item_id
  and i30.location = s.location
  left join lead_time lt on lt.item_id = s.item_id
  and lt.location = s.location;


create materialized view public.inv_stock_mv as
with
  ledger_balance as (
    select
      l.item_id,
      upper(COALESCE(l.location, 'MAIN'::text)) as location,
      sum(
        COALESCE(l.qty_in, 0::numeric) - COALESCE(l.qty_out, 0::numeric)
      ) as qty,
      max(l.created_at) as last_movement_at,
      sum(
        case
          when COALESCE(l.qty_in, 0::numeric) > 0::numeric then COALESCE(l.value, 0::numeric)
          else 0::numeric
        end
      ) as inbound_value,
      sum(
        case
          when COALESCE(l.qty_in, 0::numeric) > 0::numeric then COALESCE(l.qty_in, 0::numeric)
          else 0::numeric
        end
      ) as inbound_qty
    from
      inv_ledger l
    group by
      l.item_id,
      (upper(COALESCE(l.location, 'MAIN'::text)))
  ),
  lot_balance as (
    select
      lt.item_id,
      upper(COALESCE(lt.location, 'MAIN'::text)) as location,
      sum(
        case
          when COALESCE(lt.qty_available, 0::numeric) > 0::numeric then COALESCE(lt.qty_available, 0::numeric)
          else 0::numeric
        end
      ) as lot_qty,
      sum(
        case
          when COALESCE(lt.qty_available, 0::numeric) > 0::numeric then COALESCE(lt.qty_available, 0::numeric) * COALESCE(lt.rate, 0::numeric)
          else 0::numeric
        end
      ) as lot_value
    from
      inv_lots lt
    group by
      lt.item_id,
      (upper(COALESCE(lt.location, 'MAIN'::text)))
  )
select
  i.id as item_id,
  i.item_code,
  i.item_name,
  COALESCE(i.category, ''::text) as category,
  COALESCE(i.department, ''::text) as department,
  COALESCE(i.uom, ''::text) as uom,
  lb.location,
  lb.qty,
  case
    when COALESCE(lotb.lot_qty, 0::numeric) > 0::numeric then round(
      lotb.lot_value / NULLIF(lotb.lot_qty, 0::numeric),
      6
    )
    when COALESCE(lb.inbound_qty, 0::numeric) > 0::numeric then round(
      lb.inbound_value / NULLIF(lb.inbound_qty, 0::numeric),
      6
    )
    else 0::numeric
  end as avg_rate,
  round(
    lb.qty * case
      when COALESCE(lotb.lot_qty, 0::numeric) > 0::numeric then lotb.lot_value / NULLIF(lotb.lot_qty, 0::numeric)
      when COALESCE(lb.inbound_qty, 0::numeric) > 0::numeric then lb.inbound_value / NULLIF(lb.inbound_qty, 0::numeric)
      else 0::numeric
    end,
    2
  ) as value,
  lb.last_movement_at
from
  ledger_balance lb
  join inv_items i on i.id = lb.item_id
  left join lot_balance lotb on lotb.item_id = lb.item_id
  and lotb.location = lb.location
where
  lb.qty <> 0::numeric;

create view public.so_line_billed_qty as
select
  il.so_line_id,
  sum(il.qty) as billed_qty
from
  invoice_lines il
  join invoices i on i.id = il.invoice_id
where
  i.status <> 'CANCELLED'::text
group by
  il.so_line_id;

create view public.v_billing_dispatch_summary as
with
  dispatch_totals as (
    select
      dr.so_line_id,
      COALESCE(sum(dr.dispatch_qty), 0::numeric)::numeric(18, 3) as dispatched_qty
    from
      dispatch_records dr
    where
      dr.so_line_id is not null
    group by
      dr.so_line_id
  ),
  dispatch_latest as (
    select distinct
      on (dr.so_line_id) dr.so_line_id,
      dr.dispatch_no as latest_dispatch_no,
      COALESCE(
        dr.dispatch_date::timestamp without time zone,
        dr.created_at
      ) as latest_dispatch_date,
      dr.transporter,
      dr.vehicle_no,
      dr.lr_no
    from
      dispatch_records dr
    where
      dr.so_line_id is not null
    order by
      dr.so_line_id,
      (
        COALESCE(
          dr.dispatch_date::timestamp without time zone,
          dr.created_at
        )
      ) desc,
      dr.created_at desc,
      dr.id desc
  )
select
  t.so_line_id,
  t.dispatched_qty,
  l.latest_dispatch_no,
  l.latest_dispatch_date,
  l.transporter,
  l.vehicle_no,
  l.lr_no
from
  dispatch_totals t
  left join dispatch_latest l on l.so_line_id = t.so_line_id;


create view public.v_billing_document_register as
with
  base as (
    select
      inv.id::text as id,
      inv.invoice_no,
      inv.invoice_date,
      case
        when upper(COALESCE(inv.document_type, ''::text)) = any (array['INVOICE'::text, 'CHALLAN'::text]) then upper(inv.document_type)
        when COALESCE(inv.invoice_no, ''::text) ~~ 'DC-H/%'::text then 'CHALLAN'::text
        else 'INVOICE'::text
      end as document_type,
      COALESCE(inv.client_code, 'MANUAL'::text) as client_code,
      COALESCE(
        NULLIF(inv.bill_to_name, ''::text),
        NULLIF(inv.client_name, ''::text),
        c.client_name,
        inv.client_code,
        'MANUAL'::text
      ) as client_name,
      COALESCE(inv.status, 'DRAFT'::text) as status,
      COALESCE(inv.billing_mode, ''::text) as billing_mode,
      COALESCE(inv.remarks, ''::text) as remarks,
      COALESCE(inv.subtotal, 0::numeric)::numeric(18, 2) as subtotal,
      COALESCE(
        inv.tax_total,
        GREATEST(
          COALESCE(inv.grand_total, 0::numeric) - COALESCE(inv.subtotal, 0::numeric) - COALESCE(inv.freight, 0::numeric),
          0::numeric
        )
      )::numeric(18, 2) as tax_total,
      COALESCE(inv.grand_total, 0::numeric)::numeric(18, 2) as grand_total,
      inv.created_at,
      inv.posted_at
    from
      invoices inv
      left join clients c on c.client_code = inv.client_code
  ),
  line_meta as (
    select
      il.invoice_id::text as id,
      count(*)::integer as line_count,
      COALESCE(sum(il.qty), 0::numeric)::numeric(18, 3) as total_qty,
      min(
        upper(
          COALESCE(NULLIF(il.source_mode, ''::text), 'UNKNOWN'::text)
        )
      ) as derived_billing_mode
    from
      invoice_lines il
    group by
      il.invoice_id
  ),
  line_preview_raw as (
    select
      s.id,
      s.product_label,
      row_number() over (
        partition by
          s.id
        order by
          s.product_label
      ) as rn
    from
      (
        select distinct
          il.invoice_id::text as id,
          COALESCE(
            NULLIF(il.product_name, ''::text),
            NULLIF(il.product_code, ''::text),
            'Item'::text
          ) as product_label
        from
          invoice_lines il
      ) s
  ),
  line_preview as (
    select
      line_preview_raw.id,
      string_agg(
        line_preview_raw.product_label,
        ', '::text
        order by
          line_preview_raw.product_label
      ) as product_preview
    from
      line_preview_raw
    where
      line_preview_raw.rn <= 2
    group by
      line_preview_raw.id
  )
select
  b.id,
  b.invoice_no,
  b.invoice_date,
  b.document_type,
  b.client_code,
  b.client_name,
  b.status,
  COALESCE(
    NULLIF(b.billing_mode, ''::text),
    lm.derived_billing_mode,
    'UNKNOWN'::text
  ) as billing_mode,
  COALESCE(lm.total_qty, 0::numeric)::numeric(18, 3) as total_qty,
  COALESCE(lm.line_count, 0) as line_count,
  COALESCE(lp.product_preview, ''::text) as product_preview,
  b.remarks,
  b.subtotal,
  b.tax_total,
  b.grand_total,
  b.created_at,
  b.posted_at,
  COALESCE(
    b.posted_at,
    b.created_at::timestamp with time zone,
    b.invoice_date::timestamp without time zone::timestamp with time zone
  ) as sort_ts
from
  base b
  left join line_meta lm on lm.id = b.id
  left join line_preview lp on lp.id = b.id;


create view public.v_billing_invoice_usage_summary as
with
  usage_base as (
    select
      il.so_line_id,
      il.qty,
      COALESCE(inv.invoice_no, ''::text) as invoice_no,
      COALESCE(inv.status, 'DRAFT'::text) as invoice_status
    from
      invoice_lines il
      left join invoices inv on inv.id = il.invoice_id
    where
      il.so_line_id is not null
  ),
  usage_totals as (
    select
      ub.so_line_id,
      COALESCE(sum(ub.qty), 0::numeric)::numeric(18, 3) as billed_qty,
      COALESCE(
        sum(
          case
            when upper(ub.invoice_status) = 'POSTED'::text then ub.qty
            else 0::numeric
          end
        ),
        0::numeric
      )::numeric(18, 3) as posted_qty,
      COALESCE(
        sum(
          case
            when upper(ub.invoice_status) <> 'POSTED'::text then ub.qty
            else 0::numeric
          end
        ),
        0::numeric
      )::numeric(18, 3) as draft_qty
    from
      usage_base ub
    group by
      ub.so_line_id
  ),
  usage_refs as (
    select
      x.so_line_id,
      string_agg(
        x.invoice_ref,
        ' || '::text
        order by
          x.invoice_ref
      ) as invoice_refs
    from
      (
        select distinct
          ub.so_line_id,
          (
            (ub.invoice_no || ' ['::text) || ub.invoice_status
          ) || ']'::text as invoice_ref
        from
          usage_base ub
        where
          ub.invoice_no <> ''::text
      ) x
    group by
      x.so_line_id
  )
select
  t.so_line_id,
  t.billed_qty,
  t.posted_qty,
  t.draft_qty,
  r.invoice_refs
from
  usage_totals t
  left join usage_refs r on r.so_line_id = t.so_line_id;

create view public.v_packing_board as
with
  prod as (
    select
      pe.wo_id,
      pe.routing_id,
      pe.produced_qty,
      row_number() over (
        partition by
          pe.wo_id
        order by
          pe.created_at desc
      ) as rn
    from
      production_entries pe
  ),
  pack as (
    select
      packing_records.so_number,
      packing_records.line_no,
      sum(packing_records.packed_qty) as packed_qty
    from
      packing_records
    group by
      packing_records.so_number,
      packing_records.line_no
  )
select
  woj.wo_id,
  woj.so_number,
  woj.line_no,
  woj.product_name,
  woj.client_name,
  woj.artwork_no,
  woj.qty as order_qty,
  COALESCE(prod.produced_qty, 0::numeric) as produced_qty,
  COALESCE(pack.packed_qty, 0::numeric) as packed_qty,
  COALESCE(prod.produced_qty, 0::numeric) - COALESCE(pack.packed_qty, 0::numeric) as balance_qty
from
  work_order_jobs woj
  left join prod on prod.wo_id = woj.wo_id
  and prod.rn = 1
  left join pack on pack.so_number = woj.so_number
  and pack.line_no = woj.line_no
where
  COALESCE(prod.produced_qty, 0::numeric) > COALESCE(pack.packed_qty, 0::numeric);


create view public.v_packing_queue_fast as
with
  artwork_category as (
    select
      a.artwork_no,
      max(
        case
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%FLEXO%'::text then 'FLEXO'::text
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%CORR%'::text then 'CORRUGATION'::text
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%OFFSET%'::text then 'OFFSET'::text
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%DIGITAL%'::text then 'DIGITAL'::text
          else null::text
        end
      ) as department_category
    from
      artworks a
    where
      COALESCE(a.artwork_no, ''::text) <> ''::text
    group by
      a.artwork_no
  ),
  job_base as (
    select
      woj.wo_id,
      wo_1.wo_number,
      wo_1.wo_date,
      woj.so_number,
      woj.line_no,
      max(NULLIF(woj.artwork_no, ''::text)) as artwork_no,
      max(NULLIF(woj.client_name, ''::text)) as client_name,
      string_agg(
        distinct woj.product_name,
        ' / '::text
        order by
          woj.product_name
      ) as product_name,
      COALESCE(
        max(ac.department_category),
        case
          when upper(COALESCE(max(woj.category), ''::text)) ~~ '%FLEXO%'::text then 'FLEXO'::text
          when upper(COALESCE(max(woj.category), ''::text)) ~~ '%CORR%'::text then 'CORRUGATION'::text
          when upper(COALESCE(max(woj.category), ''::text)) ~~ '%OFFSET%'::text then 'OFFSET'::text
          when upper(COALESCE(max(woj.category), ''::text)) ~~ '%DIGITAL%'::text then 'DIGITAL'::text
          else max(woj.category)
        end
      ) as department_category
    from
      work_order_jobs woj
      join work_orders wo_1 on wo_1.id = woj.wo_id
      left join artwork_category ac on ac.artwork_no = woj.artwork_no
    group by
      woj.wo_id,
      wo_1.wo_number,
      wo_1.wo_date,
      woj.so_number,
      woj.line_no
  ),
  final_production_routing as (
    select
      r.wo_id,
      r.id as routing_id,
      row_number() over (
        partition by
          r.wo_id
        order by
          r.sequence_no desc,
          r.id desc
      ) as rn
    from
      work_order_routing r
    where
      upper(
        TRIM(
          both
          from
            COALESCE(r.process_name, r.department, ''::text)
        )
      ) <> all (
        array[
          'PACKING'::text,
          'DISPATCH'::text,
          'CUTTING PRE'::text,
          'CUTTING POST'::text,
          'QC'::text
        ]
      )
  ),
  production_totals as (
    select
      pe.routing_id,
      sum(
        COALESCE(
          pe.ok_qty,
          GREATEST(
            COALESCE(pe.produced_qty, 0::numeric) - COALESCE(pe.rejected_qty, 0::numeric),
            0::numeric
          )
        )
      ) as produced_qty
    from
      production_entries pe
    group by
      pe.routing_id
  ),
  packing_totals as (
    select
      pr.so_line_id,
      pr.so_number,
      pr.line_no,
      (
        array_agg(
          pr.id
          order by
            pr.packed_at desc nulls last,
            pr.id
        )
      ) [1] as pack_id,
      sum(COALESCE(pr.packed_qty, 0::numeric)) as packed_qty,
      bool_or(pr.ready_to_dispatch = true) as ready_to_dispatch
    from
      packing_records pr
    group by
      pr.so_line_id,
      pr.so_number,
      pr.line_no
  )
select
  pt.pack_id,
  so.id as so_id,
  sol.id as so_line_id,
  jb.so_number,
  jb.line_no,
  sol.product_code,
  COALESCE(sol.product_name, jb.product_name) as product_name,
  jb.client_name,
  COALESCE(sol.category, jb.department_category) as category,
  jb.department_category,
  COALESCE(sol.qty, 0::numeric) as order_qty,
  COALESCE(prod.produced_qty, 0::numeric) as produced_qty,
  COALESCE(pt.packed_qty, 0::numeric) as packed_qty,
  pt.ready_to_dispatch,
  sol.expected_delivery,
  sol.final_delivery,
  sol.division,
  sol.quote_no,
  sol.pm_code,
  sol.product_remarks,
  sol.prepress_remarks,
  so.remarks as so_remarks,
  jb.artwork_no,
  jb.wo_number,
  jb.wo_date,
  COALESCE(
    wo.snapshot_json ->> 'modeOfTransport'::text,
    ''::text
  ) as transport_mode
from
  job_base jb
  join work_orders wo on wo.id = jb.wo_id
  left join final_production_routing fr on fr.wo_id = jb.wo_id
  and fr.rn = 1
  left join production_totals prod on prod.routing_id = fr.routing_id
  left join sales_orders so on so.so_number = jb.so_number
  left join sales_order_lines sol on sol.so_id = so.id
  and sol.line_no = jb.line_no
  left join packing_totals pt on pt.so_line_id = sol.id
where
  COALESCE(prod.produced_qty, 0::numeric) > 0::numeric;


create view public.v_production_board as
with
  stage_production as (
    select
      production_entries.routing_id,
      sum(production_entries.produced_qty) as produced_total
    from
      production_entries
    group by
      production_entries.routing_id
  )
select
  wor.id as routing_id,
  wo.id as wo_id,
  wo.wo_number,
  ((wo.snapshot_json -> 'jobs'::text) -> 0) ->> 'artworkNo'::text as artwork_no,
  ((wo.snapshot_json -> 'jobs'::text) -> 0) ->> 'client'::text as client_name,
  wor.process_name,
  wor.sequence_no,
  wor.department,
  wor.planned_machine,
  case
    when wor.sequence_no = 1 then (
      (
        ((wo.snapshot_json -> 'jobs'::text) -> 0) ->> 'coreSheets'::text
      )::numeric
    ) + (
      (
        (wo.snapshot_json -> 'wastage'::text) ->> 'processSheets'::text
      )::numeric
    )
    else COALESCE(prev_stage_prod.produced_total, 0::numeric)
  end as planned_qty,
  COALESCE(curr_stage_prod.produced_total, 0::numeric) as produced_qty
from
  work_order_routing wor
  join work_orders wo on wo.id = wor.wo_id
  left join stage_production curr_stage_prod on curr_stage_prod.routing_id = wor.id
  left join work_order_routing prev_stage on prev_stage.wo_id = wor.wo_id
  and prev_stage.sequence_no = (wor.sequence_no - 1)
  left join stage_production prev_stage_prod on prev_stage_prod.routing_id = prev_stage.id;

create view public.v_production_stage_queue_fast as
with recursive
  artwork_category as (
    select
      a.artwork_no,
      max(
        case
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%FLEXO%'::text then 'FLEXO'::text
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%CORR%'::text then 'CORRUGATION'::text
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%OFFSET%'::text then 'OFFSET'::text
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%DIGITAL%'::text then 'DIGITAL'::text
          else null::text
        end
      ) as department_category
    from
      artworks a
    where
      COALESCE(a.artwork_no, ''::text) <> ''::text
    group by
      a.artwork_no
  ),
  snapshot_metrics as (
    select
      wo.id as wo_id,
      COALESCE(
        (
          select
            sum(
              COALESCE(
                NULLIF(p.value ->> 'requiredQty'::text, ''::text)::numeric,
                NULLIF(p.value ->> 'sheetsWithWaste'::text, ''::text)::numeric,
                0::numeric
              )
            ) as sum
          from
            jsonb_array_elements(
              COALESCE(wo.snapshot_json -> 'papers'::text, '[]'::jsonb)
            ) p (value)
        ),
        0::numeric
      ) as paper_sheets_with_waste,
      COALESCE(
        (
          select
            sum(
              COALESCE(
                NULLIF(p.value ->> 'coreSheets'::text, ''::text)::numeric,
                NULLIF(p.value ->> 'sheets'::text, ''::text)::numeric,
                0::numeric
              )
            ) as sum
          from
            jsonb_array_elements(
              COALESCE(wo.snapshot_json -> 'papers'::text, '[]'::jsonb)
            ) p (value)
        ),
        0::numeric
      ) as paper_core_sheets,
      COALESCE(
        (
          select
            sum(
              COALESCE(
                NULLIF(j.value ->> 'coreSheets'::text, ''::text)::numeric,
                0::numeric
              )
            ) as sum
          from
            jsonb_array_elements(
              COALESCE(wo.snapshot_json -> 'jobs'::text, '[]'::jsonb)
            ) j (value)
        ),
        0::numeric
      ) as job_core_sheets,
      COALESCE(
        NULLIF(
          (wo.snapshot_json -> 'flexoDetails'::text) ->> 'totalRunningMeter'::text,
          ''::text
        )::numeric,
        NULLIF(
          (wo.snapshot_json -> 'flexoDetails'::text) ->> 'baseRunningMeter'::text,
          ''::text
        )::numeric,
        0::numeric
      ) as flexo_running_meter,
      COALESCE(
        NULLIF(
          (wo.snapshot_json -> 'wastage'::text) ->> 'processSheets'::text,
          ''::text
        )::numeric,
        0::numeric
      ) as process_waste_sheets
    from
      work_orders wo
  ),
  stores_issue_plan as (
    select
      wo.id as wo_id,
      sum(
        case
          when upper(
            COALESCE(split_part(l.remarks, '|'::text, 5), ''::text)
          ) = any (array['SHEET'::text, 'SHEETS'::text]) then COALESCE(l.qty_out, 0::numeric)
          else 0::numeric
        end
      ) as issued_sheet_qty,
      sum(
        case
          when upper(
            COALESCE(split_part(l.remarks, '|'::text, 5), ''::text)
          ) = any (
            array[
              'RM'::text,
              'R.M.'::text,
              'RUNNING METER'::text,
              'RUNNING METERS'::text
            ]
          ) then COALESCE(l.qty_out, 0::numeric)
          else 0::numeric
        end
      ) as issued_rm_qty
    from
      work_orders wo
      join inv_ledger l on l.ref_type = 'ISSUE'::text
      and l.ref_no = wo.wo_number
      left join inv_ledger rev on rev.ref_no = l.id::text
      and upper(COALESCE(rev.ref_type, ''::text)) ~~ 'REV-%'::text
    where
      rev.id is null
    group by
      wo.id
  ),
  initial_routing as (
    select distinct
      on (r.wo_id) r.wo_id,
      COALESCE(
        NULLIF(r.process_name, ''::text),
        NULLIF(r.department, ''::text),
        'PROCESS'::text
      ) as first_process_name
    from
      work_order_routing r
    where
      upper(
        COALESCE(
          NULLIF(r.process_name, ''::text),
          NULLIF(r.department, ''::text),
          'PROCESS'::text
        )
      ) <> all (
        array[
          'PACKING'::text,
          'DISPATCH'::text,
          'CUTTING PRE'::text,
          'CUTTING POST'::text,
          'QC'::text
        ]
      )
    order by
      r.wo_id,
      r.sequence_no,
      r.id
  ),
  job_meta as (
    select
      woj.wo_id,
      wo.wo_number,
      wo.wo_date,
      string_agg(
        distinct woj.so_number,
        ', '::text
        order by
          woj.so_number
      ) as so_numbers,
      string_agg(
        distinct woj.line_no::text,
        ', '::text
        order by
          (woj.line_no::text)
      ) as line_nos,
      string_agg(
        distinct woj.artwork_no,
        ', '::text
        order by
          woj.artwork_no
      ) as artwork_nos,
      max(NULLIF(woj.client_name, ''::text)) as client_name,
      string_agg(
        distinct woj.product_name,
        ' / '::text
        order by
          woj.product_name
      ) as product_names,
      max(NULLIF(woj.job_priority, ''::text)) as job_priority,
      max(woj.expected_delivery) as expected_delivery,
      COALESCE(
        NULLIF(sum(COALESCE(woj.ups, 0::numeric)), 0::numeric),
        case
          when COALESCE(
            NULLIF(max(sm.paper_core_sheets), 0::numeric),
            NULLIF(max(sm.job_core_sheets), 0::numeric),
            0::numeric
          ) > 0::numeric then sum(COALESCE(woj.qty, 0::numeric)) / COALESCE(
            NULLIF(max(sm.paper_core_sheets), 0::numeric),
            NULLIF(max(sm.job_core_sheets), 0::numeric)
          )
          else 1::numeric
        end
      ) as ups_factor,
      case
        when upper(
          COALESCE(
            (wo.snapshot_json -> 'jobDetails'::text) ->> 'type'::text,
            ''::text
          )
        ) ~~ '%FLEXO%'::text then 'FLEXO'::text
        else COALESCE(
          max(ac.department_category),
          case
            when upper(
              COALESCE(
                (wo.snapshot_json -> 'jobDetails'::text) ->> 'type'::text,
                ''::text
              )
            ) ~~ '%CORR%'::text then 'CORRUGATION'::text
            when upper(
              COALESCE(
                (wo.snapshot_json -> 'jobDetails'::text) ->> 'type'::text,
                ''::text
              )
            ) ~~ '%OFFSET%'::text then 'OFFSET'::text
            when upper(
              COALESCE(
                (wo.snapshot_json -> 'jobDetails'::text) ->> 'type'::text,
                ''::text
              )
            ) ~~ '%DIGITAL%'::text then 'DIGITAL'::text
            when upper(COALESCE(max(woj.category), ''::text)) ~~ '%FLEXO%'::text then 'FLEXO'::text
            when upper(COALESCE(max(woj.category), ''::text)) ~~ '%CORR%'::text then 'CORRUGATION'::text
            when upper(COALESCE(max(woj.category), ''::text)) ~~ '%OFFSET%'::text then 'OFFSET'::text
            when upper(COALESCE(max(woj.category), ''::text)) ~~ '%DIGITAL%'::text then 'DIGITAL'::text
            else null::text
          end
        )
      end as department_category,
      COALESCE(
        case
          when upper(
            COALESCE(
              (wo.snapshot_json -> 'jobDetails'::text) ->> 'type'::text,
              ''::text
            )
          ) ~~ '%FLEXO%'::text
          or upper(COALESCE(max(ir.first_process_name), ''::text)) ~~ '%INSPECTION%'::text
          or upper(COALESCE(max(ir.first_process_name), ''::text)) ~~ '%SLITTING%'::text
          or upper(COALESCE(max(ac.department_category), ''::text)) = 'FLEXO'::text
          or upper(COALESCE(max(woj.category), ''::text)) ~~ '%FLEXO%'::text then NULLIF(max(sip.issued_rm_qty), 0::numeric)
          else NULLIF(max(sip.issued_sheet_qty), 0::numeric)
        end,
        case
          when upper(
            COALESCE(
              (wo.snapshot_json -> 'jobDetails'::text) ->> 'type'::text,
              ''::text
            )
          ) ~~ '%FLEXO%'::text
          or upper(COALESCE(max(ir.first_process_name), ''::text)) ~~ '%INSPECTION%'::text
          or upper(COALESCE(max(ir.first_process_name), ''::text)) ~~ '%SLITTING%'::text
          or upper(COALESCE(max(ac.department_category), ''::text)) = 'FLEXO'::text
          or upper(COALESCE(max(woj.category), ''::text)) ~~ '%FLEXO%'::text then NULLIF(max(sm.flexo_running_meter), 0::numeric)
          else null::numeric
        end,
        NULLIF(max(sm.paper_sheets_with_waste), 0::numeric),
        NULLIF(
          max(sm.paper_core_sheets) + max(sm.process_waste_sheets),
          0::numeric
        ),
        NULLIF(max(sm.flexo_running_meter), 0::numeric),
        max(sm.job_core_sheets) + max(sm.process_waste_sheets)
      ) as stage1_plan_qty
    from
      work_order_jobs woj
      join work_orders wo on wo.id = woj.wo_id
      join snapshot_metrics sm on sm.wo_id = wo.id
      left join stores_issue_plan sip on sip.wo_id = wo.id
      left join initial_routing ir on ir.wo_id = wo.id
      left join artwork_category ac on ac.artwork_no = woj.artwork_no
    group by
      woj.wo_id,
      wo.wo_number,
      wo.wo_date,
      wo.snapshot_json
  ),
  production_totals as (
    select
      pe.routing_id,
      sum(
        GREATEST(COALESCE(pe.produced_qty, 0::numeric), 0::numeric)
      ) as gross_produced_qty,
      sum(
        COALESCE(
          pe.ok_qty,
          GREATEST(
            COALESCE(pe.produced_qty, 0::numeric) - COALESCE(pe.rejected_qty, 0::numeric),
            0::numeric
          )
        )
      ) as ok_produced_qty
    from
      production_entries pe
    group by
      pe.routing_id
  ),
  routing_base as (
    select
      r.id as routing_id,
      r.wo_id,
      jm.wo_number,
      jm.wo_date,
      jm.so_numbers,
      jm.line_nos,
      jm.artwork_nos,
      jm.client_name,
      jm.product_names,
      jm.job_priority,
      jm.expected_delivery,
      jm.ups_factor,
      jm.department_category,
      r.sequence_no,
      COALESCE(
        NULLIF(r.process_name, ''::text),
        NULLIF(r.department, ''::text),
        'PROCESS'::text
      ) as process_name,
      r.department,
      r.planned_machine,
      upper(COALESCE(r.status, ''::text)) as routing_status,
      COALESCE(pt.gross_produced_qty, 0::numeric) as gross_produced_qty,
      COALESCE(pt.ok_produced_qty, 0::numeric) as ok_produced_qty,
      jm.stage1_plan_qty,
      row_number() over (
        partition by
          r.wo_id
        order by
          r.sequence_no,
          r.id
      ) as rn
    from
      work_order_routing r
      join job_meta jm on jm.wo_id = r.wo_id
      left join production_totals pt on pt.routing_id = r.id
    where
      upper(
        COALESCE(
          NULLIF(r.process_name, ''::text),
          NULLIF(r.department, ''::text),
          'PROCESS'::text
        )
      ) <> all (
        array[
          'PACKING'::text,
          'DISPATCH'::text,
          'CUTTING PRE'::text,
          'CUTTING POST'::text,
          'QC'::text
        ]
      )
  ),
  stage_calc as (
    select
      rb.routing_id,
      rb.wo_id,
      rb.wo_number,
      rb.wo_date,
      rb.so_numbers,
      rb.line_nos,
      rb.artwork_nos,
      rb.client_name,
      rb.product_names,
      rb.job_priority,
      rb.expected_delivery,
      rb.ups_factor,
      rb.department_category,
      rb.sequence_no,
      rb.process_name,
      rb.department,
      rb.planned_machine,
      rb.routing_status,
      rb.gross_produced_qty as produced_qty,
      rb.ok_produced_qty,
      rb.stage1_plan_qty as planned_qty,
      GREATEST(
        rb.stage1_plan_qty - rb.gross_produced_qty,
        0::numeric
      ) as balance_qty,
      rb.rn
    from
      routing_base rb
    where
      rb.rn = 1
    union all
    select
      rb.routing_id,
      rb.wo_id,
      rb.wo_number,
      rb.wo_date,
      rb.so_numbers,
      rb.line_nos,
      rb.artwork_nos,
      rb.client_name,
      rb.product_names,
      rb.job_priority,
      rb.expected_delivery,
      rb.ups_factor,
      rb.department_category,
      rb.sequence_no,
      rb.process_name,
      rb.department,
      rb.planned_machine,
      rb.routing_status,
      rb.gross_produced_qty as produced_qty,
      rb.ok_produced_qty,
      case
        when upper(rb.process_name) ~~ '%PRINT%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%LAMINAT%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%COAT%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%DIE CUT%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%DIECUT%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%INSPECTION%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%SLITTING%'::text then sc_1.ok_produced_qty
        when upper(sc_1.process_name) ~~ '%PRINT%'::text then round(
          sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
        )
        when upper(sc_1.process_name) ~~ '%LAMINAT%'::text then round(
          sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
        )
        when upper(sc_1.process_name) ~~ '%COAT%'::text then round(
          sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
        )
        when upper(sc_1.process_name) ~~ '%DIE CUT%'::text then round(
          sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
        )
        when upper(sc_1.process_name) ~~ '%DIECUT%'::text then round(
          sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
        )
        else sc_1.ok_produced_qty
      end as planned_qty,
      GREATEST(
        case
          when upper(rb.process_name) ~~ '%PRINT%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%LAMINAT%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%COAT%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%DIE CUT%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%DIECUT%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%INSPECTION%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%SLITTING%'::text then sc_1.ok_produced_qty
          when upper(sc_1.process_name) ~~ '%PRINT%'::text then round(
            sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
          )
          when upper(sc_1.process_name) ~~ '%LAMINAT%'::text then round(
            sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
          )
          when upper(sc_1.process_name) ~~ '%COAT%'::text then round(
            sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
          )
          when upper(sc_1.process_name) ~~ '%DIE CUT%'::text then round(
            sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
          )
          when upper(sc_1.process_name) ~~ '%DIECUT%'::text then round(
            sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
          )
          else sc_1.ok_produced_qty
        end - rb.gross_produced_qty,
        0::numeric
      ) as balance_qty,
      rb.rn
    from
      routing_base rb
      join stage_calc sc_1 on sc_1.wo_id = rb.wo_id
      and rb.rn = (sc_1.rn + 1)
  )
select
  routing_id,
  wo_id,
  wo_number,
  wo_date,
  so_numbers,
  line_nos,
  artwork_nos,
  client_name,
  product_names,
  department_category,
  process_name,
  department,
  planned_machine,
  sequence_no,
  expected_delivery,
  job_priority,
  planned_qty,
  produced_qty,
  balance_qty,
  case
    when routing_status = 'SHORT_CLOSED'::text then 'SHORT_CLOSED'::text
    when routing_status = 'HOLD'::text then 'HOLD'::text
    when planned_qty <= 0::numeric
    and produced_qty <= 0::numeric then 'PENDING'::text
    when balance_qty <= 0::numeric then 'COMPLETED'::text
    when produced_qty > 0::numeric then 'IN_PROGRESS'::text
    else 'PENDING'::text
  end as status
from
  stage_calc sc;


create view public.v_production_stage_rows_fast as
with recursive
  split_marker as (
    select
      q.wo_id,
      min(q.sequence_no) as split_seq
    from
      v_production_stage_queue_fast q
    where
      upper(COALESCE(q.process_name, ''::text)) ~~ '%DIE CUT%'::text
      or upper(COALESCE(q.process_name, ''::text)) ~~ '%DIECUT%'::text
    group by
      q.wo_id
  ),
  queue_meta as (
    select distinct
      q.wo_id,
      q.wo_date,
      q.wo_number,
      q.so_numbers,
      q.line_nos,
      q.artwork_nos,
      q.client_name,
      q.product_names,
      q.department_category,
      q.expected_delivery,
      q.job_priority
    from
      v_production_stage_queue_fast q
  ),
  combined_rows as (
    select
      q.routing_id::text as row_key,
      0 as row_sort,
      ''::text as job_sort,
      'COMBINED'::text as row_kind,
      'Combined Sheet Flow'::text as row_description,
      case
        when upper(COALESCE(q.department_category, ''::text)) = 'FLEXO'::text then 'RM'::text
        else 'SHEETS'::text
      end as plan_unit,
      q.wo_id,
      q.wo_date,
      q.routing_id,
      q.wo_number,
      q.so_numbers,
      q.so_numbers as so_number_display,
      null::text as so_number,
      null::text as line_no,
      null::text as job_reference,
      null::numeric as job_ups,
      q.client_name,
      q.product_names as product_name,
      q.product_names,
      q.artwork_nos as artwork_no,
      q.artwork_nos,
      q.process_name,
      case
        when upper(COALESCE(q.department_category, ''::text)) = 'DIGITAL'::text
        and upper(COALESCE(q.process_name, ''::text)) ~~ '%DIGITAL%'::text then 'Printing'::text
        when upper(COALESCE(q.department_category, ''::text)) = 'FLEXO'::text
        and (
          upper(COALESCE(q.process_name, ''::text)) ~~ '%INSPECTION%'::text
          or upper(COALESCE(q.process_name, ''::text)) ~~ '%SLITTING%'::text
        ) then 'Inspection/Slitting'::text
        when upper(COALESCE(q.department_category, ''::text)) = 'FLEXO'::text
        and upper(COALESCE(q.process_name, ''::text)) ~~ '%DIE SHEET%'::text then 'Flexo Die Sheeting'::text
        when upper(COALESCE(q.department_category, ''::text)) = 'FLEXO'::text
        and (
          upper(COALESCE(q.process_name, ''::text)) ~~ '%FLEXO PRINT%'::text
          or upper(COALESCE(q.process_name, ''::text)) ~~ '%ONLINE GOLD FOIL%'::text
          or upper(COALESCE(q.process_name, ''::text)) ~~ '%ONLINE SILVER FOIL%'::text
        ) then 'Flexo Printing'::text
        when upper(COALESCE(q.department_category, ''::text)) = 'FLEXO'::text
        and (
          upper(COALESCE(q.process_name, ''::text)) ~~ '%DIE CUT%'::text
          or upper(COALESCE(q.process_name, ''::text)) ~~ '%DIECUT%'::text
        ) then 'Flexo Die Cutting Offline'::text
        when upper(COALESCE(q.process_name, ''::text)) = 'CORRUGATION SHEET PASTING'::text then 'Sheet Pasting'::text
        else q.process_name
      end as process_display_name,
      q.department,
      q.planned_machine,
      q.sequence_no,
      q.expected_delivery,
      q.job_priority,
      q.department_category,
      q.planned_qty,
      q.produced_qty,
      q.balance_qty,
      q.status
    from
      v_production_stage_queue_fast q
      left join split_marker sm on sm.wo_id = q.wo_id
    where
      (
        sm.split_seq is null
        or q.sequence_no <= sm.split_seq
      )
      and (
        upper(COALESCE(q.process_name, ''::text)) <> all (
          array[
            'PACKING'::text,
            'DISPATCH'::text,
            'CUTTING PRE'::text,
            'CUTTING POST'::text,
            'QC'::text
          ]
        )
      )
  ),
  job_dimension as (
    select
      woj.wo_id,
      woj.so_number,
      woj.line_no::text as line_no,
      woj.job_reference,
      COALESCE(
        NULLIF(woj.job_reference, ''::text),
        (COALESCE(woj.so_number, ''::text) || '||'::text) || COALESCE(woj.line_no::text, ''::text)
      ) as job_key,
      max(NULLIF(woj.client_name, ''::text)) as client_name,
      max(NULLIF(woj.product_name, ''::text)) as product_name,
      max(NULLIF(woj.artwork_no, ''::text)) as artwork_no,
      max(NULLIF(woj.job_priority, ''::text)) as job_priority,
      max(woj.expected_delivery) as expected_delivery,
      COALESCE(
        NULLIF(max(COALESCE(woj.ups, 0::numeric)), 0::numeric),
        NULLIF(
          max(COALESCE(woj.group_ups, 0::numeric)),
          0::numeric
        ),
        1::numeric
      ) as job_ups
    from
      work_order_jobs woj
    group by
      woj.wo_id,
      woj.so_number,
      (woj.line_no::text),
      woj.job_reference,
      (
        COALESCE(
          NULLIF(woj.job_reference, ''::text),
          (COALESCE(woj.so_number, ''::text) || '||'::text) || COALESCE(woj.line_no::text, ''::text)
        )
      )
  ),
  routing_entry_totals as (
    select
      pe.routing_id,
      sum(
        GREATEST(COALESCE(pe.produced_qty, 0::numeric), 0::numeric)
      ) as gross_produced_qty,
      sum(
        COALESCE(
          pe.ok_qty,
          GREATEST(
            COALESCE(pe.produced_qty, 0::numeric) - COALESCE(pe.rejected_qty, 0::numeric),
            0::numeric
          )
        )
      ) as ok_produced_qty
    from
      production_entries pe
    group by
      pe.routing_id
  ),
  job_entry_totals as (
    select
      pe.routing_id,
      COALESCE(
        NULLIF(pe.so_number, ''::text),
        NULLIF(
          TRIM(
            both
            from
              "substring" (
                COALESCE(pe.downtime_reason, ''::text),
                'SO:([^|]+)'::text
              )
          ),
          ''::text
        )
      ) as so_number,
      COALESCE(
        NULLIF(pe.line_no, ''::text),
        NULLIF(
          TRIM(
            both
            from
              "substring" (
                COALESCE(pe.downtime_reason, ''::text),
                'LINE:([^|]+)'::text
              )
          ),
          ''::text
        )
      ) as line_no,
      COALESCE(
        NULLIF(pe.job_reference, ''::text),
        NULLIF(
          TRIM(
            both
            from
              "substring" (
                COALESCE(pe.downtime_reason, ''::text),
                'JOBREF:([^|]+)'::text
              )
          ),
          ''::text
        )
      ) as job_reference,
      sum(
        GREATEST(COALESCE(pe.produced_qty, 0::numeric), 0::numeric)
      ) as gross_produced_qty,
      sum(
        COALESCE(
          pe.ok_qty,
          GREATEST(
            COALESCE(pe.produced_qty, 0::numeric) - COALESCE(pe.rejected_qty, 0::numeric),
            0::numeric
          )
        )
      ) as ok_produced_qty
    from
      production_entries pe
    group by
      pe.routing_id,
      (
        COALESCE(
          NULLIF(pe.so_number, ''::text),
          NULLIF(
            TRIM(
              both
              from
                "substring" (
                  COALESCE(pe.downtime_reason, ''::text),
                  'SO:([^|]+)'::text
                )
            ),
            ''::text
          )
        )
      ),
      (
        COALESCE(
          NULLIF(pe.line_no, ''::text),
          NULLIF(
            TRIM(
              both
              from
                "substring" (
                  COALESCE(pe.downtime_reason, ''::text),
                  'LINE:([^|]+)'::text
                )
            ),
            ''::text
          )
        )
      ),
      (
        COALESCE(
          NULLIF(pe.job_reference, ''::text),
          NULLIF(
            TRIM(
              both
              from
                "substring" (
                  COALESCE(pe.downtime_reason, ''::text),
                  'JOBREF:([^|]+)'::text
                )
            ),
            ''::text
          )
        )
      )
  ),
  job_totals as (
    select
      jet.routing_id,
      jet.so_number,
      jet.line_no,
      jet.job_reference,
      COALESCE(
        NULLIF(jet.job_reference, ''::text),
        (COALESCE(jet.so_number, ''::text) || '||'::text) || COALESCE(jet.line_no, ''::text)
      ) as job_key,
      jet.gross_produced_qty,
      jet.ok_produced_qty
    from
      job_entry_totals jet
  ),
  post_split_routing as (
    select
      r.id as routing_id,
      r.wo_id,
      qm.wo_date,
      qm.wo_number,
      qm.so_numbers,
      qm.line_nos,
      qm.artwork_nos,
      qm.client_name,
      qm.product_names,
      qm.department_category,
      qm.expected_delivery,
      qm.job_priority,
      r.sequence_no,
      COALESCE(
        NULLIF(r.process_name, ''::text),
        NULLIF(r.department, ''::text),
        'PROCESS'::text
      ) as process_name,
      r.department,
      r.planned_machine,
      upper(COALESCE(r.status, ''::text)) as routing_status,
      sm.split_seq,
      row_number() over (
        partition by
          r.wo_id
        order by
          r.sequence_no,
          r.id
      ) as post_rn
    from
      work_order_routing r
      join split_marker sm on sm.wo_id = r.wo_id
      join queue_meta qm on qm.wo_id = r.wo_id
    where
      r.sequence_no > sm.split_seq
      and (
        upper(COALESCE(r.process_name, ''::text)) <> all (
          array[
            'PACKING'::text,
            'DISPATCH'::text,
            'CUTTING PRE'::text,
            'CUTTING POST'::text,
            'QC'::text
          ]
        )
      )
  ),
  split_seed as (
    select
      psr.wo_id,
      psr.sequence_no,
      psr.post_rn,
      psr.routing_id,
      jd.job_key,
      jd.so_number,
      jd.line_no,
      jd.job_reference,
      jd.client_name,
      jd.product_name,
      jd.artwork_no,
      jd.job_priority,
      jd.expected_delivery,
      jd.job_ups,
      case
        when upper(COALESCE(psr.process_name, ''::text)) ~~ '%INSPECTION%'::text
        or upper(COALESCE(psr.process_name, ''::text)) ~~ '%SLITTING%'::text then COALESCE(ret.ok_produced_qty, 0::numeric)
        else COALESCE(ret.ok_produced_qty, 0::numeric) * COALESCE(NULLIF(jd.job_ups, 0::numeric), 1::numeric)
      end as planned_qty,
      COALESCE(jt.gross_produced_qty, 0::numeric) as produced_qty,
      COALESCE(jt.ok_produced_qty, 0::numeric) as ok_produced_qty
    from
      post_split_routing psr
      join split_marker sm on sm.wo_id = psr.wo_id
      and psr.post_rn = 1
      join job_dimension jd on jd.wo_id = psr.wo_id
      left join v_production_stage_queue_fast dc on dc.wo_id = psr.wo_id
      and dc.sequence_no = sm.split_seq
      left join routing_entry_totals ret on ret.routing_id = dc.routing_id
      left join job_totals jt on jt.routing_id = psr.routing_id
      and jt.job_key = jd.job_key
  ),
  job_stage_calc as (
    select
      ss.wo_id,
      ss.sequence_no,
      ss.post_rn,
      ss.routing_id,
      ss.job_key,
      ss.so_number,
      ss.line_no,
      ss.job_reference,
      ss.client_name,
      ss.product_name,
      ss.artwork_no,
      ss.job_priority,
      ss.expected_delivery,
      ss.job_ups,
      ss.planned_qty,
      ss.produced_qty,
      ss.ok_produced_qty
    from
      split_seed ss
    union all
    select
      nxt.wo_id,
      nxt.sequence_no,
      nxt.post_rn,
      nxt.routing_id,
      prev.job_key,
      prev.so_number,
      prev.line_no,
      prev.job_reference,
      prev.client_name,
      prev.product_name,
      prev.artwork_no,
      prev.job_priority,
      prev.expected_delivery,
      prev.job_ups,
      prev.ok_produced_qty as planned_qty,
      COALESCE(jt.gross_produced_qty, 0::numeric) as produced_qty,
      COALESCE(jt.ok_produced_qty, 0::numeric) as ok_produced_qty
    from
      job_stage_calc prev
      join post_split_routing nxt on nxt.wo_id = prev.wo_id
      and nxt.post_rn = (prev.post_rn + 1)
      left join job_totals jt on jt.routing_id = nxt.routing_id
      and jt.job_key = prev.job_key
  )
select
  cr.row_key,
  cr.row_sort,
  cr.job_sort,
  cr.row_kind,
  cr.row_description,
  cr.plan_unit,
  cr.wo_id,
  cr.wo_date,
  cr.routing_id,
  cr.wo_number,
  cr.so_numbers,
  cr.so_number_display,
  cr.so_number,
  cr.line_no,
  cr.job_reference,
  cr.job_ups,
  cr.client_name,
  cr.product_name,
  cr.product_names,
  cr.artwork_no,
  cr.artwork_nos,
  cr.process_name,
  cr.process_display_name,
  cr.department,
  cr.planned_machine,
  cr.sequence_no,
  cr.expected_delivery,
  cr.job_priority,
  cr.department_category,
  cr.planned_qty,
  cr.produced_qty,
  cr.balance_qty,
  cr.status
from
  combined_rows cr
union all
select
  (jsc.routing_id::text || '||'::text) || jsc.job_key as row_key,
  1 as row_sort,
  COALESCE(
    NULLIF(jsc.job_reference, ''::text),
    (COALESCE(jsc.so_number, ''::text) || '-'::text) || COALESCE(jsc.line_no, ''::text)
  ) as job_sort,
  'JOB'::text as row_kind,
  'Separated Job'::text as row_description,
  case
    when upper(COALESCE(psr.process_name, ''::text)) ~~ '%INSPECTION%'::text
    or upper(COALESCE(psr.process_name, ''::text)) ~~ '%SLITTING%'::text then 'RM'::text
    else 'UNITS'::text
  end as plan_unit,
  psr.wo_id,
  psr.wo_date,
  psr.routing_id,
  psr.wo_number,
  psr.so_numbers,
  COALESCE(jsc.so_number, ''::text) as so_number_display,
  jsc.so_number,
  jsc.line_no,
  jsc.job_reference,
  jsc.job_ups,
  COALESCE(
    NULLIF(jsc.client_name, ''::text),
    psr.client_name
  ) as client_name,
  jsc.product_name,
  psr.product_names,
  jsc.artwork_no,
  psr.artwork_nos,
  psr.process_name,
  case
    when upper(COALESCE(psr.department_category, ''::text)) = 'DIGITAL'::text
    and upper(COALESCE(psr.process_name, ''::text)) ~~ '%DIGITAL%'::text then 'Printing'::text
    when upper(COALESCE(psr.department_category, ''::text)) = 'FLEXO'::text
    and (
      upper(COALESCE(psr.process_name, ''::text)) ~~ '%INSPECTION%'::text
      or upper(COALESCE(psr.process_name, ''::text)) ~~ '%SLITTING%'::text
    ) then 'Inspection/Slitting'::text
    when upper(COALESCE(psr.department_category, ''::text)) = 'FLEXO'::text
    and upper(COALESCE(psr.process_name, ''::text)) ~~ '%DIE SHEET%'::text then 'Flexo Die Sheeting'::text
    when upper(COALESCE(psr.department_category, ''::text)) = 'FLEXO'::text
    and (
      upper(COALESCE(psr.process_name, ''::text)) ~~ '%FLEXO PRINT%'::text
      or upper(COALESCE(psr.process_name, ''::text)) ~~ '%ONLINE GOLD FOIL%'::text
      or upper(COALESCE(psr.process_name, ''::text)) ~~ '%ONLINE SILVER FOIL%'::text
    ) then 'Flexo Printing'::text
    when upper(COALESCE(psr.department_category, ''::text)) = 'FLEXO'::text
    and (
      upper(COALESCE(psr.process_name, ''::text)) ~~ '%DIE CUT%'::text
      or upper(COALESCE(psr.process_name, ''::text)) ~~ '%DIECUT%'::text
    ) then 'Flexo Die Cutting Offline'::text
    when upper(COALESCE(psr.process_name, ''::text)) = 'CORRUGATION SHEET PASTING'::text then 'Sheet Pasting'::text
    else psr.process_name
  end as process_display_name,
  psr.department,
  psr.planned_machine,
  psr.sequence_no,
  COALESCE(jsc.expected_delivery, psr.expected_delivery) as expected_delivery,
  COALESCE(jsc.job_priority, psr.job_priority) as job_priority,
  psr.department_category,
  jsc.planned_qty,
  jsc.produced_qty,
  GREATEST(jsc.planned_qty - jsc.produced_qty, 0::numeric) as balance_qty,
  case
    when psr.routing_status = 'SHORT_CLOSED'::text then 'SHORT_CLOSED'::text
    when psr.routing_status = 'HOLD'::text then 'HOLD'::text
    when jsc.planned_qty <= 0::numeric
    and jsc.produced_qty <= 0::numeric then 'PENDING'::text
    when GREATEST(jsc.planned_qty - jsc.produced_qty, 0::numeric) <= 0::numeric then 'COMPLETED'::text
    when jsc.produced_qty > 0::numeric then 'IN_PROGRESS'::text
    else 'PENDING'::text
  end as status
from
  job_stage_calc jsc
  join post_split_routing psr on psr.routing_id = jsc.routing_id;


create view public.v_production_summary_fast as
select
  wo_date,
  department_category,
  process_display_name,
  status,
  count(*) as total_rows,
  sum(COALESCE(balance_qty, 0::numeric)) as pending_qty,
  sum(
    case
      when upper(COALESCE(status, ''::text)) = 'IN_PROGRESS'::text then 1
      else 0
    end
  ) as in_progress_rows,
  sum(
    case
      when upper(COALESCE(status, ''::text)) = 'SHORT_CLOSED'::text then 1
      else 0
    end
  ) as short_closed_rows
from
  v_production_stage_rows_fast
group by
  wo_date,
  department_category,
  process_display_name,
  status;

create view public.v_report_delivery_performance as
with
  dispatch_base as (
    select
      dr.so_line_id,
      max(dr.dispatch_date) as last_dispatch_date,
      sum(COALESCE(dr.dispatch_qty, 0::numeric)) as dispatched_qty
    from
      dispatch_records dr
    group by
      dr.so_line_id
  )
select
  so.so_number,
  sol.line_no,
  COALESCE(c.client_name, so.client_code) as client_name,
  sol.product_name,
  COALESCE(sol.final_delivery, sol.expected_delivery) as delivery_date,
  db.last_dispatch_date,
  COALESCE(sol.qty, 0::numeric) as order_qty,
  COALESCE(db.dispatched_qty, 0::numeric) as dispatched_qty,
  case
    when db.last_dispatch_date is null then 'PENDING'::text
    when COALESCE(sol.final_delivery, sol.expected_delivery) is null then 'DISPATCHED'::text
    when db.last_dispatch_date <= COALESCE(sol.final_delivery, sol.expected_delivery) then 'ON_TIME'::text
    else 'DELAYED'::text
  end as delivery_status,
  case
    when db.last_dispatch_date is null
    or COALESCE(sol.final_delivery, sol.expected_delivery) is null then 0
    else GREATEST(
      db.last_dispatch_date - COALESCE(sol.final_delivery, sol.expected_delivery),
      0
    )
  end as delay_days
from
  sales_orders so
  join sales_order_lines sol on sol.so_id = so.id
  left join clients c on c.client_code = so.client_code
  left join dispatch_base db on db.so_line_id = sol.id
where
  upper(COALESCE(so.status, 'OPEN'::text)) <> 'CANCELLED'::text;


create view public.v_report_order_line_traceability as
with
  artwork_base as (
    select
      a.so_id,
      a.line_no,
      max(a.artwork_no) as artwork_no,
      max(a.status) as artwork_status,
      max(a.plate_status) as plate_status,
      max(a.die_status) as die_status
    from
      artworks a
    group by
      a.so_id,
      a.line_no
  ),
  wo_base as (
    select
      woj.so_number,
      woj.line_no,
      count(distinct woj.wo_id) as wo_count,
      string_agg(
        distinct wo.wo_number,
        ', '::text
        order by
          wo.wo_number
      ) as wo_numbers
    from
      work_order_jobs woj
      join work_orders wo on wo.id = woj.wo_id
    group by
      woj.so_number,
      woj.line_no
  ),
  pack_base as (
    select
      pr.so_line_id,
      sum(COALESCE(pr.packed_qty, 0::numeric)) as packed_qty
    from
      packing_records pr
    group by
      pr.so_line_id
  ),
  dispatch_base as (
    select
      dr.so_line_id,
      sum(COALESCE(dr.dispatch_qty, 0::numeric)) as dispatched_qty,
      max(dr.dispatch_date) as last_dispatch_date
    from
      dispatch_records dr
    group by
      dr.so_line_id
  ),
  invoice_base as (
    select
      il.so_line_id,
      sum(COALESCE(il.qty, 0::numeric)) as billed_qty
    from
      invoice_lines il

  )

create view public.v_report_planning_lines as
with
  artwork_ranked as (
    select
      a.so_id,
      a.line_no,
      a.id,
      a.artwork_no,
      a.product_type,
      a.plate_status,
      a.die_status,
      COALESCE(
        NULLIF(a.status, ''::text),
        case
          when COALESCE(a.artwork_no, ''::text) <> ''::text then 'PENDING_APPROVAL'::text
          else 'NO_ART'::text
        end
      ) as artwork_status,
      a.artwork_at,
      a.approved_at,
      row_number() over (
        partition by
          a.so_id,
          a.line_no
        order by
          a.approved_at desc nulls last,
          a.artwork_at desc nulls last,
          a.id desc
      ) as rn
    from
      artworks a
  ),
  artwork_base as (
    select
      artwork_ranked.so_id,
      artwork_ranked.line_no,
      artwork_ranked.artwork_no,
      artwork_ranked.product_type,
      artwork_ranked.plate_status,
      artwork_ranked.die_status,
      artwork_ranked.artwork_status,
      artwork_ranked.artwork_at,
      artwork_ranked.approved_at as artwork_approved_at
    from
      artwork_ranked
    where
      artwork_ranked.rn = 1
  ),
  wo_base as (
    select
      woj.so_number,
      woj.line_no,
      count(distinct woj.wo_id) as wo_count,
      string_agg(
        distinct wo.wo_number,
        ', '::text
        order by
          wo.wo_number
      ) as wo_numbers,
      min(wo.wo_date) as first_wo_at,
      max(wo.wo_date) as last_wo_at
    from
      work_order_jobs woj
      join work_orders wo on wo.id = woj.wo_id
    group by
      woj.so_number,
      woj.line_no
  ),
  routing_meta as (
    select
      woj.so_number,
      woj.line_no,
      count(distinct r.id) as routing_count,
      min(wo.wo_date) as routing_marked_at
    from
      work_order_jobs woj
      join work_orders wo on wo.id = woj.wo_id
      join work_order_routing r on r.wo_id = woj.wo_id
    group by
      woj.so_number,
      woj.line_no
  ),
  routing_steps as (
    select
      step_rows.so_number,
      step_rows.line_no,
      string_agg(
        step_rows.step_name,
        ' | '::text
        order by
          step_rows.sequence_no,
          step_rows.step_name
      ) as routing_steps
    from
      (
        select distinct
          woj.so_number,
          woj.line_no,
          COALESCE(r.sequence_no, 0) as sequence_no,
          COALESCE(
            NULLIF(r.process_name, ''::text),
            NULLIF(r.department, ''::text),
            'PROCESS'::text
          ) as step_name
        from
          work_order_jobs woj
          join work_order_routing r on r.wo_id = woj.wo_id
      ) step_rows
    group by
      step_rows.so_number,
      step_rows.line_no
  ),
  routing_base as (
    select
      rm.so_number,
      rm.line_no,
      rm.routing_count,
      rm.routing_marked_at,
      rs.routing_steps
    from
      routing_meta rm
      left join routing_steps rs on rs.so_number = rm.so_number
      and rs.line_no = rm.line_no
  ),
  stage_rows as (
    select
      woj.so_number,
      woj.line_no,
      COALESCE(psq.process_name, 'PROCESS'::text) as process_name,
      min(COALESCE(psq.sequence_no, 0)) as sequence_no,
      sum(COALESCE(psq.planned_qty, 0::numeric)) as planned_qty,
      sum(COALESCE(psq.produced_qty, 0::numeric)) as produced_qty,
      max(pe.entry_datetime) as stage_updated_at,
      case
        when bool_or(
          upper(COALESCE(psq.status, 'PENDING'::text)) = 'SHORT_CLOSED'::text
        ) then 'SHORT_CLOSED'::text
        when bool_or(
          upper(COALESCE(psq.status, 'PENDING'::text)) = 'HOLD'::text
        ) then 'HOLD'::text
        when bool_or(
          upper(COALESCE(psq.status, 'PENDING'::text)) = 'IN_PROGRESS'::text
        ) then 'IN_PROGRESS'::text
        when bool_and(
          upper(COALESCE(psq.status, 'PENDING'::text)) = 'COMPLETED'::text
        ) then 'COMPLETED'::text
        else 'PENDING'::text
      end as stage_status
    from
      work_order_jobs woj
      join v_production_stage_queue_fast psq on psq.wo_id = woj.wo_id
      left join production_entries pe on pe.routing_id = psq.routing_id
    group by
      woj.so_number,
      woj.line_no,
      (COALESCE(psq.process_name, 'PROCESS'::text))
  ),
  stage_base as (
    select
      stage_rows.so_number,
      stage_rows.line_no,
      string_agg(
        (
          (
            (
              (
                (
                  (stage_rows.process_name || ': '::text) || TRIM(
                    both
                    from
                      to_char(
                        COALESCE(stage_rows.produced_qty, 0::numeric),
                        'FM999999990.###'::text
                      )
                  )
                ) || '/'::text
              ) || TRIM(
                both
                from
                  to_char(
                    COALESCE(stage_rows.planned_qty, 0::numeric),
                    'FM999999990.###'::text
                  )
              )
            ) || ' ['::text
          ) || stage_rows.stage_status
        ) || ']'::text,
        ' | '::text
        order by
          stage_rows.sequence_no,
          stage_rows.process_name
      ) as stage_summary,
      max(stage_rows.stage_updated_at) as stage_last_updated_at
    from
      stage_rows
    group by
      stage_rows.so_number,
      stage_rows.line_no
  ),
  current_stage as (
    select distinct
      on (stage_rows.so_number, stage_rows.line_no) stage_rows.so_number,
      stage_rows.line_no,
      stage_rows.process_name as current_stage,
      stage_rows.stage_status as current_stage_status
    from
      stage_rows
    where
      upper(
        COALESCE(stage_rows.stage_status, 'PENDING'::text)
      ) <> 'COMPLETED'::text
    order by
      stage_rows.so_number,
      stage_rows.line_no,
      (
        case
          when upper(
            COALESCE(stage_rows.stage_status, 'PENDING'::text)
          ) = 'IN_PROGRESS'::text then 0
          when upper(
            COALESCE(stage_rows.stage_status, 'PENDING'::text)
          ) = 'PENDING'::text then 1
          when upper(
            COALESCE(stage_rows.stage_status, 'PENDING'::text)
          ) = 'HOLD'::text then 2
          when upper(
            COALESCE(stage_rows.stage_status, 'PENDING'::text)
          ) = 'SHORT_CLOSED'::text then 3
          else 4
        end
      ),
      stage_rows.sequence_no,
      stage_rows.process_name
  ),
  invoice_base as (
    select
      il.so_line_id,
      sum(COALESCE(il.qty, 0::numeric)) as billed_qty,
      count(distinct il.invoice_id) as invoice_count,
      min(inv.invoice_date) as first_invoice_date,
      max(inv.invoice_date) as last_invoice_date,
      max(inv.created_at) as last_billed_at,
      string_agg(
        distinct inv.invoice_no,
        ', '::text
        order by
          inv.invoice_no
      ) filter (
        where
          COALESCE(inv.invoice_no, ''::text) <> ''::text
      ) as invoice_nos,
      string_agg(
        distinct to_char(
          inv.invoice_date::timestamp with time zone,
          'YYYY-MM-DD'::text
        ),
        ', '::text
        order by
          (
            to_char(
              inv.invoice_date::timestamp with time zone,
              'YYYY-MM-DD'::text
            )
          )
      ) filter (
        where
          inv.invoice_date is not null
      ) as invoice_dates
    from
      invoice_lines il
      left join invoices inv on inv.id = il.invoice_id
    where
      il.so_line_id is not null
    group by
      il.so_line_id
  )
select
  so.id as so_id,
  sol.id as so_line_id,
  so.so_number,
  so.so_date,
  so.po_number,
  so.po_date,
  COALESCE(
    so.created_at,
    so.so_date::timestamp without time zone
  ) as so_datetime,
  so.created_at as so_created_at,
  to_char(
    COALESCE(
      so.created_at,
      so.so_date::timestamp without time zone
    ),
    'HH24:MI'::text
  ) as so_time,
  so.client_code,
  COALESCE(c.client_name, so.client_code) as client_name,
  so.sales_rep,
  sol.line_no,
  sol.product_code,
  sol.product_name,
  sol.category,
  COALESCE(sol.qty, 0::numeric) as order_qty,
  sol.unit,
  sol.rate,
  sol.expected_delivery,
  sol.final_delivery,
  COALESCE(sol.accounts_status, 'PENDING'::text) as accounts_status,
  sol.accounts_at,
  COALESCE(sol.business_status, 'PENDING'::text) as business_status,
  sol.business_at,
  case
    when upper(COALESCE(sol.accounts_status, 'PENDING'::text)) = 'APPROVED'::text
    and upper(COALESCE(sol.business_status, 'PENDING'::text)) = 'APPROVED'::text then 'APPROVED'::text
    when upper(COALESCE(sol.accounts_status, ''::text)) = 'HOLD'::text
    or upper(COALESCE(sol.business_status, ''::text)) = 'HOLD'::text then 'HOLD'::text
    else 'PENDING'::text
  end as sales_approval_status,
  case
    when upper(COALESCE(sol.accounts_status, 'PENDING'::text)) = 'APPROVED'::text
    and upper(COALESCE(sol.business_status, 'PENDING'::text)) = 'APPROVED'::text then GREATEST(
      COALESCE(sol.accounts_at, sol.business_at),
      COALESCE(sol.business_at, sol.accounts_at)
    )
    else null::timestamp with time zone
  end as sales_approval_at,
  ab.artwork_no,
  ab.product_type,
  ab.plate_status,
  ab.die_status,
  COALESCE(ab.artwork_status, 'NO_ART'::text) as artwork_status,
  ab.artwork_at,
  ab.artwork_approved_at,
  COALESCE(wb.wo_count, 0::bigint) as wo_count,
  COALESCE(wb.wo_numbers, ''::text) as wo_numbers,
  wb.first_wo_at,
  wb.last_wo_at,
  COALESCE(rb.routing_count, 0::bigint) as routing_count,
  COALESCE(rb.routing_steps, ''::text) as routing_steps,
  rb.routing_marked_at,
  case
    when COALESCE(rb.routing_count, 0::bigint) > 0 then 'MARKED'::text
    else 'PENDING'::text
  end as routing_status,
  COALESCE(cs.current_stage, ''::text) as current_stage,
  COALESCE(cs.current_stage_status, ''::text) as current_stage_status,
  COALESCE(sb.stage_summary, ''::text) as stage_summary,
  sb.stage_last_updated_at,
  COALESCE(ib.billed_qty, 0::numeric) as billed_qty,
  GREATEST(
    COALESCE(sol.qty, 0::numeric) - COALESCE(ib.billed_qty, 0::numeric),
    0::numeric
  ) as billing_pending_qty,
  COALESCE(ib.invoice_count, 0::bigint) as invoice_count,
  COALESCE(ib.invoice_nos, ''::text) as invoice_nos,
  COALESCE(ib.invoice_dates, ''::text) as invoice_dates,
  ib.first_invoice_date,
  ib.last_invoice_date,
  ib.last_billed_at,
  case
    when COALESCE(ib.billed_qty, 0::numeric) >= COALESCE(sol.qty, 0::numeric)
    and COALESCE(sol.qty, 0::numeric) > 0::numeric then 'CLOSED'::text
    else 'PENDING'::text
  end as billing_status
from
  sales_orders so
  join sales_order_lines sol on sol.so_id = so.id
  left join clients c on c.client_code = so.client_code
  left join artwork_base ab on ab.so_id = so.id
  and ab.line_no = sol.line_no
  left join wo_base wb on wb.so_number = so.so_number
  and wb.line_no = sol.line_no
  left join routing_base rb on rb.so_number = so.so_number
  and rb.line_no = sol.line_no
  left join stage_base sb on sb.so_number = so.so_number
  and sb.line_no = sol.line_no
  left join current_stage cs on cs.so_number = so.so_number
  and cs.line_no = sol.line_no
  left join invoice_base ib on ib.so_line_id = sol.id
where
  upper(COALESCE(so.status, 'OPEN'::text)) <> 'CANCELLED'::text;

create view public.v_report_unbilled_dispatch as
with
  dispatch_base as (
    select
      dr.so_line_id,
      max(dr.so_number) as so_number,
      max(dr.line_no) as line_no,
      max(dr.product_name) as product_name,
      max(dr.dispatch_date) as dispatch_date,
      sum(COALESCE(dr.dispatch_qty, 0::numeric)) as dispatched_qty
    from
      dispatch_records dr
    group by
      dr.so_line_id
  ),
  invoice_base as (
    select
      il.so_line_id,
      sum(COALESCE(il.qty, 0::numeric)) as billed_qty
    from
      invoice_lines il
    group by
      il.so_line_id
  )
select
  db.so_line_id,
  db.so_number,
  db.line_no,
  db.product_name,
  db.dispatch_date,
  db.dispatched_qty,
  COALESCE(ib.billed_qty, 0::numeric) as billed_qty,
  GREATEST(
    db.dispatched_qty - COALESCE(ib.billed_qty, 0::numeric),
    0::numeric
  ) as unbilled_qty
from
  dispatch_base db
  left join invoice_base ib on ib.so_line_id = db.so_line_id
where
  GREATEST(
    db.dispatched_qty - COALESCE(ib.billed_qty, 0::numeric),
    0::numeric
  ) > 0::numeric;


create view public.v_sales_order_line_details as
select
  sol.id,
  sol.so_id,
  so.so_number,
  so.so_date,
  sol.line_no,
  sol.product_code,
  sol.product_name,
  sol.category,

    sol.amount,

    ) - COALESCE(sol.disc_amount, 0::numeric) + COALESCE(sol.cgst, 0::numeric) + COALESCE(sol.sgst, 0::numeric) + COALESCE(sol.igst, 0::numeric)

  COALESCE(sol.job_reference, ''::text) as job_reference,
  COALESCE(sol.job_priority, ''::text) as job_priority,

  COALESCE(sol.business_status, 'PENDING'::text) as business_status,
  COALESCE(src.wo_status, 'PENDING'::text) as wo_status,
  COALESCE(sol.final_delivery, sol.expected_delivery) as delivery_display,
  concat_ws(
    ' | '::text,
    NULLIF(sol.product_remarks, ''::text),
    NULLIF(sol.prepress_remarks, ''::text)
  ) as remarks_display
from
  sales_order_lines sol
  join sales_orders so on so.id = sol.so_id
  left join items i on i.item_code = sol.product_code

create view public.v_sales_orders_dashboard as

  so_number,
  so_date,
  client_code,
  sales_rep,
  po_number,
  sales_type,
  currency,
  subtotal,
  discount_total,
  cgst_total,
  sgst_total,
  igst_total,
  grand_total,
  status,
  created_at,
  case
    when (
      exists (
        select
          1
        from
          work_order_jobs w
        where
          w.so_number = so.so_number
      )
    ) then 'CREATED'::text
    else 'PENDING'::text
  end as wo_status,

    when (
      exists (
        select
          1
        from
          sales_order_lines l
        where
          l.so_id = so.id
          and l.accounts_status = 'REJECTED'::text
      )
    ) then 'REJECTED'::text
    when (
      exists (
        select
          1
        from
          sales_order_lines l
        where
          l.so_id = so.id
          and (
            l.accounts_status is null
            or l.accounts_status = 'PENDING'::text
          )
      )
    ) then 'PENDING'::text
    else 'APPROVED'::text
  end as accounts_approved,
  case
    when (
      exists (
        select
          1
        from
          sales_order_lines l
        where
          l.so_id = so.id
          and l.business_status = 'REJECTED'::text
      )
    ) then 'REJECTED'::text
    when (
      exists (
        select
          1
        from
          sales_order_lines l
        where
          l.so_id = so.id
          and (
            l.business_status is null
            or l.business_status = 'PENDING'::text
          )
      )
    ) then 'PENDING'::text
    else 'APPROVED'::text
  end as business_approved
from
  sales_orders so;

  upper(COALESCE(so.status, 'OPEN'::text)) as status,
  COALESCE(
    src.wo_status,
    NULLIF(upper(COALESCE(so.wo_status, ''::text)), ''::text),
    'PENDING'::text
  ) as wo_status,
  COALESCE(src.line_count, 0) as line_count,
  COALESCE(src.total_qty, 0::numeric) as total_qty,
  src.latest_delivery,
  COALESCE(src.is_approval_hold, false) as is_approval_hold,
  COALESCE(
    src.can_edit,
    upper(COALESCE(so.status, 'OPEN'::text)) <> 'CANCELLED'::text
  ) as can_edit,
  COALESCE(
    src.can_hold,
    upper(COALESCE(so.status, 'OPEN'::text)) <> 'CANCELLED'::text
  ) as can_hold,
  COALESCE(
    src.can_cancel,
    upper(COALESCE(so.status, 'OPEN'::text)) <> 'CANCELLED'::text
  ) as can_cancel,
  COALESCE(
    src.hold_target,
    case
      when upper(COALESCE(so.status, 'OPEN'::text)) = 'HOLD'::text then 'OPEN'::text
      else 'HOLD'::text
    end
  ) as hold_target,
  COALESCE(
    src.hold_label,
    case
      when upper(COALESCE(so.status, 'OPEN'::text)) = 'HOLD'::text then 'Reopen'::text
      else 'Hold'::text
    end
  ) as hold_label
from
  sales_orders so
  left join clients c on c.client_code = so.client_code
  left join sales_order_rollup_cache src on src.so_id = so.id::text;


create view public.v_workorder_candidates as
with
  service_only_items as (
    select
      v.item_code,
      v.item_name,
      v.category
    from
      (
        values
          (
            'SLC001'::text,
            'FLEXO PRINTING PLATE'::text,
            'FLEXO PRINTING PLATE'::text
          ),
          (
            'SLC002'::text,
            'OFFSET PRINTING PLATE'::text,
            'OFFSET PRINTING PLATE'::text
          ),
          ('SLC003'::text, 'DIE'::text, 'DIE'::text)
      ) v (item_code, item_name, category)
  ),
  artwork_ranked as (
    select
      a.so_id,
      a.line_no,
      a.artwork_no,
      a.product_type,
      a.sheet_length,
      a.sheet_width,
      a.sheet_ups,
      a.printing_colors,
      a.status,
      a.approved_at,
      row_number() over (
        partition by
          a.so_id,
          a.line_no
        order by
          (COALESCE(a.approved_at, a.artwork_at)) desc nulls last,
          a.id desc
      ) as rn
    from
      artworks a
  ),
  artwork_pick as (
    select
      artwork_ranked.so_id,
      artwork_ranked.line_no,
      artwork_ranked.artwork_no,
      artwork_ranked.product_type,
      artwork_ranked.sheet_length,
      artwork_ranked.sheet_width,
      artwork_ranked.sheet_ups,
      artwork_ranked.printing_colors,
      artwork_ranked.status,
      artwork_ranked.approved_at
    from
      artwork_ranked
    where
      artwork_ranked.rn = 1
  ),
  wo_totals as (
    select
      woj.so_number,
      woj.line_no,
      sum(COALESCE(woj.qty, 0::numeric)) as processed_qty,
      string_agg(
        distinct NULLIF(wo.wo_number, ''::text),
        ', '::text
        order by
          (NULLIF(wo.wo_number, ''::text))
      ) as wo_list
    from
      work_order_jobs woj
      left join work_orders wo on wo.id = woj.wo_id
    group by
      woj.so_number,
      woj.line_no
  ),
  base as (
    select
      so.id as so_id,
      sol.id as so_line_id,
      so.so_number,
      so.so_date,
      so.created_at as so_created_at,
      to_char(
        COALESCE(
          so.created_at,
          so.so_date::timestamp without time zone
        ),
        'HH24:MI'::text
      ) as so_time,
      so.sales_rep,
      so.client_code,
      COALESCE(c.client_name, so.client_code, ''::text) as client_name,
      sol.line_no,
      sol.product_code,
      sol.product_name,
      sol.category,
      sol.qty,
      sol.unit,
      so.po_number,
      sol.expected_delivery,
      sol.job_priority,
      sol.job_type,
      so.remarks as so_remarks,
      sol.product_remarks,
      sol.prepress_remarks as prepress_remark,
      sol.job_reference,
      upper(
        COALESCE(
          NULLIF(sol.accounts_status, ''::text),
          'PENDING'::text
        )
      ) as accounts_status,
      upper(
        COALESCE(
          NULLIF(sol.business_status, ''::text),
          'PENDING'::text
        )
      ) as business_status,
      ap.artwork_no,
      COALESCE(
        NULLIF(ap.product_type, ''::text),
        case
          when upper(COALESCE(sol.category, ''::text)) ~~ '%FLEXO%'::text then 'Flexo'::text
          when upper(COALESCE(sol.category, ''::text)) ~~ '%CORR%'::text then 'Corrugated'::text
          when upper(COALESCE(sol.category, ''::text)) ~~ '%DIGITAL%'::text then 'Digital'::text
          when upper(COALESCE(sol.category, ''::text)) ~~ '%OFFSET%'::text then 'Offset'::text
          else ''::text
        end
      ) as product_type,
      ap.sheet_length,
      ap.sheet_width,
      ap.sheet_ups,
      ap.printing_colors,
      case
        when ap.so_id is null then 'NO_ART'::text
        else upper(
          COALESCE(NULLIF(ap.status, ''::text), 'PENDING'::text)
        )
      end as artwork_status,
      ap.approved_at as artwork_approved_at,
      COALESCE(wt.processed_qty, 0::numeric) as processed_qty,
      GREATEST(
        COALESCE(sol.qty, 0::numeric) - COALESCE(wt.processed_qty, 0::numeric),
        0::numeric
      ) as remaining_qty,
      COALESCE(wt.wo_list, ''::text) as wo_list,
      case
        when upper(
          COALESCE(
            NULLIF(ap.product_type, ''::text),
            sol.category,
            ''::text
          )
        ) ~~ '%FLEXO%'::text then 'FLEXO'::text
        when upper(
          COALESCE(
            NULLIF(ap.product_type, ''::text),
            sol.category,
            ''::text
          )
        ) ~~ '%CORR%'::text then 'CORRUGATION'::text
        when upper(
          COALESCE(
            NULLIF(ap.product_type, ''::text),
            sol.category,
            ''::text
          )
        ) ~~ '%DIGITAL%'::text then 'DIGITAL'::text
        when upper(
          COALESCE(
            NULLIF(ap.product_type, ''::text),
            sol.category,
            ''::text
          )
        ) ~~ '%OFFSET%'::text then 'OFFSET'::text
        else upper(COALESCE(sol.category, ''::text))
      end as department_category
    from
      sales_order_lines sol
      join sales_orders so on so.id = sol.so_id
      left join clients c on c.client_code = so.client_code
      left join artwork_pick ap on ap.so_id = sol.so_id
      and ap.line_no = sol.line_no
      left join wo_totals wt on wt.so_number = so.so_number
      and wt.line_no = sol.line_no
    where
      upper(COALESCE(so.status, ''::text)) <> 'CANCELLED'::text
      and not (
        exists (
          select
            1
          from
            service_only_items s
          where
            s.item_code = upper(
              TRIM(
                both
                from
                  COALESCE(sol.product_code, ''::text)
              )
            )
            or s.item_name = upper(
              TRIM(
                both
                from
                  COALESCE(sol.product_name, ''::text)
              )
            )
            or s.category = upper(
              TRIM(
                both
                from
                  COALESCE(sol.category, ''::text)
              )
            )
        )
      )
  ),
  scored as (
    select
      b.so_id,
      b.so_line_id,
      b.so_number,
      b.so_date,
      b.so_created_at,
      b.so_time,
      b.sales_rep,
      b.client_code,
      b.client_name,
      b.line_no,
      b.product_code,
      b.product_name,
      b.category,
      b.qty,
      b.unit,
      b.po_number,
      b.expected_delivery,
      b.job_priority,
      b.job_type,
      b.so_remarks,
      b.product_remarks,
      b.prepress_remark,
      b.job_reference,
      b.accounts_status,
      b.business_status,
      b.artwork_no,
      b.product_type,
      b.sheet_length,
      b.sheet_width,
      b.sheet_ups,
      b.printing_colors,
      b.artwork_status,
      b.artwork_approved_at,
      b.processed_qty,
      b.remaining_qty,
      b.wo_list,
      b.department_category,
      concat_ws(
        ' / '::text,
        case
          when b.accounts_status = 'REJECTED'::text then 'Accounts'::text
          else null::text
        end,
        case
          when b.business_status = 'REJECTED'::text then 'Business'::text
          else null::text
        end,
        case
          when b.artwork_status = 'REJECTED'::text then 'Artwork'::text
          else null::text
        end
      ) as rejected_stage_list,
      concat_ws(
        ' / '::text,
        case
          when b.accounts_status <> 'APPROVED'::text then 'Accounts'::text
          else null::text
        end,
        case
          when b.business_status <> 'APPROVED'::text then 'Business'::text
          else null::text
        end,
        case
          when b.artwork_status <> 'APPROVED'::text then 'Artwork'::text
          else null::text
        end
      ) as pending_stage_list
    from
      base b
    where
      b.department_category !~~ '%FLEXO%'::text
  )
select
  so_id,
  so_line_id,
  so_number,
  so_date,
  so_created_at,
  so_time,
  sales_rep,
  client_code,
  client_name,
  line_no,
  product_code,
  product_name,
  category,
  department_category,
  qty,
  unit,
  po_number,
  expected_delivery,
  job_priority,
  job_type,
  so_remarks,
  product_remarks,
  prepress_remark,
  job_reference,
  accounts_status,
  business_status,
  artwork_no,
  product_type,
  sheet_length,
  sheet_width,
  sheet_ups,
  printing_colors,
  artwork_status,
  artwork_approved_at,
  processed_qty,
  remaining_qty,
  wo_list,
  case
    when rejected_stage_list <> ''::text then 'Rejected at '::text || rejected_stage_list
    when pending_stage_list <> ''::text then 'Pending at '::text || pending_stage_list
    else 'Approved'::text
  end as approval_stage,
  rejected_stage_list = ''::text
  and pending_stage_list = ''::text as can_create_wo
from
  scored;


create view public.wo_material_pending as
select
  wom.wo_id,
  wom.material_key,
  wom.required_qty,
  wom.uom,
  COALESCE(sum(l.qty_out), 0::numeric) as issued_qty,
  wom.required_qty - COALESCE(sum(l.qty_out), 0::numeric) as pending_qty
from
  work_order_materials wom
  left join inv_ledger l on l.ref_type = 'ISSUE'::text
  and l.ref_no = wom.wo_id::text
  and l.remarks = wom.material_key
group by
  wom.wo_id,
  wom.material_key,
  wom.required_qty,
  wom.uom;

        when eb.raw_machine_upper ~~ '%CD102%'::text
        or eb.raw_machine_upper ~~ '%CD-102%'::text
        or eb.raw_machine_upper ~~ '%CD 102%'::text then 'CD102'::text
        when eb.raw_machine_upper ~~ '%LAMINATION 01%'::text
        or eb.raw_machine_upper ~~ '%LAMINATION 1%'::text
        or eb.raw_machine_upper ~~ '%LAMINATION ROLL-1%'::text
        or eb.raw_machine_upper ~~ '%LAMINATION ROLL 1%'::text
        or eb.raw_machine_upper ~~ '%LAM-1%'::text
        or eb.raw_machine_upper = 'LAM1'::text then 'LAM1'::text
        when eb.raw_machine_upper ~~ '%LAMINATION 02%'::text
        or eb.raw_machine_upper ~~ '%LAMINATION 2%'::text
        or eb.raw_machine_upper ~~ '%LAM-2%'::text
        or eb.raw_machine_upper = 'LAM2'::text then 'LAM2'::text
        when eb.raw_machine_upper ~~ '%LAMINATION 03%'::text
        or eb.raw_machine_upper ~~ '%LAMINATION 3%'::text
        or eb.raw_machine_upper ~~ '%LAM-3%'::text
        or eb.raw_machine_upper = 'LAM3'::text then 'LAM3'::text
        when eb.raw_machine_upper ~~ '%RHYGUAN%'::text
        or eb.raw_process_upper ~~ '%INSPECTION%'::text
        or eb.raw_process_upper ~~ '%SLITTING%'::text
        or eb.raw_process_upper ~~ '%SLIT%'::text then 'FLEXOSLIT'::text
        when eb.raw_process_upper ~~ '%FLEXO%'::text
        and (
          eb.raw_process_upper ~~ '%DIE%'::text
          or eb.raw_process_upper ~~ '%CUT%'::text
        )
        or eb.raw_machine_upper ~~ '%FLEXO%'::text
        and (
          eb.raw_machine_upper ~~ '%DIE%'::text
          or eb.raw_machine_upper ~~ '%CUT%'::text
        ) then 'FLEXODIE'::text
        when eb.raw_process_upper ~~ '%FLEXO%'::text
        or eb.raw_machine_upper ~~ '%FLEXO%'::text
        or eb.raw_machine_upper ~~ '%MARKANY%'::text then 'FLEXOPRINT'::text
        when eb.raw_machine_upper ~~ '%DIGITAL%'::text
        or eb.raw_machine_upper ~~ '%CANON%'::text then 'DIGITAL'::text
        when eb.raw_machine_upper ~~ '%2 PLY%'::text
        or eb.raw_machine_upper ~~ '%TWO PLY%'::text
        or eb.raw_machine_upper ~~ '%PLY MAKING%'::text then '2PLY'::text
        when eb.raw_machine_upper ~~ '%FMZ%'::text
        or eb.raw_machine_upper ~~ '%FLUTE%'::text then 'FMZ'::text
        when eb.raw_machine_upper ~~ '%MANUAL PASTING%'::text
        or eb.raw_machine_upper ~~ '%MANUAL SHEET PASTING 2%'::text
        or eb.raw_machine_upper ~~ '%MANUAL SHEET PASTING - 2%'::text then 'MSP2'::text
        when eb.raw_machine_upper ~~ '%MANUAL SHEET PASTING%'::text then 'MSP1'::text
        when eb.raw_machine_upper ~~ '%MANUAL DIE CUTTING 05%'::text
        or eb.raw_machine_upper ~~ '%DIE CUTTING MANUAL 2%'::text
        or eb.raw_machine_upper ~~ '%DIE CUTTING MANUAL - 2%'::text then 'DCMAN2'::text
        when eb.raw_machine_upper ~~ '%MANUAL DIE%'::text
        or eb.raw_machine_upper ~~ '%DIE CUTTING MANUAL%'::text
        or eb.raw_machine_upper ~~ '%MANUAL DIE CUTTING 04%'::text then 'DCMAN1'::text
        when eb.raw_machine_upper ~~ '%AUTOMATIC DIE%'::text
        or eb.raw_machine_upper ~~ '%AUTO DIE%'::text
        or eb.raw_machine_upper ~~ '%DIE CUTTING AUTO%'::text then 'DCAUTO'::text
        when eb.raw_machine_upper ~~ '%DIE CUTTING 01%'::text
        or eb.raw_machine_upper ~~ '%DIE CUTTING-01%'::text
        or eb.raw_machine_upper ~~ '%DIE-CUTTING-01%'::text
        or eb.raw_machine_upper ~~ '%DIE-1%'::text
        or eb.raw_machine_upper ~~ '%DYE-1%'::text then 'DIE1'::text
        when eb.raw_machine_upper ~~ '%DIE CUTTING 02%'::text
        or eb.raw_machine_upper ~~ '%DIE CUTTING-02%'::text
        or eb.raw_machine_upper ~~ '%DIE-CUTTING-02%'::text
        or eb.raw_machine_upper ~~ '%DIE-2%'::text
        or eb.raw_machine_upper ~~ '%DYE-2%'::text then 'DIE2'::text
        when eb.raw_machine_upper ~~ '%DIE CUTTING 03%'::text
        or eb.raw_machine_upper ~~ '%DIE CUTTING-03%'::text
        or eb.raw_machine_upper ~~ '%DIE-CUTTING-03%'::text
        or eb.raw_machine_upper ~~ '%DIE-3%'::text
        or eb.raw_machine_upper ~~ '%DYE-3%'::text then 'DIE3'::text
        when eb.raw_machine_upper ~~ '%DIE CUTTING 04%'::text
        or eb.raw_machine_upper ~~ '%DIE-4%'::text
        or eb.raw_machine_upper ~~ '%DYE-4%'::text then 'DIE4'::text
        when eb.raw_machine_upper ~~ '%DIE CUTTING 05%'::text
        or eb.raw_machine_upper ~~ '%DIE-5%'::text
        or eb.raw_machine_upper ~~ '%DYE-5%'::text then 'DIE5'::text
        when eb.raw_machine_upper ~~ '%STITCHING MANUAL 02%'::text
        or eb.raw_machine_upper ~~ '%STITCHING MANUAL - 2%'::text then 'STMAN2'::text
        when eb.raw_machine_upper ~~ '%STITCHING MANUAL%'::text
        or eb.raw_machine_upper ~~ '%STITCHING MANUAL 01%'::text
        or eb.raw_machine_upper ~~ '%STITCHING MANUAL - 1%'::text then 'STMAN1'::text
        when eb.raw_machine_upper ~~ '%AUTO STITCH%'::text
        or eb.raw_machine_upper ~~ '%STITCHING AUTO%'::text then 'STAUTO'::text
        when eb.raw_machine_upper ~~ '%FOLDER GLUER%'::text then 'FOLDER'::text
        when eb.raw_machine_upper ~~ '%RS4%'::text then 'RS4'::text
        else regexp_replace(
          eb.raw_machine_upper,
          '[^A-Z0-9]+'::text,
          ''::text,
          'g'::text
        )
      end as machine_key
    from
      entry_base eb
  ),
  daily_agg as (
    select
      en.production_date,
      en.machine_key,
      COALESCE(ms.machine, max(en.raw_machine)) as machine,
      COALESCE(ms.standard_make_ready_minutes, 0::numeric) as standard_make_ready_minutes,
      COALESCE(ms.target_hourly_output, 0::numeric) as target_hourly_output,
      string_agg(
        distinct en.operator_name,
        ', '::text
        order by
          en.operator_name
      ) filter (
        where
          en.operator_name is not null
      ) as operator_names,
      max(en.entry_local_ts) as last_completion_at,
      count(distinct en.job_key) as make_ready_jobs,
      sum(en.produced_qty) as actual_production,
      sum(en.downtime_minutes) / 60.0 as downtime_hours,
      string_agg(
        distinct en.downtime_label,
        ', '::text
        order by
          en.downtime_label
      ) filter (
        where
          en.downtime_label is not null
      ) as downtime_reasons
    from
      entry_normalized en
      left join machine_standards ms on ms.machine_key = en.machine_key
    group by
      en.production_date,
      en.machine_key,
      ms.machine,
      ms.standard_make_ready_minutes,
      ms.target_hourly_output
  ),
  daily_calc as (
    select
      da.production_date,
      da.machine_key,
      da.machine,
      da.standard_make_ready_minutes,
      da.target_hourly_output,
      da.operator_names,
      da.last_completion_at,
      da.make_ready_jobs,
      da.actual_production,
      da.downtime_hours,
      da.downtime_reasons,
      round(
        8::numeric + case
          when da.last_completion_at::time without time zone > '17:30:00'::time without time zone then EXTRACT(
            epoch
            from
              da.last_completion_at - (
                da.production_date::timestamp without time zone + '17:30:00'::time without time zone::interval
              )
          ) / 3600.0
          else 0::numeric
        end,
        2
      ) as working_hours,
      round(
        da.make_ready_jobs::numeric * da.standard_make_ready_minutes / 60.0,
        2
      ) as total_make_ready_hours
    from
      daily_agg da
  ),
  final_calc as (
    select
      dc.production_date,
      dc.machine_key,
      dc.machine,
      dc.standard_make_ready_minutes,
      dc.target_hourly_output,
      dc.operator_names,
      dc.last_completion_at,
      dc.make_ready_jobs,
      dc.actual_production,
      dc.downtime_hours,
      dc.downtime_reasons,
      dc.working_hours,
      dc.total_make_ready_hours,
      round(
        GREATEST(
          dc.working_hours - COALESCE(dc.downtime_hours, 0::numeric) - dc.total_make_ready_hours,
          0::numeric
        ),
        2
      ) as actual_running_hours
    from
      daily_calc dc
  )
select
  production_date,
  machine_key,
  machine,
  COALESCE(operator_names, ''::text) as operator_names,
  round(working_hours, 2) as working_hours,
  COALESCE(downtime_reasons, ''::text) as downtime_reasons,
  round(COALESCE(downtime_hours, 0::numeric), 2) as downtime_hours,
  make_ready_jobs,
  standard_make_ready_minutes,
  total_make_ready_hours,
  actual_running_hours,
  target_hourly_output,
  round(
    actual_running_hours * case
      when machine_key = any (
        array[
          'FLEXOPRINT'::text,
          'FLEXODIE'::text,
          'FLEXOSLIT'::text
        ]
      ) then target_hourly_output * 60::numeric
      else target_hourly_output
    end,
    2
  ) as target_production,
  round(actual_production, 2) as actual_production,
  round(
    actual_production / NULLIF(actual_running_hours, 0::numeric) / case
      when machine_key = any (
        array[
          'FLEXOPRINT'::text,
          'FLEXODIE'::text,
          'FLEXOSLIT'::text
        ]
      ) then 60
      else 1
    end::numeric,
    2
  ) as actual_nop,
  round(
    actual_production / NULLIF(actual_running_hours, 0::numeric) / case
      when machine_key = any (
        array[
          'FLEXOPRINT'::text,
          'FLEXODIE'::text,
          'FLEXOSLIT'::text
        ]
      ) then 60
      else 1
    end::numeric - NULLIF(target_hourly_output, 0::numeric),
    2
  ) as nop_deviation,
  round(
    actual_production / NULLIF(actual_running_hours, 0::numeric) / case
      when machine_key = any (
        array[
          'FLEXOPRINT'::text,
          'FLEXODIE'::text,
          'FLEXOSLIT'::text
        ]
      ) then 60
      else 1
    end::numeric / NULLIF(target_hourly_output, 0::numeric) * 100.0,
    2
  ) as nop_achievement_pct,
  case
    when COALESCE(target_hourly_output, 0::numeric) <= 0::numeric then 'NO_TARGET'::text
    when (
      actual_production / NULLIF(actual_running_hours, 0::numeric) / case
        when machine_key = any (
          array[
            'FLEXOPRINT'::text,
            'FLEXODIE'::text,
            'FLEXOSLIT'::text
          ]
        ) then 60
        else 1
      end::numeric
    ) >= target_hourly_output then 'ON_TARGET'::text
    else 'BELOW_TARGET'::text
  end as performance_status,
  last_completion_at
from
  final_calc;
