  id uuid not null default gen_random_uuid (),
  item_id uuid not null,
  location text not null default 'MAIN'::text,

  qty_in numeric null default 0,
  qty_out numeric null default 0,
  rate numeric null default 0,

  constraint inv_ledger_pkey primary key (id),

  constraint chk_single_direction check (
    (
      (
        (qty_in > (0)::numeric)

create materialized view public.inv_stock_analytics_mv as
with
  lot_ranked as (
    select
      lt_1.item_id,
      upper(COALESCE(lt_1.location, 'MAIN'::text)) as location,
      TRIM(
        both
        from
          COALESCE(lt_1.batch_no, ''::text)
      ) as batch_no,
      lt_1.receipt_date,
      lt_1.created_at,
      COALESCE(lt_1.qty_available, 0::numeric) as qty_available,
      row_number() over (
        partition by
          lt_1.item_id,
          (upper(COALESCE(lt_1.location, 'MAIN'::text)))
        order by
          lt_1.receipt_date,
          lt_1.created_at,
          lt_1.id
      ) as rn
    from
      inv_lots lt_1
    where
      COALESCE(lt_1.qty_available, 0::numeric) > 0::numeric
  ),
  lot_agg as (
    select
      lr.item_id,
      lr.location,
      count(*)::integer as batch_count,
      min(lr.receipt_date) as oldest_receipt_at,
      max(
        case
          when lr.rn = 1 then lr.batch_no
          else null::text
        end
      ) as next_batch_no
    from
      lot_ranked lr
    group by
      lr.item_id,
      lr.location
  ),
  issue_30 as (
    select
      l.item_id,
      upper(COALESCE(l.location, 'MAIN'::text)) as location,
      sum(COALESCE(l.qty_out, 0::numeric)) as issue_qty_30
    from
      inv_ledger l
    where
      l.ref_type = 'ISSUE'::text
      and l.created_at >= (now() - '30 days'::interval)
    group by
      l.item_id,
      (upper(COALESCE(l.location, 'MAIN'::text)))
  ),
  pr_first_receipt as (
    select
      l.item_id,
      upper(COALESCE(l.location, 'MAIN'::text)) as location,
      l.ref_no as pr_no,
      min(l.created_at) as first_receipt_at
    from
      inv_ledger l
    where
      l.ref_type = 'PR-RECEIPT'::text
      and COALESCE(l.qty_in, 0::numeric) > 0::numeric
      and l.ref_no is not null
    group by
      l.item_id,
      (upper(COALESCE(l.location, 'MAIN'::text))),
      l.ref_no
  ),
  lead_time as (
    select
      pr.item_id,
      r.location,
      avg(
        GREATEST(
          EXTRACT(
            epoch
            from
              r.first_receipt_at - pr.created_at
          ) / 86400.0,
          0::numeric
        )
      ) as avg_lead_time_days
    from
      inv_purchase_requests pr
      join pr_first_receipt r on r.pr_no = pr.pr_no
      and r.item_id = pr.item_id
    group by
      pr.item_id,
      r.location
  )
select
  s.item_id,
  s.location,
  s.last_movement_at,
  la.oldest_receipt_at,
  case
    when COALESCE(la.oldest_receipt_at, s.last_movement_at) is null then null::integer
    else ceil(
      EXTRACT(
        epoch
        from
          now() - COALESCE(la.oldest_receipt_at, s.last_movement_at)
      ) / 86400.0
    )::integer
  end as ageing_days,
  case
    when COALESCE(la.oldest_receipt_at, s.last_movement_at) is null then 'UNKNOWN'::text
    when ceil(
      EXTRACT(
        epoch
        from
          now() - COALESCE(la.oldest_receipt_at, s.last_movement_at)
      ) / 86400.0
    ) > 60::numeric then 'NON_MOVING'::text
    when ceil(
      EXTRACT(
        epoch
        from
          now() - COALESCE(la.oldest_receipt_at, s.last_movement_at)
      ) / 86400.0
    ) > 30::numeric then 'SLOW_MOVING'::text
    else 'FAST_MOVING'::text
  end as movement_class,
  COALESCE(la.next_batch_no, ''::text) as next_batch_no,
  COALESCE(la.batch_count, 0) as batch_count,
  round(COALESCE(i30.issue_qty_30, 0::numeric) / 30.0, 4) as avg_daily_consumption,
  round(COALESCE(lt.avg_lead_time_days, 7::numeric), 2) as avg_lead_time_days,
  1.2 as safety_factor,
  round(
    COALESCE(lt.avg_lead_time_days, 7::numeric) * (COALESCE(i30.issue_qty_30, 0::numeric) / 30.0) * 1.2,
    2
  ) as minimum_stock_level
from
  inv_stock_mv s
  left join lot_agg la on la.item_id = s.item_id
  and la.location = s.location
  left join issue_30 i30 on i30.item_id = s.item_id
  and i30.location = s.location
  left join lead_time lt on lt.item_id = s.item_id
  and lt.location = s.location;


create materialized view public.inv_stock_mv as
with
  ledger_balance as (
    select
      l.item_id,
      upper(COALESCE(l.location, 'MAIN'::text)) as location,
      sum(
        COALESCE(l.qty_in, 0::numeric) - COALESCE(l.qty_out, 0::numeric)
      ) as qty,
      max(l.created_at) as last_movement_at,
      sum(
        case
          when COALESCE(l.qty_in, 0::numeric) > 0::numeric then COALESCE(l.value, 0::numeric)
          else 0::numeric
        end
      ) as inbound_value,
      sum(
        case
          when COALESCE(l.qty_in, 0::numeric) > 0::numeric then COALESCE(l.qty_in, 0::numeric)
          else 0::numeric
        end
      ) as inbound_qty
    from
      inv_ledger l
    group by
      l.item_id,
      (upper(COALESCE(l.location, 'MAIN'::text)))
  ),
  lot_balance as (
    select
      lt.item_id,
      upper(COALESCE(lt.location, 'MAIN'::text)) as location,
      sum(
        case
          when COALESCE(lt.qty_available, 0::numeric) > 0::numeric then COALESCE(lt.qty_available, 0::numeric)
          else 0::numeric
        end
      ) as lot_qty,
      sum(
        case
          when COALESCE(lt.qty_available, 0::numeric) > 0::numeric then COALESCE(lt.qty_available, 0::numeric) * COALESCE(lt.rate, 0::numeric)
          else 0::numeric
        end
      ) as lot_value
    from
      inv_lots lt
    group by
      lt.item_id,
      (upper(COALESCE(lt.location, 'MAIN'::text)))
  )
select
  i.id as item_id,
  i.item_code,
  i.item_name,
  COALESCE(i.category, ''::text) as category,
  COALESCE(i.department, ''::text) as department,
  COALESCE(i.uom, ''::text) as uom,
  lb.location,
  lb.qty,
  case
    when COALESCE(lotb.lot_qty, 0::numeric) > 0::numeric then round(
      lotb.lot_value / NULLIF(lotb.lot_qty, 0::numeric),
      6
    )
    when COALESCE(lb.inbound_qty, 0::numeric) > 0::numeric then round(
      lb.inbound_value / NULLIF(lb.inbound_qty, 0::numeric),
      6
    )
    else 0::numeric
  end as avg_rate,
  round(
    lb.qty * case
      when COALESCE(lotb.lot_qty, 0::numeric) > 0::numeric then lotb.lot_value / NULLIF(lotb.lot_qty, 0::numeric)
      when COALESCE(lb.inbound_qty, 0::numeric) > 0::numeric then lb.inbound_value / NULLIF(lb.inbound_qty, 0::numeric)
      else 0::numeric
    end,
    2
  ) as value,
  lb.last_movement_at
from
  ledger_balance lb
  join inv_items i on i.id = lb.item_id
  left join lot_balance lotb on lotb.item_id = lb.item_id
  and lotb.location = lb.location
where
  lb.qty <> 0::numeric;

create view public.v_billing_dispatch_summary
with
  (security_invoker = on) as
with
  dispatch_totals as (
    select
      dr.so_line_id,
      COALESCE(sum(dr.dispatch_qty), 0::numeric)::numeric(18, 3) as dispatched_qty
    from
      dispatch_records dr
    where
      dr.so_line_id is not null
    group by
      dr.so_line_id
  ),
  dispatch_latest as (
    select distinct
      on (dr.so_line_id) dr.so_line_id,
      dr.dispatch_no as latest_dispatch_no,
      COALESCE(
        dr.dispatch_date::timestamp without time zone,
        dr.created_at
      ) as latest_dispatch_date,
      dr.transporter,
      dr.vehicle_no,
      dr.lr_no
    from
      dispatch_records dr
    where
      dr.so_line_id is not null
    order by
      dr.so_line_id,
      (
        COALESCE(
          dr.dispatch_date::timestamp without time zone,
          dr.created_at
        )
      ) desc,
      dr.created_at desc,
      dr.id desc
  )
select
  t.so_line_id,
  t.dispatched_qty,
  l.latest_dispatch_no,
  l.latest_dispatch_date,
  l.transporter,
  l.vehicle_no,
  l.lr_no
from
  dispatch_totals t
  left join dispatch_latest l on l.so_line_id = t.so_line_id;


create view public.v_billing_document_register
with
  (security_invoker = on) as
with
  base as (
    select
      inv.id::text as id,
      inv.invoice_no,
      inv.invoice_date,
      case
        when upper(COALESCE(inv.document_type, ''::text)) = any (array['INVOICE'::text, 'CHALLAN'::text]) then upper(inv.document_type)
        when COALESCE(inv.invoice_no, ''::text) ~~ 'DC-H/%'::text then 'CHALLAN'::text
        else 'INVOICE'::text
      end as document_type,
      COALESCE(inv.client_code, 'MANUAL'::text) as client_code,
      COALESCE(
        NULLIF(inv.bill_to_name, ''::text),
        NULLIF(inv.client_name, ''::text),
        c.client_name,
        inv.client_code,
        'MANUAL'::text
      ) as client_name,
      COALESCE(inv.status, 'DRAFT'::text) as status,
      COALESCE(inv.billing_mode, ''::text) as billing_mode,
      COALESCE(inv.remarks, ''::text) as remarks,
      COALESCE(inv.subtotal, 0::numeric)::numeric(18, 2) as subtotal,
      COALESCE(
        inv.tax_total,
        GREATEST(
          COALESCE(inv.grand_total, 0::numeric) - COALESCE(inv.subtotal, 0::numeric) - COALESCE(inv.freight, 0::numeric),
          0::numeric
        )
      )::numeric(18, 2) as tax_total,
      COALESCE(inv.grand_total, 0::numeric)::numeric(18, 2) as grand_total,
      inv.created_at,
      inv.posted_at
    from
      invoices inv
      left join clients c on c.client_code = inv.client_code
  ),
  line_meta as (
    select
      il.invoice_id::text as id,
      count(*)::integer as line_count,
      COALESCE(sum(il.qty), 0::numeric)::numeric(18, 3) as total_qty,
      min(
        upper(
          COALESCE(NULLIF(il.source_mode, ''::text), 'UNKNOWN'::text)
        )
      ) as derived_billing_mode
    from
      invoice_lines il
    group by
      il.invoice_id
  ),
  line_preview_raw as (
    select
      s.id,
      s.product_label,
      row_number() over (
        partition by
          s.id
        order by
          s.product_label
      ) as rn
    from
      (
        select distinct
          il.invoice_id::text as id,
          COALESCE(
            NULLIF(il.product_name, ''::text),
            NULLIF(il.product_code, ''::text),
            'Item'::text
          ) as product_label
        from
          invoice_lines il
      ) s
  ),
  line_preview as (
    select
      line_preview_raw.id,
      string_agg(
        line_preview_raw.product_label,
        ', '::text
        order by
          line_preview_raw.product_label
      ) as product_preview
    from
      line_preview_raw
    where
      line_preview_raw.rn <= 2
    group by
      line_preview_raw.id
  )
select
  b.id,
  b.invoice_no,
  b.invoice_date,
  b.document_type,
  b.client_code,
  b.client_name,
  b.status,
  COALESCE(
    NULLIF(b.billing_mode, ''::text),
    lm.derived_billing_mode,
    'UNKNOWN'::text
  ) as billing_mode,
  COALESCE(lm.total_qty, 0::numeric)::numeric(18, 3) as total_qty,
  COALESCE(lm.line_count, 0) as line_count,
  COALESCE(lp.product_preview, ''::text) as product_preview,
  b.remarks,
  b.subtotal,
  b.tax_total,
  b.grand_total,
  b.created_at,
  b.posted_at,
  COALESCE(
    b.posted_at,
    b.created_at::timestamp with time zone,
    b.invoice_date::timestamp without time zone::timestamp with time zone
  ) as sort_ts
from
  base b
  left join line_meta lm on lm.id = b.id
  left join line_preview lp on lp.id = b.id;


create view public.v_billing_invoice_usage_summary
with
  (security_invoker = on) as
with
  usage_base as (
    select
      il.so_line_id,
      il.qty,
      COALESCE(inv.invoice_no, ''::text) as invoice_no,
      COALESCE(inv.status, 'DRAFT'::text) as invoice_status
    from
      invoice_lines il
      left join invoices inv on inv.id = il.invoice_id
    where
      il.so_line_id is not null
  ),
  usage_totals as (
    select
      ub.so_line_id,
      COALESCE(sum(ub.qty), 0::numeric)::numeric(18, 3) as billed_qty,
      COALESCE(
        sum(
          case
            when upper(ub.invoice_status) = 'POSTED'::text then ub.qty
            else 0::numeric
          end
        ),
        0::numeric
      )::numeric(18, 3) as posted_qty,
      COALESCE(
        sum(
          case
            when upper(ub.invoice_status) <> 'POSTED'::text then ub.qty
            else 0::numeric
          end
        ),
        0::numeric
      )::numeric(18, 3) as draft_qty
    from
      usage_base ub
    group by
      ub.so_line_id
  ),
  usage_refs as (
    select
      x.so_line_id,
      string_agg(
        x.invoice_ref,
        ' || '::text
        order by
          x.invoice_ref
      ) as invoice_refs
    from
      (
        select distinct
          ub.so_line_id,
          (
            (ub.invoice_no || ' ['::text) || ub.invoice_status
          ) || ']'::text as invoice_ref
        from
          usage_base ub
        where
          ub.invoice_no <> ''::text
      ) x
    group by
      x.so_line_id
  )
select
  t.so_line_id,
  t.billed_qty,
  t.posted_qty,
  t.draft_qty,
  r.invoice_refs
from
  usage_totals t
  left join usage_refs r on r.so_line_id = t.so_line_id;

create view public.v_packing_board

  COALESCE(pack.packed_qty, 0::numeric) as packed_qty,
  COALESCE(prod.produced_qty, 0::numeric) - COALESCE(pack.packed_qty, 0::numeric) as balance_qty
from
  work_order_jobs woj
  left join prod on prod.wo_id = woj.wo_id
  and prod.rn = 1
  left join pack on pack.so_number = woj.so_number
  and pack.line_no = woj.line_no
where
  COALESCE(prod.produced_qty, 0::numeric) > COALESCE(pack.packed_qty, 0::numeric);


create view public.v_packing_queue_fast as
with
  artwork_category as (
    select
      a.artwork_no,
      max(
        case
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%FLEXO%'::text then 'FLEXO'::text
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%CORR%'::text then 'CORRUGATION'::text
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%OFFSET%'::text then 'OFFSET'::text
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%DIGITAL%'::text then 'DIGITAL'::text
          else null::text
        end
      ) as department_category
    from
      artworks a
    where
      COALESCE(a.artwork_no, ''::text) <> ''::text
    group by
      a.artwork_no
  ),
  job_base as (
    select
      woj.wo_id,
      wo_1.wo_number,
      wo_1.wo_date,
      woj.so_number,
      woj.line_no,
      max(NULLIF(woj.artwork_no, ''::text)) as artwork_no,
      max(NULLIF(woj.client_name, ''::text)) as client_name,
      string_agg(
        distinct woj.product_name,
        ' / '::text
        order by
          woj.product_name
      ) as product_name,
      COALESCE(
        max(ac.department_category),
        case
          when upper(COALESCE(max(woj.category), ''::text)) ~~ '%FLEXO%'::text then 'FLEXO'::text
          when upper(COALESCE(max(woj.category), ''::text)) ~~ '%CORR%'::text then 'CORRUGATION'::text
          when upper(COALESCE(max(woj.category), ''::text)) ~~ '%OFFSET%'::text then 'OFFSET'::text
          when upper(COALESCE(max(woj.category), ''::text)) ~~ '%DIGITAL%'::text then 'DIGITAL'::text
          else max(woj.category)
        end
      ) as department_category
    from
      work_order_jobs woj
      join work_orders wo_1 on wo_1.id = woj.wo_id
      left join artwork_category ac on ac.artwork_no = woj.artwork_no
    group by
      woj.wo_id,
      wo_1.wo_number,
      wo_1.wo_date,
      woj.so_number,
      woj.line_no
  ),
  final_production_routing as (
    select
      r.wo_id,
      r.id as routing_id,
      row_number() over (
        partition by
          r.wo_id
        order by
          r.sequence_no desc,
          r.id desc
      ) as rn
    from
      work_order_routing r
    where
      upper(
        TRIM(
          both
          from
            COALESCE(r.process_name, r.department, ''::text)
        )
      ) <> all (
        array[
          'PACKING'::text,
          'DISPATCH'::text,
          'CUTTING PRE'::text,
          'CUTTING POST'::text,
          'QC'::text
        ]
      )
  ),
  production_totals as (
    select
      pe.routing_id,
      sum(
        COALESCE(
          pe.ok_qty,
          GREATEST(
            COALESCE(pe.produced_qty, 0::numeric) - COALESCE(pe.rejected_qty, 0::numeric),
            0::numeric
          )
        )
      ) as produced_qty
    from
      production_entries pe
    group by
      pe.routing_id
  ),
  packing_totals as (
    select
      pr.so_line_id,
      pr.so_number,
      pr.line_no,
      (
        array_agg(
          pr.id
          order by
            pr.packed_at desc nulls last,
            pr.id
        )
      ) [1] as pack_id,
      sum(COALESCE(pr.packed_qty, 0::numeric)) as packed_qty,
      sum(COALESCE(pr.packed_weight_kg, 0::numeric)) as packed_weight_kg,
      bool_or(pr.ready_to_dispatch = true) as ready_to_dispatch
    from
      packing_records pr
    group by
      pr.so_line_id,
      pr.so_number,
      pr.line_no
  )
select
  pt.pack_id,
  so.id as so_id,
  sol.id as so_line_id,
  jb.so_number,
  jb.line_no,
  sol.product_code,
  COALESCE(sol.product_name, jb.product_name) as product_name,
  jb.client_name,
  COALESCE(sol.category, jb.department_category) as category,
  jb.department_category,
  COALESCE(sol.qty, 0::numeric) as order_qty,
  COALESCE(prod.produced_qty, 0::numeric) as produced_qty,
  COALESCE(pt.packed_qty, 0::numeric) as packed_qty,
  pt.ready_to_dispatch,
  sol.expected_delivery,
  sol.final_delivery,
  sol.division,
  sol.quote_no,
  sol.pm_code,
  sol.product_remarks,
  sol.prepress_remarks,
  so.remarks as so_remarks,
  jb.artwork_no,
  jb.wo_number,
  jb.wo_date,
  COALESCE(
    wo.snapshot_json ->> 'modeOfTransport'::text,
    ''::text
  ) as transport_mode,
  COALESCE(pt.packed_weight_kg, 0::numeric) as packed_weight_kg
from
  job_base jb
  join work_orders wo on wo.id = jb.wo_id
  left join final_production_routing fr on fr.wo_id = jb.wo_id
  and fr.rn = 1
  left join production_totals prod on prod.routing_id = fr.routing_id
  left join sales_orders so on so.so_number = jb.so_number
  left join sales_order_lines sol on sol.so_id = so.id
  and sol.line_no = jb.line_no
  left join packing_totals pt on pt.so_line_id = sol.id
where
  COALESCE(prod.produced_qty, 0::numeric) > 0::numeric;


create view public.v_production_board
with
  (security_invoker = on) as
with
  stage_production as (
    select
      production_entries.routing_id,
      sum(production_entries.produced_qty) as produced_total
    from
      production_entries
    group by
      production_entries.routing_id
  )
select
  wor.id as routing_id,
  wo.id as wo_id,
  wo.wo_number,
  ((wo.snapshot_json -> 'jobs'::text) -> 0) ->> 'artworkNo'::text as artwork_no,
  ((wo.snapshot_json -> 'jobs'::text) -> 0) ->> 'client'::text as client_name,
  wor.process_name,
  wor.sequence_no,
  wor.department,
  wor.planned_machine,
  case
    when wor.sequence_no = 1 then (
      (
        ((wo.snapshot_json -> 'jobs'::text) -> 0) ->> 'coreSheets'::text
      )::numeric
    ) + (
      (
        (wo.snapshot_json -> 'wastage'::text) ->> 'processSheets'::text
      )::numeric
    )
    else COALESCE(prev_stage_prod.produced_total, 0::numeric)
  end as planned_qty,
  COALESCE(curr_stage_prod.produced_total, 0::numeric) as produced_qty
from
  work_order_routing wor
  join work_orders wo on wo.id = wor.wo_id
  left join stage_production curr_stage_prod on curr_stage_prod.routing_id = wor.id
  left join work_order_routing prev_stage on prev_stage.wo_id = wor.wo_id
  and prev_stage.sequence_no = (wor.sequence_no - 1)
  left join stage_production prev_stage_prod on prev_stage_prod.routing_id = prev_stage.id;

create view public.v_production_stage_queue_fast as
with recursive
  artwork_category as (
    select
      a.artwork_no,
      max(
        case
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%FLEXO%'::text then 'FLEXO'::text
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%CORR%'::text then 'CORRUGATION'::text
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%OFFSET%'::text then 'OFFSET'::text
          when upper(COALESCE(a.product_type, ''::text)) ~~ '%DIGITAL%'::text then 'DIGITAL'::text
          else null::text
        end
      ) as department_category
    from
      artworks a
    where
      COALESCE(a.artwork_no, ''::text) <> ''::text
    group by
      a.artwork_no
  ),
  snapshot_metrics as (
    select
      wo.id as wo_id,
      COALESCE(
        (
          select
            sum(
              COALESCE(
                NULLIF(p.value ->> 'requiredQty'::text, ''::text)::numeric,
                NULLIF(p.value ->> 'sheetsWithWaste'::text, ''::text)::numeric,
                0::numeric
              )
            ) as sum
          from
            jsonb_array_elements(
              COALESCE(wo.snapshot_json -> 'papers'::text, '[]'::jsonb)
            ) p (value)
        ),
        0::numeric
      ) as paper_sheets_with_waste,
      COALESCE(
        (
          select
            sum(
              COALESCE(
                NULLIF(p.value ->> 'coreSheets'::text, ''::text)::numeric,
                NULLIF(p.value ->> 'sheets'::text, ''::text)::numeric,
                0::numeric
              )
            ) as sum
          from
            jsonb_array_elements(
              COALESCE(wo.snapshot_json -> 'papers'::text, '[]'::jsonb)
            ) p (value)
        ),
        0::numeric
      ) as paper_core_sheets,
      COALESCE(
        (
          select
            sum(
              COALESCE(
                NULLIF(j.value ->> 'coreSheets'::text, ''::text)::numeric,
                0::numeric
              )
            ) as sum
          from
            jsonb_array_elements(
              COALESCE(wo.snapshot_json -> 'jobs'::text, '[]'::jsonb)
            ) j (value)
        ),
        0::numeric
      ) as job_core_sheets,
      COALESCE(
        NULLIF(
          (wo.snapshot_json -> 'flexoDetails'::text) ->> 'totalRunningMeter'::text,
          ''::text
        )::numeric,
        NULLIF(
          (wo.snapshot_json -> 'flexoDetails'::text) ->> 'baseRunningMeter'::text,
          ''::text
        )::numeric,
        0::numeric
      ) as flexo_running_meter,
      COALESCE(
        NULLIF(
          (wo.snapshot_json -> 'flexoDetails'::text) ->> 'requiredKg'::text,
          ''::text
        )::numeric,
        0::numeric
      ) as flexo_required_kg,
      COALESCE(
        NULLIF(
          (wo.snapshot_json -> 'wastage'::text) ->> 'processSheets'::text,
          ''::text
        )::numeric,
        0::numeric
      ) as process_waste_sheets
    from
      work_orders wo
  ),
  stores_issue_plan as (
    select
      wo.id as wo_id,
      sum(
        case
          when upper(
            COALESCE(split_part(l.remarks, '|'::text, 5), ''::text)
          ) = any (array['SHEET'::text, 'SHEETS'::text]) then COALESCE(l.qty_out, 0::numeric)
          else 0::numeric
        end
      ) as issued_sheet_qty,
      sum(
        case
          when upper(
            COALESCE(split_part(l.remarks, '|'::text, 5), ''::text)
          ) = any (
            array[
              'RM'::text,
              'R.M.'::text,
              'RUNNING METER'::text,
              'RUNNING METERS'::text
            ]
          ) then COALESCE(l.qty_out, 0::numeric)
          else 0::numeric
        end
      ) as issued_rm_qty,
      sum(
        case
          when upper(
            COALESCE(split_part(l.remarks, '|'::text, 5), ''::text)
          ) = any (
            array[
              'KG'::text,
              'KGS'::text,
              'KILOGRAM'::text,
              'KILOGRAMS'::text
            ]
          ) then COALESCE(l.qty_out, 0::numeric)
          else 0::numeric
        end
      ) as issued_kg_qty
    from
      work_orders wo
      join inv_ledger l on l.ref_type = 'ISSUE'::text
      and l.ref_no = wo.wo_number
      left join inv_ledger rev on rev.ref_no = l.id::text
      and upper(COALESCE(rev.ref_type, ''::text)) ~~ 'REV-%'::text
    where
      rev.id is null
    group by
      wo.id
  ),
  stores_issue_plan_normalized as (
    select
      sip.wo_id,
      sip.issued_sheet_qty,
      sip.issued_rm_qty,
      sip.issued_kg_qty,
      case
        when sm.flexo_required_kg > 0::numeric
        and COALESCE(sip.issued_kg_qty, 0::numeric) > (sm.flexo_required_kg * 20::numeric)
        and (
          COALESCE(sip.issued_kg_qty, 0::numeric) / 1000::numeric
        ) <= (sm.flexo_required_kg * 2.5) then COALESCE(sip.issued_kg_qty, 0::numeric) / 1000::numeric
        else COALESCE(sip.issued_kg_qty, 0::numeric)
      end as effective_issued_kg_qty,
      case
        when sm.flexo_running_meter > 0::numeric
        and sm.flexo_required_kg > 0::numeric then case
          when case
            when sm.flexo_required_kg > 0::numeric
            and COALESCE(sip.issued_kg_qty, 0::numeric) > (sm.flexo_required_kg * 20::numeric)
            and (
              COALESCE(sip.issued_kg_qty, 0::numeric) / 1000::numeric
            ) <= (sm.flexo_required_kg * 2.5) then COALESCE(sip.issued_kg_qty, 0::numeric) / 1000::numeric * sm.flexo_running_meter / sm.flexo_required_kg
            else COALESCE(sip.issued_kg_qty, 0::numeric) * sm.flexo_running_meter / sm.flexo_required_kg
          end > (sm.flexo_running_meter * 5::numeric) then sm.flexo_running_meter
          else case
            when sm.flexo_required_kg > 0::numeric
            and COALESCE(sip.issued_kg_qty, 0::numeric) > (sm.flexo_required_kg * 20::numeric)
            and (
              COALESCE(sip.issued_kg_qty, 0::numeric) / 1000::numeric
            ) <= (sm.flexo_required_kg * 2.5) then COALESCE(sip.issued_kg_qty, 0::numeric) / 1000::numeric * sm.flexo_running_meter / sm.flexo_required_kg
            else COALESCE(sip.issued_kg_qty, 0::numeric) * sm.flexo_running_meter / sm.flexo_required_kg
          end
        end
        else 0::numeric
      end as issued_kg_equivalent_rm
    from
      stores_issue_plan sip
      join snapshot_metrics sm on sm.wo_id = sip.wo_id
  ),
  initial_routing as (
    select distinct
      on (r.wo_id) r.wo_id,
      COALESCE(
        NULLIF(r.process_name, ''::text),
        NULLIF(r.department, ''::text),
        'PROCESS'::text
      ) as first_process_name
    from
      work_order_routing r
    where
      upper(
        COALESCE(
          NULLIF(r.process_name, ''::text),
          NULLIF(r.department, ''::text),
          'PROCESS'::text
        )
      ) <> all (
        array[
          'PACKING'::text,
          'DISPATCH'::text,
          'CUTTING PRE'::text,
          'CUTTING POST'::text,
          'QC'::text
        ]
      )
    order by
      r.wo_id,
      r.sequence_no,
      r.id
  ),
  job_meta as (
    select
      woj.wo_id,
      wo.wo_number,
      wo.wo_date,
      string_agg(
        distinct woj.so_number,
        ', '::text
        order by
          woj.so_number
      ) as so_numbers,
      string_agg(
        distinct woj.line_no::text,
        ', '::text
        order by
          (woj.line_no::text)
      ) as line_nos,
      string_agg(
        distinct woj.artwork_no,
        ', '::text
        order by
          woj.artwork_no
      ) as artwork_nos,
      max(NULLIF(woj.client_name, ''::text)) as client_name,
      string_agg(
        distinct woj.product_name,
        ' / '::text
        order by
          woj.product_name
      ) as product_names,
      sum(COALESCE(woj.qty, 0::numeric)) as order_qty,
      max(sm.flexo_running_meter) as flexo_running_meter,
      max(NULLIF(woj.job_priority, ''::text)) as job_priority,
      max(woj.expected_delivery) as expected_delivery,
      COALESCE(
        NULLIF(sum(COALESCE(woj.ups, 0::numeric)), 0::numeric),
        case
          when COALESCE(
            NULLIF(max(sm.paper_core_sheets), 0::numeric),
            NULLIF(max(sm.job_core_sheets), 0::numeric),
            0::numeric
          ) > 0::numeric then sum(COALESCE(woj.qty, 0::numeric)) / COALESCE(
            NULLIF(max(sm.paper_core_sheets), 0::numeric),
            NULLIF(max(sm.job_core_sheets), 0::numeric)
          )
          else 1::numeric
        end
      ) as ups_factor,
      case
        when upper(
          COALESCE(
            (wo.snapshot_json -> 'jobDetails'::text) ->> 'type'::text,
            ''::text
          )
        ) ~~ '%FLEXO%'::text then 'FLEXO'::text
        else COALESCE(
          max(ac.department_category),
          case
            when upper(
              COALESCE(
                (wo.snapshot_json -> 'jobDetails'::text) ->> 'type'::text,
                ''::text
              )
            ) ~~ '%CORR%'::text then 'CORRUGATION'::text
            when upper(
              COALESCE(
                (wo.snapshot_json -> 'jobDetails'::text) ->> 'type'::text,
                ''::text
              )
            ) ~~ '%OFFSET%'::text then 'OFFSET'::text
            when upper(
              COALESCE(
                (wo.snapshot_json -> 'jobDetails'::text) ->> 'type'::text,
                ''::text
              )
            ) ~~ '%DIGITAL%'::text then 'DIGITAL'::text
            when upper(COALESCE(max(woj.category), ''::text)) ~~ '%FLEXO%'::text then 'FLEXO'::text
            when upper(COALESCE(max(woj.category), ''::text)) ~~ '%CORR%'::text then 'CORRUGATION'::text
            when upper(COALESCE(max(woj.category), ''::text)) ~~ '%OFFSET%'::text then 'OFFSET'::text
            when upper(COALESCE(max(woj.category), ''::text)) ~~ '%DIGITAL%'::text then 'DIGITAL'::text
            else null::text
          end
        )
      end as department_category,
      case
        when upper(
          COALESCE(
            (wo.snapshot_json -> 'jobDetails'::text) ->> 'type'::text,
            ''::text
          )
        ) ~~ '%FLEXO%'::text
        or upper(COALESCE(max(ir.first_process_name), ''::text)) ~~ '%INSPECTION%'::text
        or upper(COALESCE(max(ir.first_process_name), ''::text)) ~~ '%SLITTING%'::text
        or upper(COALESCE(max(ac.department_category), ''::text)) = 'FLEXO'::text
        or upper(COALESCE(max(woj.category), ''::text)) ~~ '%FLEXO%'::text then GREATEST(
          COALESCE(max(sipn.issued_rm_qty), 0::numeric) + COALESCE(max(sipn.issued_kg_equivalent_rm), 0::numeric),
          COALESCE(max(sm.flexo_running_meter), 0::numeric)
        )
        else COALESCE(
          NULLIF(max(sipn.issued_sheet_qty), 0::numeric),
          NULLIF(max(sm.paper_sheets_with_waste), 0::numeric),
          NULLIF(
            max(sm.paper_core_sheets) + max(sm.process_waste_sheets),
            0::numeric
          ),
          max(sm.job_core_sheets) + max(sm.process_waste_sheets)
        )
      end as stage1_plan_qty
    from
      work_order_jobs woj
      join work_orders wo on wo.id = woj.wo_id
      join snapshot_metrics sm on sm.wo_id = wo.id
      left join stores_issue_plan_normalized sipn on sipn.wo_id = wo.id
      left join initial_routing ir on ir.wo_id = wo.id
      left join artwork_category ac on ac.artwork_no = woj.artwork_no
    group by
      woj.wo_id,
      wo.wo_number,
      wo.wo_date,
      wo.snapshot_json
  ),
  production_totals as (
    select
      pe.routing_id,
      sum(
        GREATEST(COALESCE(pe.produced_qty, 0::numeric), 0::numeric)
      ) as gross_produced_qty,
      sum(
        COALESCE(
          pe.ok_qty,
          GREATEST(
            COALESCE(pe.produced_qty, 0::numeric) - COALESCE(pe.rejected_qty, 0::numeric),
            0::numeric
          )
        )
      ) as ok_produced_qty
    from
      production_entries pe
    group by
      pe.routing_id
  ),
  routing_base as (
    select
      r.id as routing_id,
      r.wo_id,
      jm.wo_number,
      jm.wo_date,
      jm.so_numbers,
      jm.line_nos,
      jm.artwork_nos,
      jm.client_name,
      jm.product_names,
      jm.order_qty,
      jm.flexo_running_meter,
      jm.job_priority,
      jm.expected_delivery,
      jm.ups_factor,
      jm.department_category,
      r.sequence_no,
      COALESCE(
        NULLIF(r.process_name, ''::text),
        NULLIF(r.department, ''::text),
        'PROCESS'::text
      ) as process_name,
      r.department,
      r.planned_machine,
      upper(COALESCE(r.status, ''::text)) as routing_status,
      COALESCE(pt.gross_produced_qty, 0::numeric) as gross_produced_qty,
      COALESCE(pt.ok_produced_qty, 0::numeric) as ok_produced_qty,
      jm.stage1_plan_qty,
      row_number() over (
        partition by
          r.wo_id
        order by
          r.sequence_no,
          r.id
      ) as rn
    from
      work_order_routing r
      join job_meta jm on jm.wo_id = r.wo_id
      left join production_totals pt on pt.routing_id = r.id
    where
      upper(
        COALESCE(
          NULLIF(r.process_name, ''::text),
          NULLIF(r.department, ''::text),
          'PROCESS'::text
        )
      ) <> all (
        array[
          'PACKING'::text,
          'DISPATCH'::text,
          'CUTTING PRE'::text,
          'CUTTING POST'::text,
          'QC'::text
        ]
      )
  ),
  stage_calc as (
    select
      rb.routing_id,
      rb.wo_id,
      rb.wo_number,
      rb.wo_date,
      rb.so_numbers,
      rb.line_nos,
      rb.artwork_nos,
      rb.client_name,
      rb.product_names,
      rb.order_qty,
      rb.flexo_running_meter,
      rb.job_priority,
      rb.expected_delivery,
      rb.ups_factor,
      rb.department_category,
      rb.sequence_no,
      rb.process_name,
      rb.department,
      rb.planned_machine,
      rb.routing_status,
      rb.gross_produced_qty as produced_qty,
      rb.ok_produced_qty,
      case
        when rb.department_category = 'FLEXO'::text
        and (
          upper(rb.process_name) ~~ '%DIE CUT%'::text
          or upper(rb.process_name) ~~ '%DIECUT%'::text
        )
        and upper(rb.process_name) !~~ '%DIE SHEET%'::text then GREATEST(rb.stage1_plan_qty, rb.gross_produced_qty)
        when rb.department_category = 'FLEXO'::text then GREATEST(rb.stage1_plan_qty, rb.gross_produced_qty)
        else rb.stage1_plan_qty
      end as planned_qty,
      GREATEST(
        case
          when rb.department_category = 'FLEXO'::text
          and (
            upper(rb.process_name) ~~ '%DIE CUT%'::text
            or upper(rb.process_name) ~~ '%DIECUT%'::text
          )
          and upper(rb.process_name) !~~ '%DIE SHEET%'::text then GREATEST(rb.stage1_plan_qty, rb.gross_produced_qty)
          when rb.department_category = 'FLEXO'::text then GREATEST(rb.stage1_plan_qty, rb.gross_produced_qty)
          else rb.stage1_plan_qty
        end - rb.gross_produced_qty,
        0::numeric
      ) as balance_qty,
      case
        when rb.department_category = 'FLEXO'::text
        and upper(rb.process_name) !~~ '%DIE CUT%'::text
        and upper(rb.process_name) !~~ '%DIECUT%'::text
        and upper(rb.process_name) !~~ '%DIE SHEET%'::text then rb.ok_produced_qty
        else 0::numeric
      end as rm_carry_qty,
      rb.rn
    from
      routing_base rb
    where
      rb.rn = 1
    union all
    select
      rb.routing_id,
      rb.wo_id,
      rb.wo_number,
      rb.wo_date,
      rb.so_numbers,
      rb.line_nos,
      rb.artwork_nos,
      rb.client_name,
      rb.product_names,
      rb.order_qty,
      rb.flexo_running_meter,
      rb.job_priority,
      rb.expected_delivery,
      rb.ups_factor,
      rb.department_category,
      rb.sequence_no,
      rb.process_name,
      rb.department,
      rb.planned_machine,
      rb.routing_status,
      rb.gross_produced_qty as produced_qty,
      rb.ok_produced_qty,
      case
        when rb.department_category = 'FLEXO'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%PRINT%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%LAMINAT%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%COAT%'::text then sc_1.ok_produced_qty
        when rb.department_category = 'FLEXO'::text
        and (
          upper(rb.process_name) ~~ '%INSPECTION%'::text
          or upper(rb.process_name) ~~ '%SLITTING%'::text
        ) then sc_1.rm_carry_qty
        when rb.department_category = 'FLEXO'::text
        and (
          upper(rb.process_name) ~~ '%DIE CUT%'::text
          or upper(rb.process_name) ~~ '%DIECUT%'::text
        )
        and upper(rb.process_name) !~~ '%DIE SHEET%'::text then sc_1.rm_carry_qty
        when upper(rb.process_name) ~~ '%DIE CUT%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%DIECUT%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%2 PLY%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%TWO PLY%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%PLY MAKING%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%CORRUGATION SHEET PASTING%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%SHEET PASTING%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%INSPECTION%'::text then sc_1.ok_produced_qty
        when upper(rb.process_name) ~~ '%SLITTING%'::text then sc_1.ok_produced_qty
        when upper(sc_1.process_name) ~~ '%PRINT%'::text then round(
          sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
        )
        when upper(sc_1.process_name) ~~ '%LAMINAT%'::text then round(
          sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
        )
        when upper(sc_1.process_name) ~~ '%COAT%'::text then round(
          sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
        )
        when upper(sc_1.process_name) ~~ '%DIE CUT%'::text then round(
          sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
        )
        when upper(sc_1.process_name) ~~ '%DIECUT%'::text then round(
          sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
        )
        else sc_1.ok_produced_qty
      end as planned_qty,
      GREATEST(
        case
          when rb.department_category = 'FLEXO'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%PRINT%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%LAMINAT%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%COAT%'::text then sc_1.ok_produced_qty
          when rb.department_category = 'FLEXO'::text
          and (
            upper(rb.process_name) ~~ '%INSPECTION%'::text
            or upper(rb.process_name) ~~ '%SLITTING%'::text
          ) then sc_1.rm_carry_qty
          when rb.department_category = 'FLEXO'::text
          and (
            upper(rb.process_name) ~~ '%DIE CUT%'::text
            or upper(rb.process_name) ~~ '%DIECUT%'::text
          )
          and upper(rb.process_name) !~~ '%DIE SHEET%'::text then sc_1.rm_carry_qty
          when upper(rb.process_name) ~~ '%DIE CUT%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%DIECUT%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%2 PLY%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%TWO PLY%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%PLY MAKING%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%CORRUGATION SHEET PASTING%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%SHEET PASTING%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%INSPECTION%'::text then sc_1.ok_produced_qty
          when upper(rb.process_name) ~~ '%SLITTING%'::text then sc_1.ok_produced_qty
          when upper(sc_1.process_name) ~~ '%PRINT%'::text then round(
            sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
          )
          when upper(sc_1.process_name) ~~ '%LAMINAT%'::text then round(
            sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
          )
          when upper(sc_1.process_name) ~~ '%COAT%'::text then round(
            sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
          )
          when upper(sc_1.process_name) ~~ '%DIE CUT%'::text then round(
            sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
          )
          when upper(sc_1.process_name) ~~ '%DIECUT%'::text then round(
            sc_1.ok_produced_qty * COALESCE(NULLIF(rb.ups_factor, 0::numeric), 1::numeric)
          )
          else sc_1.ok_produced_qty
        end - rb.gross_produced_qty,
        0::numeric
      ) as balance_qty,
      case
        when rb.department_category = 'FLEXO'::text then case
          when rb.ok_produced_qty > 0::numeric then rb.ok_produced_qty
          else sc_1.ok_produced_qty
        end
        else sc_1.rm_carry_qty
      end as rm_carry_qty,
      rb.rn
    from
      routing_base rb
      join stage_calc sc_1 on sc_1.wo_id = rb.wo_id
      and rb.rn = (sc_1.rn + 1)
  )
select
  routing_id,
  wo_id,
  wo_number,
  wo_date,
  so_numbers,
  line_nos,
  artwork_nos,
  client_name,
  product_names,
  department_category,
  process_name,
  department,
  planned_machine,
  sequence_no,
  expected_delivery,
  job_priority,
  planned_qty,
  produced_qty,
  balance_qty,
  case
    when routing_status = 'SHORT_CLOSED'::text then 'SHORT_CLOSED'::text
    when routing_status = 'HOLD'::text then 'HOLD'::text
    when planned_qty <= 0::numeric
    and produced_qty <= 0::numeric then 'PENDING'::text
    when balance_qty <= 0::numeric then 'COMPLETED'::text
    when produced_qty > 0::numeric then 'IN_PROGRESS'::text
    else 'PENDING'::text
  end as status
from
  stage_calc sc;

  psr.process_name,
  case
    when upper(COALESCE(psr.department_category, ''::text)) = 'DIGITAL'::text
    and upper(COALESCE(psr.process_name, ''::text)) ~~ '%DIGITAL%'::text then 'Printing'::text
    when upper(COALESCE(psr.department_category, ''::text)) = 'FLEXO'::text
    and (
      upper(COALESCE(psr.process_name, ''::text)) ~~ '%INSPECTION%'::text
      or upper(COALESCE(psr.process_name, ''::text)) ~~ '%SLITTING%'::text
    ) then 'Inspection/Slitting'::text
    when upper(COALESCE(psr.department_category, ''::text)) = 'FLEXO'::text
    and upper(COALESCE(psr.process_name, ''::text)) ~~ '%DIE SHEET%'::text then 'Flexo Die Sheeting'::text
    when upper(COALESCE(psr.department_category, ''::text)) = 'FLEXO'::text
    and (
      upper(COALESCE(psr.process_name, ''::text)) ~~ '%FLEXO PRINT%'::text
      or upper(COALESCE(psr.process_name, ''::text)) ~~ '%ONLINE GOLD FOIL%'::text
      or upper(COALESCE(psr.process_name, ''::text)) ~~ '%ONLINE SILVER FOIL%'::text
    ) then 'Flexo Printing'::text
    when upper(COALESCE(psr.department_category, ''::text)) = 'FLEXO'::text
    and (
      upper(COALESCE(psr.process_name, ''::text)) ~~ '%DIE CUT%'::text
      or upper(COALESCE(psr.process_name, ''::text)) ~~ '%DIECUT%'::text
    ) then 'Flexo Die Cutting Offline'::text
    when upper(COALESCE(psr.process_name, ''::text)) = 'CORRUGATION SHEET PASTING'::text then 'Sheet Pasting'::text
    else psr.process_name
  end as process_display_name,
  psr.department,
  psr.planned_machine,
  psr.sequence_no,
  COALESCE(jsc.expected_delivery, psr.expected_delivery) as expected_delivery,
  COALESCE(jsc.job_priority, psr.job_priority) as job_priority,
  psr.department_category,
  jsc.planned_qty,
  jsc.produced_qty,
  GREATEST(jsc.planned_qty - jsc.produced_qty, 0::numeric) as balance_qty,
  case
    when psr.routing_status = 'SHORT_CLOSED'::text then 'SHORT_CLOSED'::text
    when psr.routing_status = 'HOLD'::text then 'HOLD'::text
    when jsc.planned_qty <= 0::numeric
    and jsc.produced_qty <= 0::numeric then 'PENDING'::text
    when GREATEST(jsc.planned_qty - jsc.produced_qty, 0::numeric) <= 0::numeric then 'COMPLETED'::text
    when jsc.produced_qty > 0::numeric then 'IN_PROGRESS'::text
    else 'PENDING'::text
  end as status
from
  job_stage_calc jsc
  join post_split_routing psr on psr.routing_id = jsc.routing_id;

create view public.v_report_delivery_performance as
with
  dispatch_base as (
    select
      dr.so_line_id,
      max(dr.dispatch_date) as last_dispatch_date,
      sum(COALESCE(dr.dispatch_qty, 0::numeric)) as dispatched_qty
    from
      dispatch_records dr
    group by
      dr.so_line_id
  )
select
  so.so_number,
  sol.line_no,
  COALESCE(c.client_name, so.client_code) as client_name,
  sol.product_name,
  COALESCE(sol.final_delivery, sol.expected_delivery) as delivery_date,
  db.last_dispatch_date,
  COALESCE(sol.qty, 0::numeric) as order_qty,
  COALESCE(db.dispatched_qty, 0::numeric) as dispatched_qty,
  case
    when db.last_dispatch_date is null then 'PENDING'::text
    when COALESCE(sol.final_delivery, sol.expected_delivery) is null then 'DISPATCHED'::text
    when db.last_dispatch_date <= COALESCE(sol.final_delivery, sol.expected_delivery) then 'ON_TIME'::text
    else 'DELAYED'::text
  end as delivery_status,
  case
    when db.last_dispatch_date is null
    or COALESCE(sol.final_delivery, sol.expected_delivery) is null then 0
    else GREATEST(
      db.last_dispatch_date - COALESCE(sol.final_delivery, sol.expected_delivery),
      0
    )
  end as delay_days
from
  sales_orders so
  join sales_order_lines sol on sol.so_id = so.id
  left join clients c on c.client_code = so.client_code
  left join dispatch_base db on db.so_line_id = sol.id
where
  upper(COALESCE(so.status, 'OPEN'::text)) <> 'CANCELLED'::text;

create view public.v_report_planning_lines as
with
  artwork_ranked as (
    select
      a.so_id,
      a.line_no,
      a.id,
      a.artwork_no,
      a.product_type,
      a.plate_status,
      a.die_status,
      COALESCE(
        NULLIF(a.status, ''::text),
        case
          when COALESCE(a.artwork_no, ''::text) <> ''::text then 'PENDING_APPROVAL'::text
          else 'NO_ART'::text
        end
      ) as artwork_status,
      a.artwork_at,
      a.approved_at,
      row_number() over (
        partition by
          a.so_id,
          a.line_no
        order by
          a.approved_at desc nulls last,
          a.artwork_at desc nulls last,
          a.id desc
      ) as rn
    from
      artworks a
  ),
  artwork_base as (
    select
      artwork_ranked.so_id,
      artwork_ranked.line_no,
      artwork_ranked.artwork_no,
      artwork_ranked.product_type,
      artwork_ranked.plate_status,
      artwork_ranked.die_status,
      artwork_ranked.artwork_status,
      artwork_ranked.artwork_at,
      artwork_ranked.approved_at as artwork_approved_at
    from
      artwork_ranked
    where
      artwork_ranked.rn = 1
  ),
  wo_base as (
    select
      woj.so_number,
      woj.line_no,
      count(distinct woj.wo_id) as wo_count,
      string_agg(
        distinct wo.wo_number,
        ', '::text
        order by
          wo.wo_number
      ) as wo_numbers,
      min(wo.wo_date) as first_wo_at,
      max(wo.wo_date) as last_wo_at
    from
      work_order_jobs woj
      join work_orders wo on wo.id = woj.wo_id
    group by
      woj.so_number,
      woj.line_no
  ),
  routing_meta as (
    select
      woj.so_number,
      woj.line_no,
      count(distinct r.id) as routing_count,
      min(wo.wo_date) as routing_marked_at
    from
      work_order_jobs woj
      join work_orders wo on wo.id = woj.wo_id
      join work_order_routing r on r.wo_id = woj.wo_id
    group by
      woj.so_number,
      woj.line_no
  ),
  routing_steps as (
    select
      step_rows.so_number,
      step_rows.line_no,
      string_agg(
        step_rows.step_name,
        ' | '::text
        order by
          step_rows.sequence_no,
          step_rows.step_name
      ) as routing_steps
    from
      (
        select distinct
          woj.so_number,
          woj.line_no,
          COALESCE(r.sequence_no, 0) as sequence_no,
          COALESCE(
            NULLIF(r.process_name, ''::text),
            NULLIF(r.department, ''::text),
            'PROCESS'::text
          ) as step_name
        from
          work_order_jobs woj
          join work_order_routing r on r.wo_id = woj.wo_id
      ) step_rows
    group by
      step_rows.so_number,
      step_rows.line_no
  ),
  routing_base as (
    select
      rm.so_number,
      rm.line_no,
      rm.routing_count,
      rm.routing_marked_at,
      rs.routing_steps
    from
      routing_meta rm
      left join routing_steps rs on rs.so_number = rm.so_number
      and rs.line_no = rm.line_no
  ),
  stage_rows as (
    select
      woj.so_number,
      woj.line_no,
      COALESCE(psq.process_name, 'PROCESS'::text) as process_name,
      min(COALESCE(psq.sequence_no, 0)) as sequence_no,
      sum(COALESCE(psq.planned_qty, 0::numeric)) as planned_qty,
      sum(COALESCE(psq.produced_qty, 0::numeric)) as produced_qty,
      max(pe.entry_datetime) as stage_updated_at,
      case
        when bool_or(
          upper(COALESCE(psq.status, 'PENDING'::text)) = 'SHORT_CLOSED'::text
        ) then 'SHORT_CLOSED'::text
        when bool_or(
          upper(COALESCE(psq.status, 'PENDING'::text)) = 'HOLD'::text
        ) then 'HOLD'::text
        when bool_or(
          upper(COALESCE(psq.status, 'PENDING'::text)) = 'IN_PROGRESS'::text
        ) then 'IN_PROGRESS'::text
        when bool_and(
          upper(COALESCE(psq.status, 'PENDING'::text)) = 'COMPLETED'::text
        ) then 'COMPLETED'::text
        else 'PENDING'::text
      end as stage_status
    from
      work_order_jobs woj
      join v_production_stage_queue_fast psq on psq.wo_id = woj.wo_id
      left join production_entries pe on pe.routing_id = psq.routing_id
    group by
      woj.so_number,
      woj.line_no,
      (COALESCE(psq.process_name, 'PROCESS'::text))
  ),
  stage_base as (
    select
      stage_rows.so_number,
      stage_rows.line_no,
      string_agg(
        (
          (
            (
              (
                (
                  (stage_rows.process_name || ': '::text) || TRIM(
                    both
                    from
                      to_char(
                        COALESCE(stage_rows.produced_qty, 0::numeric),
                        'FM999999990.###'::text
                      )
                  )
                ) || '/'::text
              ) || TRIM(
                both
                from
                  to_char(
                    COALESCE(stage_rows.planned_qty, 0::numeric),
                    'FM999999990.###'::text
                  )
              )
            ) || ' ['::text
          ) || stage_rows.stage_status
        ) || ']'::text,
        ' | '::text
        order by
          stage_rows.sequence_no,
          stage_rows.process_name
      ) as stage_summary,
      max(stage_rows.stage_updated_at) as stage_last_updated_at
    from
      stage_rows
    group by
      stage_rows.so_number,
      stage_rows.line_no
  ),
  current_stage as (
    select distinct
      on (stage_rows.so_number, stage_rows.line_no) stage_rows.so_number,
      stage_rows.line_no,
      stage_rows.process_name as current_stage,
      stage_rows.stage_status as current_stage_status
    from
      stage_rows
    where
      upper(
        COALESCE(stage_rows.stage_status, 'PENDING'::text)
      ) <> 'COMPLETED'::text
    order by
      stage_rows.so_number,
      stage_rows.line_no,
      (
        case
          when upper(
            COALESCE(stage_rows.stage_status, 'PENDING'::text)
          ) = 'IN_PROGRESS'::text then 0
          when upper(
            COALESCE(stage_rows.stage_status, 'PENDING'::text)
          ) = 'PENDING'::text then 1
          when upper(
            COALESCE(stage_rows.stage_status, 'PENDING'::text)
          ) = 'HOLD'::text then 2
          when upper(
            COALESCE(stage_rows.stage_status, 'PENDING'::text)
          ) = 'SHORT_CLOSED'::text then 3
          else 4
        end
      ),
      stage_rows.sequence_no,
      stage_rows.process_name
  ),
  invoice_base as (
    select
      il.so_line_id,
      sum(COALESCE(il.qty, 0::numeric)) as billed_qty,
      count(distinct il.invoice_id) as invoice_count,
      min(inv.invoice_date) as first_invoice_date,
      max(inv.invoice_date) as last_invoice_date,
      max(inv.created_at) as last_billed_at,
      string_agg(
        distinct inv.invoice_no,
        ', '::text
        order by
          inv.invoice_no
      ) filter (
        where
          COALESCE(inv.invoice_no, ''::text) <> ''::text
      ) as invoice_nos,
      string_agg(
        distinct to_char(
          inv.invoice_date::timestamp with time zone,
          'YYYY-MM-DD'::text
        ),
        ', '::text
        order by
          (
            to_char(
              inv.invoice_date::timestamp with time zone,
              'YYYY-MM-DD'::text
            )
          )
      ) filter (
        where
          inv.invoice_date is not null
      ) as invoice_dates
    from
      invoice_lines il
      left join invoices inv on inv.id = il.invoice_id
    where
      il.so_line_id is not null
    group by
      il.so_line_id
  )
select

  COALESCE(ib.billed_qty, 0::numeric) as billed_qty,
  GREATEST(
    COALESCE(sol.qty, 0::numeric) - COALESCE(ib.billed_qty, 0::numeric),
    0::numeric
  ) as billing_pending_qty,
  COALESCE(ib.invoice_count, 0::bigint) as invoice_count,
  COALESCE(ib.invoice_nos, ''::text) as invoice_nos,
  COALESCE(ib.invoice_dates, ''::text) as invoice_dates,
  ib.first_invoice_date,
  ib.last_invoice_date,
  ib.last_billed_at,
  case
    when COALESCE(ib.billed_qty, 0::numeric) >= COALESCE(sol.qty, 0::numeric)
    and COALESCE(sol.qty, 0::numeric) > 0::numeric then 'CLOSED'::text
    else 'PENDING'::text
  end as billing_status
from
  sales_orders so
  join sales_order_lines sol on sol.so_id = so.id
  left join clients c on c.client_code = so.client_code
  left join artwork_base ab on ab.so_id = so.id
  and ab.line_no = sol.line_no
  left join wo_base wb on wb.so_number = so.so_number
  and wb.line_no = sol.line_no
  left join routing_base rb on rb.so_number = so.so_number
  and rb.line_no = sol.line_no
  left join stage_base sb on sb.so_number = so.so_number
  and sb.line_no = sol.line_no
  left join current_stage cs on cs.so_number = so.so_number
  and cs.line_no = sol.line_no
  left join v_report_planning_material_status ms on ms.so_number = so.so_number
  and ms.line_no::text = sol.line_no::text
  left join v_report_planning_tooling_status ts on ts.so_number = so.so_number
  and ts.line_no::text = sol.line_no::text
  left join invoice_base ib on ib.so_line_id = sol.id
where
  upper(COALESCE(so.status, 'OPEN'::text)) <> 'CANCELLED'::text;

create view public.v_report_unbilled_dispatch as
with
  dispatch_base as (
    select
      dr.so_line_id,
      max(dr.so_number) as so_number,
      max(dr.line_no) as line_no,
      max(dr.product_name) as product_name,
      max(dr.dispatch_date) as dispatch_date,
      sum(COALESCE(dr.dispatch_qty, 0::numeric)) as dispatched_qty
    from
      dispatch_records dr
    group by
      dr.so_line_id
  ),
  invoice_base as (
    select
      il.so_line_id,
      sum(COALESCE(il.qty, 0::numeric)) as billed_qty
    from
      invoice_lines il
    group by
      il.so_line_id
  )
select
  db.so_line_id,
  db.so_number,
  db.line_no,
  db.product_name,
  db.dispatch_date,
  db.dispatched_qty,
  COALESCE(ib.billed_qty, 0::numeric) as billed_qty,
  GREATEST(
    db.dispatched_qty - COALESCE(ib.billed_qty, 0::numeric),
    0::numeric
  ) as unbilled_qty
from
  dispatch_base db
  left join invoice_base ib on ib.so_line_id = db.so_line_id
where
  GREATEST(
    db.dispatched_qty - COALESCE(ib.billed_qty, 0::numeric),
    0::numeric
  ) > 0::numeric;

create view public.v_workorder_candidates
with
  (security_invoker = on) as
with
  service_only_items as (
    select
      v.item_code,
      v.item_name,
      v.category
    from
      (
        values
          (
            'SLC001'::text,
            'FLEXO PRINTING PLATE'::text,
            'FLEXO PRINTING PLATE'::text
          ),
          (
            'SLC002'::text,
            'OFFSET PRINTING PLATE'::text,
            'OFFSET PRINTING PLATE'::text
          ),
          ('SLC003'::text, 'DIE'::text, 'DIE'::text)
      ) v (item_code, item_name, category)
  ),
  artwork_ranked as (
    select
      a.so_id,
      a.line_no,
      a.artwork_no,
      a.product_type,
      a.sheet_length,
      a.sheet_width,
      a.sheet_ups,
      a.printing_colors,
      a.status,
      a.approved_at,
      row_number() over (
        partition by
          a.so_id,
          a.line_no
        order by
          (COALESCE(a.approved_at, a.artwork_at)) desc nulls last,
          a.id desc
      ) as rn
    from
      artworks a
  ),
  artwork_pick as (
    select
      artwork_ranked.so_id,
      artwork_ranked.line_no,
      artwork_ranked.artwork_no,
      artwork_ranked.product_type,
      artwork_ranked.sheet_length,
      artwork_ranked.sheet_width,
      artwork_ranked.sheet_ups,
      artwork_ranked.printing_colors,
      artwork_ranked.status,
      artwork_ranked.approved_at
    from
      artwork_ranked
    where
      artwork_ranked.rn = 1
  ),
  wo_totals as (
    select
      woj.so_number,
      woj.line_no,
      sum(COALESCE(woj.qty, 0::numeric)) as processed_qty,
      string_agg(
        distinct NULLIF(wo.wo_number, ''::text),
        ', '::text
        order by
          (NULLIF(wo.wo_number, ''::text))
      ) as wo_list
    from
      work_order_jobs woj
      left join work_orders wo on wo.id = woj.wo_id
    group by
      woj.so_number,
      woj.line_no
  ),
  base as (
    select
      so.id as so_id,
      sol.id as so_line_id,
      so.so_number,
      so.so_date,
      so.created_at as so_created_at,
      to_char(
        COALESCE(
          so.created_at,
          so.so_date::timestamp without time zone
        ),
        'HH24:MI'::text
      ) as so_time,
      so.sales_rep,
      so.client_code,
      COALESCE(c.client_name, so.client_code, ''::text) as client_name,
      sol.line_no,
      sol.product_code,
      sol.product_name,
      sol.category,
      sol.qty,
      sol.unit,
      so.po_number,
      sol.expected_delivery,
      sol.job_priority,
      sol.job_type,
      so.remarks as so_remarks,
      sol.product_remarks,
      sol.prepress_remarks as prepress_remark,
      sol.job_reference,
      upper(
        COALESCE(
          NULLIF(sol.accounts_status, ''::text),
          'PENDING'::text
        )
      ) as accounts_status,
      upper(
        COALESCE(
          NULLIF(sol.business_status, ''::text),
          'PENDING'::text
        )
      ) as business_status,
      ap.artwork_no,
      COALESCE(
        NULLIF(ap.product_type, ''::text),
        case
          when upper(COALESCE(sol.category, ''::text)) ~~ '%FLEXO%'::text then 'Flexo'::text
          when upper(COALESCE(sol.category, ''::text)) ~~ '%CORR%'::text then 'Corrugated'::text
          when upper(COALESCE(sol.category, ''::text)) ~~ '%DIGITAL%'::text then 'Digital'::text
          when upper(COALESCE(sol.category, ''::text)) ~~ '%OFFSET%'::text then 'Offset'::text
          else ''::text
        end
      ) as product_type,
      ap.sheet_length,
      ap.sheet_width,
      ap.sheet_ups,
      ap.printing_colors,
      case
        when ap.so_id is null then 'NO_ART'::text
        else upper(
          COALESCE(NULLIF(ap.status, ''::text), 'PENDING'::text)
        )
      end as artwork_status,
      ap.approved_at as artwork_approved_at,
      COALESCE(wt.processed_qty, 0::numeric) as processed_qty,
      GREATEST(
        COALESCE(sol.qty, 0::numeric) - COALESCE(wt.processed_qty, 0::numeric),
        0::numeric
      ) as remaining_qty,
      COALESCE(wt.wo_list, ''::text) as wo_list,
      case
        when upper(
          COALESCE(
            NULLIF(ap.product_type, ''::text),
            sol.category,
            ''::text
          )
        ) ~~ '%FLEXO%'::text then 'FLEXO'::text
        when upper(
          COALESCE(
            NULLIF(ap.product_type, ''::text),
            sol.category,
            ''::text
          )
        ) ~~ '%CORR%'::text then 'CORRUGATION'::text
        when upper(
          COALESCE(
            NULLIF(ap.product_type, ''::text),
            sol.category,
            ''::text
          )
        ) ~~ '%DIGITAL%'::text then 'DIGITAL'::text
        when upper(
          COALESCE(
            NULLIF(ap.product_type, ''::text),
            sol.category,
            ''::text
          )
        ) ~~ '%OFFSET%'::text then 'OFFSET'::text
        else upper(COALESCE(sol.category, ''::text))
      end as department_category
    from
      sales_order_lines sol
      join sales_orders so on so.id = sol.so_id
      left join clients c on c.client_code = so.client_code
      left join artwork_pick ap on ap.so_id = sol.so_id
      and ap.line_no = sol.line_no
      left join wo_totals wt on wt.so_number = so.so_number
      and wt.line_no = sol.line_no
    where
      upper(COALESCE(so.status, ''::text)) <> 'CANCELLED'::text
      and not (
        exists (
          select
            1
          from
            service_only_items s
          where
            s.item_code = upper(
              TRIM(
                both
                from
                  COALESCE(sol.product_code, ''::text)
              )
            )
            or s.item_name = upper(
              TRIM(
                both
                from
                  COALESCE(sol.product_name, ''::text)
              )
            )
            or s.category = upper(
              TRIM(
                both
                from
                  COALESCE(sol.category, ''::text)
              )
            )
        )
      )
  ),
  scored as (
    select
      b.so_id,
      b.so_line_id,
      b.so_number,
      b.so_date,
      b.so_created_at,
      b.so_time,
      b.sales_rep,
      b.client_code,
      b.client_name,
      b.line_no,
      b.product_code,
      b.product_name,
      b.category,
      b.qty,
      b.unit,
      b.po_number,
      b.expected_delivery,
      b.job_priority,
      b.job_type,
      b.so_remarks,
      b.product_remarks,
      b.prepress_remark,
      b.job_reference,
      b.accounts_status,
      b.business_status,
      b.artwork_no,
      b.product_type,
      b.sheet_length,
      b.sheet_width,
      b.sheet_ups,
      b.printing_colors,
      b.artwork_status,
      b.artwork_approved_at,
      b.processed_qty,
      b.remaining_qty,
      b.wo_list,
      b.department_category,
      concat_ws(
        ' / '::text,
        case
          when b.accounts_status = 'REJECTED'::text then 'Accounts'::text
          else null::text
        end,
        case
          when b.business_status = 'REJECTED'::text then 'Business'::text
          else null::text
        end,
        case
          when b.artwork_status = 'REJECTED'::text then 'Artwork'::text
          else null::text
        end
      ) as rejected_stage_list,
      concat_ws(
        ' / '::text,
        case
          when b.accounts_status <> 'APPROVED'::text then 'Accounts'::text
          else null::text
        end,
        case
          when b.business_status <> 'APPROVED'::text then 'Business'::text
          else null::text
        end,
        case
          when b.artwork_status <> 'APPROVED'::text then 'Artwork'::text
          else null::text
        end
      ) as pending_stage_list
    from
      base b
    where
      b.department_category !~~ '%FLEXO%'::text
  )
select
  so_id,
  so_line_id,
  so_number,
  so_date,
  so_created_at,
  so_time,
  sales_rep,
  client_code,
  client_name,
  line_no,
  product_code,
  product_name,
  category,
  department_category,
  qty,
  unit,
  po_number,
  expected_delivery,
  job_priority,
  job_type,
  so_remarks,
  product_remarks,
  prepress_remark,
  job_reference,
  accounts_status,
  business_status,
  artwork_no,
  product_type,
  sheet_length,
  sheet_width,
  sheet_ups,
  printing_colors,
  artwork_status,
  artwork_approved_at,
  processed_qty,
  remaining_qty,
  wo_list,
  case
    when rejected_stage_list <> ''::text then 'Rejected at '::text || rejected_stage_list
    when pending_stage_list <> ''::text then 'Pending at '::text || pending_stage_list
    else 'Approved'::text
  end as approval_stage,
  rejected_stage_list = ''::text
  and pending_stage_list = ''::text as can_create_wo
from
  scored;

        GREATEST(
          dc.working_hours - COALESCE(dc.downtime_hours, 0::numeric) - dc.total_make_ready_hours,
          0::numeric
        ),
        2
      ) as actual_running_hours
    from

  )
select
  production_date,
  machine_key,
  machine,
  COALESCE(operator_names, ''::text) as operator_names,
  round(working_hours, 2) as working_hours,
  COALESCE(downtime_reasons, ''::text) as downtime_reasons,
  round(COALESCE(downtime_hours, 0::numeric), 2) as downtime_hours,
  make_ready_jobs,
  standard_make_ready_minutes,
  total_make_ready_hours,
  actual_running_hours,
  target_hourly_output,
  round(
    actual_running_hours * case
      when machine_key = any (
        array[
          'FLEXOPRINT'::text,
          'FLEXODIE'::text,
          'FLEXOSLIT'::text
        ]
      ) then target_hourly_output * 60::numeric
      else target_hourly_output
    end,
    2
  ) as target_production,
  round(actual_production, 2) as actual_production,
  round(
    actual_production / NULLIF(actual_running_hours, 0::numeric) / case
      when machine_key = any (
        array[
          'FLEXOPRINT'::text,
          'FLEXODIE'::text,
          'FLEXOSLIT'::text
        ]
      ) then 60
      else 1
    end::numeric,
    2
  ) as actual_nop,
  round(
    actual_production / NULLIF(actual_running_hours, 0::numeric) / case
      when machine_key = any (
        array[
          'FLEXOPRINT'::text,
          'FLEXODIE'::text,
          'FLEXOSLIT'::text
        ]
      ) then 60
      else 1
    end::numeric - NULLIF(target_hourly_output, 0::numeric),
    2
  ) as nop_deviation,
  round(
    actual_production / NULLIF(actual_running_hours, 0::numeric) / case
      when machine_key = any (
        array[
          'FLEXOPRINT'::text,
          'FLEXODIE'::text,
          'FLEXOSLIT'::text
        ]
      ) then 60
      else 1
    end::numeric / NULLIF(target_hourly_output, 0::numeric) * 100.0,
    2
  ) as nop_achievement_pct,
  case
    when COALESCE(target_hourly_output, 0::numeric) <= 0::numeric then 'NO_TARGET'::text
    when (
      actual_production / NULLIF(actual_running_hours, 0::numeric) / case
        when machine_key = any (
          array[
            'FLEXOPRINT'::text,
            'FLEXODIE'::text,
            'FLEXOSLIT'::text
          ]
        ) then 60
        else 1
      end::numeric
    ) >= target_hourly_output then 'ON_TARGET'::text
    else 'BELOW_TARGET'::text
  end as performance_status,
  last_completion_at
from
  final_calc;

create view public.inv_wo_issue_requirement_v as
with
  material_rows as (
    select
      wom.wo_id,
      wom.material_key,
      max(wom.item_name) as item_name,
      max(wom.gsm) as gsm,
      max(wom.deckle) as deckle,
      max(wom.cut_size) as cut_size,
      max(wom.uom) as uom,
      sum(COALESCE(wom.required_qty, 0::numeric)) as required_qty
    from
      work_order_materials wom
    group by
      wom.wo_id,
      wom.material_key
  ),
  issued_rows as (
    select
      il.ref_no as wo_number,
      il.remarks as material_key,
      sum(COALESCE(il.qty_out, 0::numeric)) as issued_qty
    from
      inv_ledger il
    where
      upper(COALESCE(il.ref_type, ''::text)) = 'ISSUE'::text
      and COALESCE(il.qty_out, 0::numeric) > 0::numeric
      and COALESCE(il.remarks, ''::text) <> ''::text
      and not (
        exists (
          select
            1
          from
            inv_ledger rev
          where
            rev.ref_no = il.id::text
            and upper(COALESCE(rev.ref_type, ''::text)) ~~ 'REV-%'::text
        )
      )
    group by
      il.ref_no,
      il.remarks
  ),
  wo_base as (
    select
      wo.id,
      wo.wo_number,
      wo.status,
      wo.snapshot_json,
      COALESCE(
        NULLIF(
          wo.snapshot_json #>> '{jobDetails,department}'::text[],
          ''::text
        ),
        NULLIF(
          wo.snapshot_json #>> '{department}'::text[],
          ''::text
        ),
        NULLIF(
          wo.snapshot_json #>> '{departmentCategory}'::text[],
          ''::text
        ),
        NULLIF(
          wo.snapshot_json #>> '{jobDetails,category}'::text[],
          ''::text
        ),
        NULLIF(
          wo.snapshot_json #>> '{jobDetails,type}'::text[],
          ''::text
        ),
        ''::text
      ) as raw_department
    from
      work_orders wo
    where
      COALESCE(upper(wo.status), ''::text) <> 'CLOSED'::text
  ),
  prepared as (
    select
      wb.id as wo_id,
      wb.wo_number,
      wb.status,
      case
        when upper(wb.raw_department) ~~ '%FLEXO%'::text then 'Flexo'::text
        when upper(wb.raw_department) ~~ '%OFFSET%'::text then 'Offset'::text
        when upper(wb.raw_department) = 'FLAT'::text then 'Offset'::text
        when upper(wb.raw_department) ~~ '%FLAT%'::text then 'Offset'::text
        when upper(wb.raw_department) ~~ '%DIGITAL%'::text then 'Digital'::text
        when upper(wb.raw_department) ~~ '%CORR%'::text then 'Corrugation'::text
        else 'Offset'::text
      end as issue_department,
      mr.material_key,
      COALESCE(
        NULLIF(mr.item_name, ''::text),
        NULLIF(
          split_part(mr.material_key, '|'::text, 1),
          ''::text
        ),
        ''::text
      ) as item_label,
      mr.gsm,
      COALESCE(
        NULLIF(mr.deckle, 0::numeric),
        NULLIF(
          NULLIF(
            split_part(mr.material_key, '|'::text, 3),
            ''::text
          )::numeric,
          0::numeric
        ),
        0::numeric
      ) as deckle_mm,
      COALESCE(
        NULLIF(mr.cut_size, 0::numeric),
        NULLIF(
          NULLIF(
            split_part(mr.material_key, '|'::text, 4),
            ''::text
          )::numeric,
          0::numeric
        ),
        0::numeric
      ) as cut_mm,
      COALESCE(
        NULLIF(mr.uom, ''::text),
        NULLIF(
          split_part(mr.material_key, '|'::text, 5),
          ''::text
        ),
        ''::text
      ) as uom,
      COALESCE(mr.required_qty, 0::numeric) as required_qty,
      COALESCE(ir.issued_qty, 0::numeric) as issued_qty
    from
      wo_base wb
      join material_rows mr on mr.wo_id = wb.id
      left join issued_rows ir on ir.wo_number = wb.wo_number
      and ir.material_key = mr.material_key
  )
select
  wo_id,
  wo_number,
  status,
  issue_department,
  material_key,
  item_label,
  gsm,
  deckle_mm,
  cut_mm,
  uom,
  required_qty,
  issued_qty,
  GREATEST(required_qty - issued_qty, 0::numeric) as remaining_qty,
  case
    when GREATEST(required_qty - issued_qty, 0::numeric) > 0.0001 then 'PENDING'::text
    else 'ISSUED'::text
  end as line_status
from
  prepared;

create view public.v_fg_stock_available as
with
  invoice_usage as (
    select
      il.so_line_id,
      sum(COALESCE(il.qty, 0::numeric)) as billed_qty
    from
      invoice_lines il
      left join invoices i on i.id = il.invoice_id
    where
      COALESCE(upper(i.status), ''::text) <> 'CANCELLED'::text
    group by
      il.so_line_id
  ),
  fg_adjustments as (
    select
      fg_stock_adjustments.pack_id,
      fg_stock_adjustments.opening_id,
      fg_stock_adjustments.so_line_id,
      sum(
        COALESCE(fg_stock_adjustments.adjustment_qty, 0::numeric)
      ) as adjusted_qty,
      (
        array_agg(
          fg_stock_adjustments.reason
          order by
            fg_stock_adjustments.adjustment_date desc nulls last,
            fg_stock_adjustments.created_at desc
        )
      ) [1] as latest_adjustment_reason,
      (
        array_agg(
          fg_stock_adjustments.remarks
          order by
            fg_stock_adjustments.adjustment_date desc nulls last,
            fg_stock_adjustments.created_at desc
        )
      ) [1] as latest_adjustment_remarks,
      max(
        COALESCE(
          fg_stock_adjustments.adjustment_date,
          fg_stock_adjustments.created_at
        )
      ) as latest_adjustment_at
    from
      fg_stock_adjustments
    group by
      fg_stock_adjustments.pack_id,
      fg_stock_adjustments.opening_id,
      fg_stock_adjustments.so_line_id
  ),
  packed_fg as (
    select
      'PACKED::'::text || pr.id::text as row_id,
      'PACKED'::text as source_type,
      pr.id as pack_id,
      null::uuid as opening_id,
      pr.so_id,
      pr.so_line_id,
      pr.so_number,
      pr.line_no::text as line_no,
      so.client_code,
      COALESCE(c.client_name, so.client_code) as client_name,
      COALESCE(sol.product_code, pr.product_code) as product_code,
      COALESCE(sol.product_name, pr.product_name) as product_name,
      COALESCE(sol.unit, 'Pcs'::text) as uom,
      0::numeric as opening_qty,
      COALESCE(pr.packed_qty, 0::numeric) as packed_qty,
      COALESCE(iu.billed_qty, 0::numeric) as billed_qty,
      COALESCE(adj.adjusted_qty, 0::numeric) as adjusted_qty,
      GREATEST(
        COALESCE(pr.packed_qty, 0::numeric) - COALESCE(iu.billed_qty, 0::numeric) - COALESCE(adj.adjusted_qty, 0::numeric),
        0::numeric
      ) as available_qty,
      GREATEST(
        COALESCE(pr.packed_qty, 0::numeric) - COALESCE(iu.billed_qty, 0::numeric) - COALESCE(adj.adjusted_qty, 0::numeric),
        0::numeric
      ) as billable_qty,
      pr.packed_at::timestamp with time zone as stock_date,
      adj.latest_adjustment_reason,
      adj.latest_adjustment_remarks,
      adj.latest_adjustment_at,
      sol.product_remarks as remarks
    from
      packing_records pr
      left join sales_orders so on so.id = pr.so_id
      left join clients c on c.client_code = so.client_code
      left join sales_order_lines sol on sol.id = pr.so_line_id
      left join invoice_usage iu on iu.so_line_id = pr.so_line_id
      left join fg_adjustments adj on adj.pack_id = pr.id
    where
      COALESCE(pr.packed_qty, 0::numeric) > 0::numeric
  ),
  opening_fg as (
    select
      'OPENING::'::text || os.id::text as row_id,
      'OPENING'::text as source_type,
      null::uuid as pack_id,
      os.id as opening_id,
      null::uuid as so_id,
      null::uuid as so_line_id,
      ''::text as so_number,
      ''::text as line_no,
      os.client_code,
      COALESCE(os.client_name, os.client_code) as client_name,
      os.product_code,
      os.product_name,
      COALESCE(os.uom, 'Pcs'::text) as uom,
      COALESCE(os.opening_qty, 0::numeric) as opening_qty,
      0::numeric as packed_qty,
      0::numeric as billed_qty,
      COALESCE(adj.adjusted_qty, 0::numeric) as adjusted_qty,
      GREATEST(
        COALESCE(os.opening_qty, 0::numeric) - COALESCE(adj.adjusted_qty, 0::numeric),
        0::numeric
      ) as available_qty,
      0::numeric as billable_qty,
      os.opening_date as stock_date,
      adj.latest_adjustment_reason,
      adj.latest_adjustment_remarks,
      adj.latest_adjustment_at,
      os.remarks
    from
      fg_opening_stock os
      left join fg_adjustments adj on adj.opening_id = os.id
  )
select
  packed_fg.row_id,
  packed_fg.source_type,
  packed_fg.pack_id,
  packed_fg.opening_id,
  packed_fg.so_id,
  packed_fg.so_line_id,
  packed_fg.so_number,
  packed_fg.line_no,
  packed_fg.client_code,
  packed_fg.client_name,
  packed_fg.product_code,
  packed_fg.product_name,
  packed_fg.uom,
  packed_fg.opening_qty,
  packed_fg.packed_qty,
  packed_fg.billed_qty,
  packed_fg.adjusted_qty,
  packed_fg.available_qty,
  packed_fg.billable_qty,
  packed_fg.stock_date,
  packed_fg.latest_adjustment_reason,
  packed_fg.latest_adjustment_remarks,
  packed_fg.latest_adjustment_at,
  packed_fg.remarks
from
  packed_fg
union all
select
  opening_fg.row_id,
  opening_fg.source_type,
  opening_fg.pack_id,
  opening_fg.opening_id,
  opening_fg.so_id,
  opening_fg.so_line_id,
  opening_fg.so_number,
  opening_fg.line_no,
  opening_fg.client_code,

  opening_fg.product_code,

  opening_fg.uom,
  opening_fg.opening_qty,
  opening_fg.packed_qty,
  opening_fg.billed_qty,
  opening_fg.adjusted_qty,
  opening_fg.available_qty,
  opening_fg.billable_qty,
  opening_fg.stock_date,
  opening_fg.latest_adjustment_reason,
  opening_fg.latest_adjustment_remarks,
  opening_fg.latest_adjustment_at,
  opening_fg.remarks
from
  opening_fg;

  and business_status = 'APPROVED'::text
  and artwork_status = 'APPROVED'::text as can_create_wo,

    when accounts_status = 'REJECTED'::text
    or business_status = 'REJECTED'::text
    or artwork_status = 'REJECTED'::text then 'Rejected at '::text || concat_ws(
      ' / '::text,
      case
        when accounts_status = 'REJECTED'::text then 'Accounts'::text
        else null::text
      end,
      case
        when business_status = 'REJECTED'::text then 'Business'::text
        else null::text
      end,
      case
        when artwork_status = 'REJECTED'::text then 'Artwork'::text

      ' / '::text,

        else null::text
      end,
      case
        when business_status <> 'APPROVED'::text then 'Business'::text
        else null::text
      end,
      case
        when artwork_status <> 'APPROVED'::text then 'Artwork'::text
        else null::text
      end
    )
    else 'Approved'::text
  end as approval_stage,
  wo_list,
  wo_no,
  processed_teeth,
  processed_ups,
  processed_item_details,
  po_number,
  expected_delivery,
  job_priority,
  so_remarks,
  product_remarks,
  prepress_remark,
  prepress_remarks,
  job_reference,
  wo_date,
  wo_time,
  wo_date_time,
  line_status,
  is_line_active
from

where
  is_flexo_wo_candidate = true
  and (
    is_line_active = true

  );

  job_status_base as (
    select
      woj.wo_id,
      woj.so_number,

        when (

        )
        and (
          upper(COALESCE(sol.status, 'OPEN'::text)) <> all (array['CLOSED'::text, 'CANCELLED'::text])
        ) then true
        else false
      end as is_open_job
    from
      work_order_jobs woj
      left join sales_orders so on so.so_number = woj.so_number
      left join sales_order_lines sol on sol.so_id = so.id
      and sol.line_no::text = woj.line_no::text

      count(*) as total_job_count,
      count(*) filter (

      ) as open_job_count,
      COALESCE(sum(job_status_base.job_qty), 0::numeric) as total_job_qty,
      COALESCE(
        sum(job_status_base.job_qty) filter (
          where
            job_status_base.is_open_job
        ),
        0::numeric
      ) as open_job_qty,
      string_agg(
        distinct job_status_base.so_number,
        ', '::text
        order by
          job_status_base.so_number
      ) filter (
        where
          job_status_base.is_open_job
          and COALESCE(job_status_base.so_number, ''::text) <> ''::text
      ) as open_so_numbers,
      string_agg(
        distinct job_status_base.line_no::text,
        ', '::text
        order by
          (job_status_base.line_no::text)
      ) filter (
        where
          job_status_base.is_open_job
          and job_status_base.line_no is not null
      ) as open_line_nos,
      string_agg(
        distinct job_status_base.artwork_no,
        ', '::text
        order by
          job_status_base.artwork_no
      ) filter (
        where
          job_status_base.is_open_job
          and COALESCE(job_status_base.artwork_no, ''::text) <> ''::text
      ) as open_artwork_nos,

      COALESCE(ms.target_output, 0::numeric) as target_output,
      case
        when qn.machine_key = any (
          array[
            'FLEXOPRINT'::text,
            'FLEXODIE'::text,
            'FLEXOSLIT'::text
          ]
        ) then 'meter/min'::text
        else 'qty/hour'::text
      end as target_unit,
      case
        when qn.machine_key = any (
          array[
            'FLEXOPRINT'::text,
            'FLEXODIE'::text,
            'FLEXOSLIT'::text
          ]
        ) then COALESCE(ms.target_output, 0::numeric) * 60::numeric
        else COALESCE(ms.target_output, 0::numeric)
      end as target_hourly_capacity,
      case
        when COALESCE(qn.balance_qty, 0::numeric) > 0::numeric then COALESCE(qn.balance_qty, 0::numeric)
        else COALESCE(qn.planned_qty, 0::numeric)
      end as base_load_qty
    from
      queue_normalized qn
      left join machine_standards ms on ms.machine_key = qn.machine_key
  ),
  load_unit_calc as (
    select
      lb.routing_id,
      lb.wo_id,
      lb.wo_number,
      lb.wo_date,
      lb.so_numbers,
      lb.line_nos,
      lb.artwork_nos,
      lb.client_name,
      lb.product_names,
      lb.department_category,
      lb.process_name,
      lb.department,
      lb.planned_machine,
      lb.sequence_no,
      lb.expected_delivery,
      lb.job_priority,
      lb.planned_qty,
      lb.produced_qty,
      lb.balance_qty,
      lb.status,
      lb.plan_unit,
      lb.flexo_total_rm,
      lb.flexo_total_job_qty,
      lb.raw_machine,
      lb.raw_machine_upper,
      lb.raw_process_upper,
      lb.raw_category_upper,
      lb.machine_key,
      lb.normalized_machine,
      lb.standard_make_ready_minutes,
      lb.target_output,
      lb.target_unit,
      lb.target_hourly_capacity,
      lb.base_load_qty,
      case
        when lb.machine_key = any (
          array[
            'FLEXOPRINT'::text,
            'FLEXODIE'::text,
            'FLEXOSLIT'::text
          ]
        ) then 'meter'::text
        when upper(COALESCE(lb.plan_unit, ''::text)) <> ''::text then lower(lb.plan_unit)
        else 'qty'::text
      end as load_qty_unit,
      case
        when (
          lb.machine_key = any (
            array[
              'FLEXOPRINT'::text,
              'FLEXODIE'::text,
              'FLEXOSLIT'::text
            ]
          )
        )
        and COALESCE(lb.flexo_total_rm, 0::numeric) > 0::numeric
        and COALESCE(lb.flexo_total_job_qty, 0::numeric) > 0::numeric
        and (
          upper(COALESCE(lb.plan_unit, ''::text)) ~~ '%PCS%'::text
          or upper(COALESCE(lb.plan_unit, ''::text)) ~~ '%UNIT%'::text
          or upper(COALESCE(lb.plan_unit, ''::text)) ~~ '%SHEET%'::text
          or COALESCE(lb.base_load_qty, 0::numeric) > COALESCE(lb.flexo_total_rm, 0::numeric)
          or lb.machine_key = 'FLEXODIE'::text
        ) then round(
          COALESCE(lb.base_load_qty, 0::numeric) * lb.flexo_total_rm / NULLIF(lb.flexo_total_job_qty, 0::numeric),
          2
        )
        else COALESCE(lb.base_load_qty, 0::numeric)
      end as load_qty
    from
      load_base lb
  ),
  load_calc as (
    select
      luc.routing_id,
      luc.wo_id,
      luc.wo_number,
      luc.wo_date,
      luc.so_numbers,
      luc.line_nos,
      luc.artwork_nos,
      luc.client_name,
      luc.product_names,
      luc.department_category,
      luc.process_name,
      luc.department,
      luc.planned_machine,
      luc.sequence_no,
      luc.expected_delivery,
      luc.job_priority,
      luc.planned_qty,
      luc.produced_qty,
      luc.balance_qty,
      luc.status,
      luc.plan_unit,
      luc.flexo_total_rm,
      luc.flexo_total_job_qty,
      luc.raw_machine,
      luc.raw_machine_upper,
      luc.raw_process_upper,
      luc.raw_category_upper,
      luc.machine_key,
      luc.normalized_machine,
      luc.standard_make_ready_minutes,
      luc.target_output,
      luc.target_unit,
      luc.target_hourly_capacity,
      luc.base_load_qty,
      luc.load_qty_unit,
      luc.load_qty,
      round(
        COALESCE(luc.standard_make_ready_minutes, 0::numeric) / 60.0,
        2
      ) as make_ready_hours,
      round(
        COALESCE(luc.load_qty, 0::numeric) / NULLIF(luc.target_hourly_capacity, 0::numeric),
        2
      ) as running_load_hours
    from
      load_unit_calc luc
  )
select
  routing_id,
  wo_id,
  wo_number,
  wo_date,
  so_numbers,
  line_nos,
  artwork_nos,
  client_name,
  product_names,
  department_category,
  process_name,
  department,
  planned_machine,
  machine_key,
  normalized_machine as machine,
  expected_delivery,
  planned_qty,
  produced_qty,
  balance_qty,
  plan_unit,
  base_load_qty,
  load_qty,
  load_qty_unit,
  flexo_total_rm,
  flexo_total_job_qty,
  status,
  standard_make_ready_minutes,
  target_output,
  target_unit,
  round(target_hourly_capacity, 2) as target_hourly_capacity,
  make_ready_hours,
  COALESCE(running_load_hours, 0::numeric) as running_load_hours,
  round(
    COALESCE(running_load_hours, 0::numeric) + COALESCE(make_ready_hours, 0::numeric),
    2
  ) as total_load_hours,
  round(
    (
      COALESCE(running_load_hours, 0::numeric) + COALESCE(make_ready_hours, 0::numeric)
    ) / 8.0,
    2
  ) as load_days,
  case
    when COALESCE(target_hourly_capacity, 0::numeric) <= 0::numeric then 'NO_TARGET'::text
    else 'OPEN_LOAD'::text
  end as load_status
from
  load_calc;



create view public.v_report_planning_material_status as
with
  job_qty as (
    select
      woj.wo_id,
      woj.so_number,
      woj.line_no,
      sum(COALESCE(woj.qty, 0::numeric)) as line_qty
    from
      work_order_jobs woj
    group by
      woj.wo_id,
      woj.so_number,
      woj.line_no
  ),
  job_share as (
    select
      jq.wo_id,
      jq.so_number,
      jq.line_no,
      jq.line_qty,
      case
        when sum(jq.line_qty) over (
          partition by
            jq.wo_id
        ) > 0::numeric then jq.line_qty / NULLIF(
          sum(jq.line_qty) over (
            partition by
              jq.wo_id
          ),
          0::numeric
        )
        else 1.0 / NULLIF(
          count(*) over (
            partition by
              jq.wo_id
          ),
          0
        )::numeric
      end as share_ratio
    from
      job_qty jq
  ),
  material_rows as (
    select
      wom.wo_id,
      wom.material_key,
      COALESCE(
        NULLIF(max(wom.item_name), ''::text),
        NULLIF(
          split_part(wom.material_key, '|'::text, 1),
          ''::text
        ),
        ''::text
      ) as item_name,
      max(wom.uom) as uom,
      sum(COALESCE(wom.required_qty, 0::numeric)) as required_qty
    from
      work_order_materials wom
    group by
      wom.wo_id,
      wom.material_key
  ),
  issued_rows as (
    select
      wo.id as wo_id,
      il.remarks as material_key,
      sum(COALESCE(il.qty_out, 0::numeric)) as issued_qty
    from
      inv_ledger il
      join work_orders wo on wo.wo_number = il.ref_no
    where
      upper(COALESCE(il.ref_type, ''::text)) = 'ISSUE'::text
      and COALESCE(il.qty_out, 0::numeric) > 0::numeric
      and COALESCE(il.remarks, ''::text) <> ''::text
      and not (
        exists (
          select
            1
          from
            inv_ledger rev
          where
            rev.ref_no = il.id::text
            and upper(COALESCE(rev.ref_type, ''::text)) ~~ 'REV-%'::text
        )
      )
    group by
      wo.id,
      il.remarks
  ),
  line_materials as (
    select
      js.so_number,
      js.line_no,
      mr.material_key,
      mr.item_name,
      mr.uom,
      round(
        COALESCE(mr.required_qty, 0::numeric) * COALESCE(js.share_ratio, 1::numeric),
        3
      ) as required_qty,
      round(
        COALESCE(ir.issued_qty, 0::numeric) * COALESCE(js.share_ratio, 1::numeric),
        3
      ) as issued_qty,
      round(
        GREATEST(
          COALESCE(mr.required_qty, 0::numeric) - COALESCE(ir.issued_qty, 0::numeric),
          0::numeric
        ) * COALESCE(js.share_ratio, 1::numeric),
        3
      ) as pending_qty
    from
      job_share js
      join material_rows mr on mr.wo_id = js.wo_id
      left join issued_rows ir on ir.wo_id = mr.wo_id
      and ir.material_key = mr.material_key
  ),
  stock_by_item as (
    select
      upper(COALESCE(inv_stock_mv.item_name, ''::text)) as item_key,
      sum(COALESCE(inv_stock_mv.qty, 0::numeric)) as stock_qty
    from
      inv_stock_mv
    group by
      (upper(COALESCE(inv_stock_mv.item_name, ''::text)))
  ),
  receipt_by_line as (
    select
      purchase_po_receipts.po_line_id,
      sum(COALESCE(purchase_po_receipts.qty, 0::numeric)) as received_qty,
      max(purchase_po_receipts.receipt_date) as last_receipt_date,
      string_agg(
        distinct purchase_po_receipts.challan_no,
        ', '::text
        order by
          purchase_po_receipts.challan_no
      ) filter (
        where
          COALESCE(purchase_po_receipts.challan_no, ''::text) <> ''::text
      ) as receipt_refs
    from
      purchase_po_receipts
    group by
      purchase_po_receipts.po_line_id
  ),
  open_po_by_item as (
    select
      upper(COALESCE(pol.item_name, ''::text)) as item_key,
      sum(COALESCE(pol.qty, 0::numeric)) as ordered_qty,
      sum(COALESCE(r.received_qty, 0::numeric)) as received_qty,
      sum(
        GREATEST(
          COALESCE(pol.qty, 0::numeric) - COALESCE(r.received_qty, 0::numeric),
          0::numeric
        )
      ) as pending_receipt_qty,
      min(po.order_date) filter (
        where
          GREATEST(
            COALESCE(pol.qty, 0::numeric) - COALESCE(r.received_qty, 0::numeric),
            0::numeric
          ) > 0::numeric
      ) as first_open_po_date,
      string_agg(
        distinct pol.po_no,
        ', '::text
        order by
          pol.po_no
      ) filter (
        where
          GREATEST(
            COALESCE(pol.qty, 0::numeric) - COALESCE(r.received_qty, 0::numeric),
            0::numeric
          ) > 0::numeric
          and COALESCE(pol.po_no, ''::text) <> ''::text
      ) as po_nos,
      max(r.last_receipt_date) as latest_receipt_date,
      string_agg(
        distinct r.receipt_refs,
        ', '::text
        order by
          r.receipt_refs
      ) filter (
        where
          COALESCE(r.receipt_refs, ''::text) <> ''::text
      ) as receipt_refs
    from
      purchase_order_lines pol
      left join purchase_orders po on po.po_no = pol.po_no
      left join receipt_by_line r on r.po_line_id = pol.id
    where
      upper(COALESCE(pol.source_type, ''::text)) <> all (
        array[
          'ARTWORK_PLATE'::text,
          'ARTWORK_DIE'::text,
          'DIRECT_PLATE'::text,
          'DIRECT_DIE'::text
        ]
      )
    group by
      (upper(COALESCE(pol.item_name, ''::text)))
  ),
  open_pr_by_item as (
    select
      upper(
        COALESCE(inv_purchase_requests.item_name, ''::text)
      ) as item_key,
      sum(
        GREATEST(
          COALESCE(inv_purchase_requests.requested_qty, 0::numeric) - COALESCE(inv_purchase_requests.received_qty, 0::numeric),
          0::numeric
        )
      ) as pending_pr_qty,
      string_agg(
        distinct inv_purchase_requests.pr_no,
        ', '::text
        order by
          inv_purchase_requests.pr_no
      ) filter (
        where
          COALESCE(inv_purchase_requests.pr_no, ''::text) <> ''::text
      ) as pr_nos
    from
      inv_purchase_requests
    where
      upper(
        COALESCE(inv_purchase_requests.status, 'OPEN'::text)
      ) <> 'CLOSED'::text
    group by
      (
        upper(
          COALESCE(inv_purchase_requests.item_name, ''::text)
        )
      )
  ),
  line_material_enriched as (
    select
      lm.so_number,
      lm.line_no,
      lm.material_key,
      lm.item_name,
      lm.uom,
      lm.required_qty,
      lm.issued_qty,
      lm.pending_qty,
      COALESCE(st.stock_qty, 0::numeric) as stock_qty,
      COALESCE(po.ordered_qty, 0::numeric) as ordered_qty,
      COALESCE(po.received_qty, 0::numeric) as received_qty,
      COALESCE(po.pending_receipt_qty, 0::numeric) as pending_receipt_qty,
      po.first_open_po_date,
      COALESCE(po.po_nos, ''::text) as po_nos,
      po.latest_receipt_date,
      COALESCE(po.receipt_refs, ''::text) as receipt_refs,
      COALESCE(pr.pending_pr_qty, 0::numeric) as pending_pr_qty,
      COALESCE(pr.pr_nos, ''::text) as pr_nos,
      case
        when COALESCE(lm.pending_qty, 0::numeric) <= 0::numeric then 'ISSUED'::text
        when COALESCE(st.stock_qty, 0::numeric) >= COALESCE(lm.pending_qty, 0::numeric) then 'IN_STOCK'::text
        when COALESCE(po.pending_receipt_qty, 0::numeric) > 0::numeric then 'ORDERED_PENDING_RECEIPT'::text
        when COALESCE(pr.pending_pr_qty, 0::numeric) > 0::numeric then 'PR_PENDING_PO'::text
        else 'SHORTAGE'::text
      end as material_line_status
    from
      line_materials lm
      left join stock_by_item st on st.item_key = upper(COALESCE(lm.item_name, ''::text))
      left join open_po_by_item po on po.item_key = upper(COALESCE(lm.item_name, ''::text))
      left join open_pr_by_item pr on pr.item_key = upper(COALESCE(lm.item_name, ''::text))
  )
select
  so_number,
  line_no,
  case
    when sum(pending_qty) <= 0::numeric then 'ISSUED'::text
    when bool_and(
      material_line_status = any (array['ISSUED'::text, 'IN_STOCK'::text])
    ) then 'IN_STOCK'::text
    when bool_or(
      material_line_status = 'ORDERED_PENDING_RECEIPT'::text
    ) then 'ORDERED_PENDING_RECEIPT'::text
    when bool_or(material_line_status = 'PR_PENDING_PO'::text) then 'PR_PENDING_PO'::text
    else 'SHORTAGE'::text
  end as rm_availability_status,
  round(sum(required_qty), 3) as rm_required_qty,
  round(sum(issued_qty), 3) as rm_issued_qty,
  round(sum(pending_qty), 3) as rm_pending_qty,
  round(sum(stock_qty), 3) as rm_in_stock_qty,
  round(
    sum(
      GREATEST(
        pending_qty - stock_qty - pending_receipt_qty,
        0::numeric
      )
    ),
    3
  ) as rm_shortage_qty,
  round(sum(pending_receipt_qty), 3) as rm_pending_receipt_qty,
  string_agg(
    distinct po_nos,
    ', '::text
    order by
      po_nos
  ) filter (
    where
      COALESCE(po_nos, ''::text) <> ''::text
  ) as rm_po_nos,
  min(first_open_po_date) as rm_first_po_date,
  max(latest_receipt_date) as rm_latest_grn_date,
  string_agg(
    distinct receipt_refs,
    ', '::text
    order by
      receipt_refs
  ) filter (
    where
      COALESCE(receipt_refs, ''::text) <> ''::text
  ) as rm_receipt_refs,
  string_agg(
    distinct pr_nos,
    ', '::text
    order by
      pr_nos
  ) filter (
    where
      COALESCE(pr_nos, ''::text) <> ''::text
  ) as rm_pr_nos,
  string_agg(
    (
      (
        (
          (
            (
              (
                (
                  (
                    (item_name || ': Req '::text) || TRIM(
                      both
                      from
                        to_char(required_qty, 'FM999999990.###'::text)
                    )
                  ) || ', Pending '::text
                ) || TRIM(
                  both
                  from
                    to_char(pending_qty, 'FM999999990.###'::text)
                )
              ) || ', Stock '::text
            ) || TRIM(
              both
              from
                to_char(stock_qty, 'FM999999990.###'::text)
            )
          ) || case
            when COALESCE(po_nos, ''::text) <> ''::text then ', PO '::text || po_nos
            else ''::text
          end
        ) || ' ['::text
      ) || material_line_status
    ) || ']'::text,
    ' | '::text
    order by
      item_name,
      material_key
  ) as rm_detail
from
  line_material_enriched
group by
  so_number,
  line_no;


create view public.v_report_planning_tooling_status as
with
  artwork_ranked as (
    select
      so.so_number,
      a.so_id,
      a.line_no,
      COALESCE(
        NULLIF(a.artwork_no, ''::text),
        (a.so_id::text || '||'::text) || a.line_no::text
      ) as artwork_key,
      a.artwork_no,
      a.plate_status,
      a.die_status,
      row_number() over (
        partition by
          a.so_id,
          a.line_no
        order by
          a.approved_at desc nulls last,
          a.artwork_at desc nulls last,
          a.id desc
      ) as rn
    from
      artworks a
      join sales_orders so on so.id = a.so_id
  ),
  artwork_base as (
    select
      artwork_ranked.so_number,
      artwork_ranked.so_id,
      artwork_ranked.line_no,
      artwork_ranked.artwork_key,
      artwork_ranked.artwork_no,
      artwork_ranked.plate_status,
      artwork_ranked.die_status,
      artwork_ranked.rn
    from
      artwork_ranked
    where
      artwork_ranked.rn = 1
  ),
  manual_proc as (
    select
      purchase_artwork_procurement.artwork_key,
      upper(
        COALESCE(purchase_artwork_procurement.type, ''::text)
      ) as tooling_type,
      max(purchase_artwork_procurement.vendor) filter (
        where
          COALESCE(purchase_artwork_procurement.vendor, ''::text) <> ''::text
      ) as vendor,
      min(purchase_artwork_procurement.ordered_on) as ordered_on,
      max(purchase_artwork_procurement.received_on) as received_on,
      string_agg(
        distinct purchase_artwork_procurement.challan_no,
        ', '::text
        order by
          purchase_artwork_procurement.challan_no
      ) filter (
        where
          COALESCE(purchase_artwork_procurement.challan_no, ''::text) <> ''::text
      ) as challan_nos,
      sum(
        COALESCE(purchase_artwork_procurement.count, 0::numeric)
      ) as manual_qty
    from
      purchase_artwork_procurement
    where
      upper(
        COALESCE(
          purchase_artwork_procurement.status,
          'ACTIVE'::text
        )
      ) <> 'CANCELLED'::text
    group by
      purchase_artwork_procurement.artwork_key,
      (
        upper(
          COALESCE(purchase_artwork_procurement.type, ''::text)
        )
      )
  ),
  receipt_by_line as (
    select
      purchase_po_receipts.po_line_id,
      sum(COALESCE(purchase_po_receipts.qty, 0::numeric)) as received_qty,
      max(purchase_po_receipts.receipt_date) as last_receipt_date,
      string_agg(
        distinct purchase_po_receipts.challan_no,
        ', '::text
        order by
          purchase_po_receipts.challan_no
      ) filter (
        where
          COALESCE(purchase_po_receipts.challan_no, ''::text) <> ''::text
      ) as challan_nos
    from
      purchase_po_receipts
    group by
      purchase_po_receipts.po_line_id
  ),
  po_tooling_lines as (
    select
      pol.source_ref as artwork_key,
      case
        when upper(COALESCE(pol.source_type, ''::text)) = any (array['ARTWORK_DIE'::text, 'DIRECT_DIE'::text]) then 'DIE'::text
        when upper(COALESCE(pol.source_type, ''::text)) = any (
          array['ARTWORK_PLATE'::text, 'DIRECT_PLATE'::text]
        ) then 'PLATE'::text
        else ''::text
      end as tooling_type,
      pol.po_no,
      po.order_date,
      COALESCE(po.vendor_name, ''::text) as vendor_name,
      COALESCE(pol.qty, 0::numeric) as ordered_qty,
      COALESCE(r.received_qty, 0::numeric) as received_qty,
      GREATEST(
        COALESCE(pol.qty, 0::numeric) - COALESCE(r.received_qty, 0::numeric),
        0::numeric
      ) as pending_qty,
      r.last_receipt_date,
      r.challan_nos
    from
      purchase_order_lines pol
      left join purchase_orders po on po.po_no = pol.po_no
      left join receipt_by_line r on r.po_line_id = pol.id
    where
      upper(COALESCE(pol.source_type, ''::text)) = any (
        array[
          'ARTWORK_PLATE'::text,
          'ARTWORK_DIE'::text,
          'DIRECT_PLATE'::text,
          'DIRECT_DIE'::text
        ]
      )
  ),
  po_tooling as (
    select
      po_tooling_lines.artwork_key,
      po_tooling_lines.tooling_type,
      sum(po_tooling_lines.ordered_qty) as ordered_qty,
      sum(po_tooling_lines.received_qty) as received_qty,
      sum(po_tooling_lines.pending_qty) as pending_qty,
      min(po_tooling_lines.order_date) as first_order_date,
      max(po_tooling_lines.last_receipt_date) as last_receipt_date,
      string_agg(
        distinct po_tooling_lines.po_no,
        ', '::text
        order by
          po_tooling_lines.po_no
      ) filter (
        where
          COALESCE(po_tooling_lines.po_no, ''::text) <> ''::text
      ) as po_nos,
      string_agg(
        distinct po_tooling_lines.vendor_name,
        ', '::text
        order by
          po_tooling_lines.vendor_name
      ) filter (
        where
          COALESCE(po_tooling_lines.vendor_name, ''::text) <> ''::text
      ) as vendors,
      string_agg(
        distinct po_tooling_lines.challan_nos,
        ', '::text
        order by
          po_tooling_lines.challan_nos
      ) filter (
        where
          COALESCE(po_tooling_lines.challan_nos, ''::text) <> ''::text
      ) as challan_nos
    from
      po_tooling_lines
    where
      po_tooling_lines.tooling_type <> ''::text
    group by
      po_tooling_lines.artwork_key,
      po_tooling_lines.tooling_type
  )
select
  ab.so_number,
  ab.line_no,
  ab.artwork_key,
  case
    when COALESCE(pt.pending_qty, 0::numeric) <= 0::numeric
    and (
      COALESCE(pt.received_qty, 0::numeric) > 0::numeric
      or mp.received_on is not null
    ) then 'RECEIVED'::text
    when COALESCE(pt.received_qty, 0::numeric) > 0::numeric then 'PARTIAL'::text
    when COALESCE(pt.ordered_qty, 0::numeric) > 0::numeric
    or mp.ordered_on is not null then 'ORDERED'::text
    when upper(COALESCE(ab.plate_status, ''::text)) = any (
      array[
        'NEW'::text,
        'YES'::text,
        'REQUIRED'::text,
        'SELECTED'::text
      ]
    ) then 'PENDING_ORDER'::text
    else COALESCE(NULLIF(ab.plate_status, ''::text), 'NA'::text)
  end as plate_procurement_status,
  COALESCE(pt.po_nos, ''::text) as plate_po_nos,
  COALESCE(pt.first_order_date, mp.ordered_on) as plate_ordered_on,
  COALESCE(pt.last_receipt_date, mp.received_on) as plate_received_on,
  COALESCE(pt.ordered_qty, mp.manual_qty, 0::numeric) as plate_ordered_qty,
  COALESCE(
    pt.received_qty,
    case
      when mp.received_on is not null then mp.manual_qty
      else 0::numeric
    end,
    0::numeric
  ) as plate_received_qty,
  COALESCE(
    pt.pending_qty,
    GREATEST(
      COALESCE(mp.manual_qty, 0::numeric) - case
        when mp.received_on is not null then COALESCE(mp.manual_qty, 0::numeric)
        else 0::numeric
      end,
      0::numeric
    ),
    0::numeric
  ) as plate_pending_qty,
  COALESCE(NULLIF(pt.vendors, ''::text), mp.vendor, ''::text) as plate_vendors,
  COALESCE(
    NULLIF(pt.challan_nos, ''::text),
    mp.challan_nos,
    ''::text
  ) as plate_receipt_refs,
  case
    when COALESCE(dt.pending_qty, 0::numeric) <= 0::numeric
    and (
      COALESCE(dt.received_qty, 0::numeric) > 0::numeric
      or md.received_on is not null
    ) then 'RECEIVED'::text
    when COALESCE(dt.received_qty, 0::numeric) > 0::numeric then 'PARTIAL'::text
    when COALESCE(dt.ordered_qty, 0::numeric) > 0::numeric
    or md.ordered_on is not null then 'ORDERED'::text
    when upper(COALESCE(ab.die_status, ''::text)) = any (
      array[
        'NEW'::text,
        'YES'::text,
        'REQUIRED'::text,
        'SELECTED'::text
      ]
    ) then 'PENDING_ORDER'::text
    else COALESCE(NULLIF(ab.die_status, ''::text), 'NA'::text)
  end as die_procurement_status,
  COALESCE(dt.po_nos, ''::text) as die_po_nos,
  COALESCE(dt.first_order_date, md.ordered_on) as die_ordered_on,
  COALESCE(dt.last_receipt_date, md.received_on) as die_received_on,
  COALESCE(dt.ordered_qty, md.manual_qty, 0::numeric) as die_ordered_qty,
  COALESCE(
    dt.received_qty,
    case
      when md.received_on is not null then md.manual_qty
      else 0::numeric
    end,
    0::numeric
  ) as die_received_qty,
  COALESCE(
    dt.pending_qty,
    GREATEST(
      COALESCE(md.manual_qty, 0::numeric) - case
        when md.received_on is not null then COALESCE(md.manual_qty, 0::numeric)
        else 0::numeric
      end,
      0::numeric
    ),
    0::numeric
  ) as die_pending_qty,
  COALESCE(NULLIF(dt.vendors, ''::text), md.vendor, ''::text) as die_vendors,
  COALESCE(
    NULLIF(dt.challan_nos, ''::text),
    md.challan_nos,
    ''::text
  ) as die_receipt_refs
from
  artwork_base ab
  left join po_tooling pt on pt.artwork_key = ab.artwork_key
  and pt.tooling_type = 'PLATE'::text
  left join po_tooling dt on dt.artwork_key = ab.artwork_key
  and dt.tooling_type = 'DIE'::text
  left join manual_proc mp on mp.artwork_key = ab.artwork_key
  and mp.tooling_type = 'PLATE'::text
  left join manual_proc md on md.artwork_key = ab.artwork_key
  and md.tooling_type = 'DIE'::text;
